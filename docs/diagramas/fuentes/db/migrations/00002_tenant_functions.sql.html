   2 -- The two functions the isolation design rests on (data-model.md §3.2, §3.3, ADR-005).

   4 -- app.current_tenant_id(): the company encoded in the current role, or NULL. Every RLS policy compares
   5 -- tenant_id with it, so it is the ONLY place that knows how a role maps to a company: changing the
   6 -- isolation strategy (ADR-005, alternative B) means changing this function and TxRunner, nothing else.
   7 -- SECURITY INVOKER: it reads current_user, which is the role of the statement being checked. It has no
   8 -- SET search_path clause (that would stop PostgreSQL from inlining it into the policies), so the body
   9 -- names pg_catalog explicitly for every operator, function and type: the caller's search_path cannot
  10 -- change what it answers.
  11 -- +goose StatementBegin
  12 CREATE FUNCTION app.current_tenant_id() RETURNS uuid
  13 LANGUAGE sql STABLE PARALLEL SAFE AS $$
  14   SELECT CASE
  15     WHEN current_user::pg_catalog.text OPERATOR(pg_catalog.~) '^crm_t_[0-9a-f]{32}$'
  16     THEN pg_catalog.substr(current_user::pg_catalog.text, 7)::pg_catalog.uuid
  17   END
  18 $$;
  19 -- +goose StatementEnd
  20 
  21 -- Default privileges revoke EXECUTE from PUBLIC (migration 00001); this one is meant for everybody:
  22 -- the function reveals nothing and every role's policies evaluate it.
  23 GRANT EXECUTE ON FUNCTION app.current_tenant_id() TO PUBLIC;
  24 
  25 -- provisioning.provision_tenant_role(uuid): creates the role of a company. It is created while
  26 -- acting as crm_provisioner (the only role with CREATEROLE), so that role owns it and, being
  27 -- SECURITY DEFINER, is the one that runs it. crm_app cannot create roles by itself (plan §10.4).
  28 SET LOCAL ROLE crm_provisioner;
  29 
  30 -- +goose StatementBegin
  31 CREATE FUNCTION provisioning.provision_tenant_role(p_tenant_id uuid) RETURNS text
  32 LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $$
  33 DECLARE
  34   v_role text := 'crm_t_' || replace(p_tenant_id::text, '-', '');
  35 BEGIN
  36   IF NOT EXISTS (SELECT 1 FROM pg_catalog.pg_roles WHERE rolname = v_role) THEN
  37     EXECUTE format('CREATE ROLE %I NOLOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE NOREPLICATION NOBYPASSRLS', v_role);
  38   END IF;
  39   -- The company role gets the group's privileges (INHERIT) but cannot become the group (no SET).
  40   EXECUTE format('GRANT crm_tenant TO %I WITH INHERIT TRUE, SET FALSE', v_role);
  41   -- The runtime role can become the company role (SET) without inheriting anything from it (INV-02).
  42   EXECUTE format('GRANT %I TO crm_app WITH INHERIT FALSE, SET TRUE', v_role);
  43   RETURN v_role;
  44 END
  45 $$;
  46 -- +goose StatementEnd
  47 
  48 REVOKE ALL ON FUNCTION provisioning.provision_tenant_role(uuid) FROM PUBLIC;
  49 GRANT EXECUTE ON FUNCTION provisioning.provision_tenant_role(uuid) TO crm_signup;
  50 RESET ROLE;
  51 
  52 -- +goose Down
  53 SET LOCAL ROLE crm_provisioner;
  54 DROP FUNCTION provisioning.provision_tenant_role(uuid);
  55 RESET ROLE;
  56 DROP FUNCTION app.current_tenant_id();
