  15 // runMigrate applies the embedded migrations as crm_owner (DATABASE_MIGRATION_URL), never as the
  16 // runtime role crm_app (ADR-004). It is a deployment step: `serve` does not migrate.
  17 func runMigrate(ctx context.Context, args []string, e env) error {
  18 	fs := flag.NewFlagSet("migrate", flag.ContinueOnError)
  19 	fs.SetOutput(e.stderr)
  20 	if err := fs.Parse(args); err != nil {
  21 		return usagef("migrate: %v", err)
  22 	}
  23 	if fs.NArg() != 1 {
  24 		return usagef("migrate: expected exactly one of up, down, status")
  25 	}
  26 	direction := fs.Arg(0)
  27 	if direction != "up" && direction != "down" && direction != "status" {
  28 		return usagef("migrate: unknown direction %q (want up, down or status)", direction)
  29 	}
  30 
  31 	// Read directly: only this command needs the owner URL, so it is not part of the serve
  32 	// configuration. The value is a secret and is never printed.
  33 	url := e.getenv("DATABASE_MIGRATION_URL")
  34 	if url == "" {
  35 		return errors.New("migrate: DATABASE_MIGRATION_URL is required")
  36 	}
  37 	db, err := sql.Open("pgx", url)
  38 	if err != nil {
  39 		return errors.New("migrate: DATABASE_MIGRATION_URL is not a valid connection string")
  40 	}
  41 	defer func() { _ = db.Close() }() // the work is done by now: a close error changes nothing
  42 
  43 	provider, err := goose.NewProvider(goose.DialectPostgres, db, migrations.FS)
  44 	if err != nil {
  45 		return fmt.Errorf("migrate: creating provider: %w", err)
  46 	}
  47 	switch direction {
  48 	case "up":
  49 		results, err := provider.Up(ctx)
  50 		if err != nil {
  51 			return fmt.Errorf("migrate up: %w", err)
  52 		}
  53 		if len(results) == 0 {
  54 			fmt.Fprintln(e.stdout, "no migrations to apply")
  55 		}
  56 		for _, r := range results {
  57 			fmt.Fprintf(e.stdout, "applied %s (%s)\n", r.Source.Path, r.Duration)
  58 		}
  59 	case "down":
  60 		r, err := provider.Down(ctx)
  61 		if err != nil {
  62 			return fmt.Errorf("migrate down: %w", err)
  63 		}
  64 		fmt.Fprintf(e.stdout, "rolled back %s (%s)\n", r.Source.Path, r.Duration)
  65 	case "status":
  66 		statuses, err := provider.Status(ctx)
  67 		if err != nil {
  68 			return fmt.Errorf("migrate status: %w", err)
  69 		}
  70 		for _, s := range statuses {
  71 			fmt.Fprintf(e.stdout, "%-10s %s\n", s.State, s.Source.Path)
  72 		}
  73 	}
  74 	return nil
  75 }
