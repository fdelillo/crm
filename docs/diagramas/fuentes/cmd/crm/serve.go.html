  28 // runServe starts the HTTP server: HTTP by default, HTTPS with the local certificate when
  29 // TLS_CERT_FILE and TLS_KEY_FILE are set in local mode (DD-24). It returns nil after a graceful
  30 // shutdown (ctx cancelled by SIGTERM).
  31 func runServe(ctx context.Context, args []string, e env) error {
  32 	fs := flag.NewFlagSet("serve", flag.ContinueOnError)
  33 	fs.SetOutput(e.stderr)
  34 	if err := fs.Parse(args); err != nil {
  35 		return usagef("serve: %v", err)
  36 	}
  37 	if fs.NArg() > 0 {
  38 		return usagef("serve: unexpected argument %q", fs.Arg(0))
  39 	}
  40 
  41 	cfg, err := config.Load(e.getenv)
  42 	if err != nil {
  43 		return err
  44 	}
  45 	expected, err := migrations.ExpectedVersion(migrations.FS)
  46 	if err != nil {
  47 		return err
  48 	}
  49 	logger := slog.New(slog.NewJSONHandler(e.stdout, nil))
  50 	// Behind a proxy that is not listed here every client looks like the proxy (one shared rate limit
  51 	// bucket, useless IPs in logs and audit): the first line says what is trusted (DD-32).
  52 	trusted := make([]string, 0, len(cfg.TrustedProxies))
  53 	for _, p := range cfg.TrustedProxies {
  54 		trusted = append(trusted, p.String())
  55 	}
  56 	logger.Info("starting", "local_mode", cfg.IsLocal(), "trusted_proxies", trusted)
  57 	pool, err := pgxpool.New(ctx, cfg.DatabaseURL)
  58 	if err != nil {
  59 		return fmt.Errorf("DATABASE_URL: %w", err)
  60 	}
  61 	defer pool.Close()
  62 	runner := db.NewTxRunner(pool, db.WithLogger(logger))
  63 	app.LogServerVersion(ctx, runner, logger)
  64 	c := clock.Real{}
  65 	hasher := password.NewHasher(4)
  66 	recorder := audit.NewRecorder()
  67 	users := identity.NewService(runner, outbox.NewEnqueuer(c), c, cfg.SessionIdle, cfg.SessionAbsolute,
  68 		identity.WithAuthentication(hasher, recorder, cfg.AuthHMACKey, logger))
  69 	storage, err := objectstore.NewS3(objectstore.S3Config{Endpoint: cfg.S3Endpoint, Bucket: cfg.S3Bucket,
  70 		AccessKey: cfg.S3AccessKey, SecretKey: cfg.S3SecretKey, UseSSL: cfg.S3UseSSL})
  71 	if err != nil {
  72 		return err
  73 	}
  74 	companies := tenant.NewService(runner, users, industrytemplate.NoopSeeder{}, hasher, recorder, logger, tenant.WithObjectStorage(storage))
  75 	smtp, err := mailer.NewSMTP(mailer.SMTPConfig{Host: cfg.SMTPHost, Port: cfg.SMTPPort,
  76 		Username: cfg.SMTPUsername, Password: cfg.SMTPPassword, From: cfg.SMTPFrom})
  77 	if err != nil {
  78 		return err
  79 	}
  80 	emailHandler, err := emails.NewHandler(smtp, cfg.AppBaseURL.String(),
  81 		emails.LinkPaths{Reset: cfg.AppLinkReset, Verify: cfg.AppLinkVerify, Invitation: cfg.AppLinkInvitation})
  82 	if err != nil {
  83 		return err
  84 	}
  85 	dispatcher := outbox.NewDispatcher(runner, emailHandler, c, logger, app.PeriodicTasks(runner, c, logger)...)
  86 
  87 	api := app.BuildAPIRouter(users, companies, c, logger)
  88 	root := app.NewRootHandler(app.RootDeps{
  89 		API:       api,
  90 		Liveness:  app.LivenessHandler(),
  91 		Readiness: app.ReadinessHandler(runner, expected, logger),
  92 		// Until the web package embeds the built SPA (T-F008) the interface answers 503.
  93 		SPA: app.SPAUnavailableHandler(),
  94 	}, app.NewCommonMiddleware(logger, cfg.IsLocal(), cfg.TrustedProxies))
  95 
  96 	// A bad certificate pair fails here, before anything listens.
  97 	srv, err := app.NewServer(cfg, root)
  98 	if err != nil {
  99 		return err
 100 	}
 101 	serverCtx, stopServers := context.WithCancel(ctx)
 102 	defer stopServers()
 103 	ln, err := e.listen(serverCtx, "tcp", cfg.HTTPAddr)
 104 	if err != nil {
 105 		return fmt.Errorf("listening on HTTP_ADDR %s: %w", cfg.HTTPAddr, err)
 106 	}
 107 	// Serve owns each listener. Starting this goroutine here also releases HTTP_ADDR
 108 	// if the second bind fails, without retaining or reusing the listener.
 109 	apiDone := make(chan error, 1)
 110 	go func() {
 111 		apiDone <- app.Serve(serverCtx, srv, ln, logger)
 112 		stopServers()
 113 	}()
 114 	metricsSrv := app.NewMetricsServer(cfg)
 115 	metricsLn, err := e.listen(serverCtx, "tcp", cfg.MetricsAddr)
 116 	if err != nil {
 117 		stopServers()
 118 		return errors.Join(fmt.Errorf("listening on METRICS_ADDR %s: %w", cfg.MetricsAddr, err), <-apiDone)
 119 	}
 120 	metricsDone := make(chan error, 1)
 121 	go func() {
 122 		metricsDone <- app.Serve(serverCtx, metricsSrv, metricsLn, logger.With("interface", "metrics"))
 123 		stopServers()
 124 	}()
 125 	workerCtx, stopWorker := context.WithCancel(serverCtx)
 126 	defer stopWorker()
 127 	workerDone := make(chan error, 1)
 128 	go func() { workerDone <- dispatcher.Run(workerCtx) }()
 129 	serveErr := <-apiDone
 130 	metricsErr := <-metricsDone
 131 	if serveErr == nil {
 132 		serveErr = metricsErr
 133 	}
 134 	stopWorker()
 135 	select {
 136 	case workerErr := <-workerDone:
 137 		if serveErr == nil && workerErr != nil {
 138 			return workerErr
 139 		}
 140 	case <-time.After(app.ShutdownTimeout):
 141 		return fmt.Errorf("outbox worker did not stop within %s", app.ShutdownTimeout)
 142 	}
 143 	return serveErr
 144 }
