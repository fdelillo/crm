  29 // globalSignupQuota is the constant-key quota of DD-49. A request that does not reach next was
  30 // refused by the quota (its key cannot fail), so the metric and the event are emitted here; the
  31 // 429 itself is written by ratelimit.MiddlewareBy. The event carries the address and no email.
  32 func globalSignupQuota(limiter *ratelimit.Limiter, logger *slog.Logger) func(http.Handler) http.Handler {
  33 	by := ratelimit.MiddlewareBy(limiter, func(*http.Request) (string, bool) { return "signup", true })
  34 	return func(next http.Handler) http.Handler {
  35 		return http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
  36 			reached := false
  37 			by(http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
  38 				reached = true
  39 				next.ServeHTTP(w, req)
  40 			})).ServeHTTP(w, req)
  41 			if !reached {
  42 				signupGlobalLimitedCount.Add(1)
  43 				logger.WarnContext(req.Context(), "signup global quota exhausted", "event", "signup_global_limited",
  44 					"ip", httpx.ClientIPFrom(req.Context()).String())
  45 			}
  46 		})
  47 	}
  48 }

  50 // RegisterRoutes registers the public signup endpoint. Both limiters spend a token before decoding
  51 // or validating the body, so rejected attempts also consume the budget. The IP quota (DD-9) runs
  52 // first, so a refusal by IP does not spend the global quota (DD-49).
  53 func RegisterRoutes(r chi.Router, service *Service, ipLimiter, globalLimiter *ratelimit.Limiter, logger *slog.Logger) {
  54 	r.With(ratelimit.Middleware(ipLimiter), globalSignupQuota(globalLimiter, logger)).Post("/api/v1/auth/signup", func(w http.ResponseWriter, req *http.Request) {
  55 		var input Signup
  56 		if err := httpx.DecodeJSON(w, req, &input); err != nil {
  57 			return
  58 		}
  59 		meta := identity.RequestMeta{IP: httpx.ClientIPFrom(req.Context()), UserAgent: req.UserAgent(),
  60 			RequestID: httpx.RequestIDFrom(req.Context())}
  61 		result, err := service.Register(req.Context(), input, meta)
  62 		if err != nil {
  63 			var fields FieldErrors
  64 			if errors.As(err, &fields) {
  65 				httpx.ValidationError(w, req, fields)
  66 				return
  67 			}
  68 			if errors.Is(err, identity.ErrEmailTaken) {
  69 				signupEmailExistsCount.Add(1)
  70 				logger.WarnContext(req.Context(), "signup email exists", "security_event", "signup_email_exists", "ip", meta.IP.String())
  71 				httpx.WriteProblem(w, req, httpx.CodeEmailAlreadyRegistered,
  72 					httpx.WithSuggestedAction(httpx.SuggestedPasswordReset))
  73 				return
  74 			}
  75 			var pgErr *pgconn.PgError
  76 			if errors.Is(err, db.ErrUnavailable) && errors.As(err, &pgErr) && pgErr.Code == "55P03" {
  77 				signupLockTimeoutCount.Add(1)
  78 				logger.WarnContext(req.Context(), "signup lock timeout", "event", "signup_lock_timeout")
  79 				httpx.WriteProblem(w, req, httpx.CodeServiceUnavailable, httpx.RetryAfter(SignupLockTimeout))
  80 				return
  81 			}
  82 			httpx.WriteDBError(w, req, err, logger)
  83 			return
  84 		}
  85 		identity.SetSessionCookie(w, result.Session)
  86 		body, err := json.Marshal(struct {
  87 			User        identity.CurrentUser `json:"user"`
  88 			Tenant      Summary              `json:"tenant"`
  89 			Permissions []authz.Permission   `json:"permissions"`
  90 		}{User: result.User, Tenant: result.Tenant, Permissions: authz.Permissions(result.User.Role)})
  91 		if err != nil {
  92 			panic(err)
  93 		}
  94 		w.Header().Set("Content-Type", "application/json")
  95 		w.WriteHeader(http.StatusCreated)
  96 		_, _ = w.Write(body)
  97 	})
  98 }

 100 // RegisterCompanyRoutes expects Authenticate outside this group. Permission checks
 101 // precede decoding multipart/JSON or looking up resources (INV-12).
 102 func RegisterCompanyRoutes(r chi.Router, service *Service, logger *slog.Logger) {
 103 	r.Get("/api/v1/tenant", func(w http.ResponseWriter, req *http.Request) {
 104 		p, _ := authz.PrincipalFrom(req.Context())
 105 		out, err := service.Get(req.Context(), p)
 106 		if writeCompanyError(w, req, err, logger) {
 107 			return
 108 		}
 109 		writeCompanyJSON(w, out)
 110 	})
 111 	r.Get("/api/v1/tenant/logo", func(w http.ResponseWriter, req *http.Request) {
 112 		p, _ := authz.PrincipalFrom(req.Context())
 113 		out, err := service.GetLogo(req.Context(), p, req.Header.Get("If-None-Match"))
 114 		if writeCompanyError(w, req, err, logger) {
 115 			return
 116 		}
 117 		w.Header().Set("Cache-Control", "private, no-cache")
 118 		w.Header().Set("ETag", out.ETag)
 119 		w.Header().Set("X-Content-Type-Options", "nosniff")
 120 		w.Header().Set("Content-Type", out.ContentType)
 121 		if out.NotModified {
 122 			w.WriteHeader(http.StatusNotModified)
 123 			return
 124 		}
 125 		defer func() { _ = out.Body.Close() }()
 126 		w.Header().Set("Content-Length", strconv.FormatInt(out.Size, 10))
 127 		w.WriteHeader(http.StatusOK)
 128 		if _, err := io.Copy(w, out.Body); err != nil {
 129 			if req.Context().Err() != nil {
 130 				// The headers already went out; mark the existing request log as 499.
 131 				httpx.WriteDBError(w, req, db.ErrCanceled, logger)
 132 			} else {
 133 				logger.WarnContext(req.Context(), "logo response stream failed", "event", "logo_stream_failed", "error", err)
 134 			}
 135 		}
 136 	})
 137 	r.Group(func(r chi.Router) {
 138 		r.Use(authz.RequirePermission(authz.SettingsManage))
 139 		r.Patch("/api/v1/tenant", func(w http.ResponseWriter, req *http.Request) {
 140 			var input Update
 141 			if err := httpx.DecodeJSON(w, req, &input); err != nil {
 142 				return
 143 			}
 144 			p, _ := authz.PrincipalFrom(req.Context())
 145 			out, err := service.Update(req.Context(), p, input, companyRequestMeta(req))
 146 			if writeCompanyError(w, req, err, logger) {
 147 				return
 148 			}
 149 			writeCompanyJSON(w, out)
 150 		})
 151 		r.Put("/api/v1/tenant/logo", func(w http.ResponseWriter, req *http.Request) {
 152 			data, err := readLogoMultipart(w, req)
 153 			if writeCompanyError(w, req, err, logger) {
 154 				return
 155 			}
 156 			p, _ := authz.PrincipalFrom(req.Context())
 157 			out, err := service.SetLogo(req.Context(), p, data, companyRequestMeta(req))
 158 			if writeCompanyError(w, req, err, logger) {
 159 				return
 160 			}
 161 			writeCompanyJSON(w, out)
 162 		})
 163 		r.Delete("/api/v1/tenant/logo", func(w http.ResponseWriter, req *http.Request) {
 164 			p, _ := authz.PrincipalFrom(req.Context())
 165 			if writeCompanyError(w, req, service.RemoveLogo(req.Context(), p, companyRequestMeta(req)), logger) {
 166 				return
 167 			}
 168 			w.WriteHeader(http.StatusNoContent)
 169 		})
 170 	})
 171 }

 175 func readLogoMultipart(w http.ResponseWriter, req *http.Request) ([]byte, error) {
 176 	mediaType, _, err := mime.ParseMediaType(req.Header.Get("Content-Type"))
 177 	if err != nil || mediaType != "multipart/form-data" {
 178 		return nil, ErrLogoUnsupportedType
 179 	}
 180 	req.Body = http.MaxBytesReader(w, req.Body, LogoMaxBodyBytes)
 181 	reader, err := req.MultipartReader()
 182 	if err != nil {
 183 		return nil, errMalformedMultipart
 184 	}
 185 	part, err := reader.NextPart()
 186 	if errors.Is(err, io.EOF) {
 187 		return nil, FieldErrors{"file": "required"}
 188 	}
 189 	if err != nil {
 190 		return nil, multipartError(err)
 191 	}
 192 	if part.FormName() != "file" {
 193 		return nil, errMalformedMultipart
 194 	}
 195 	// Do not Close/drain an oversize part: Close would read its unread remainder.
 196 	data, err := io.ReadAll(io.LimitReader(part, LogoMaxBytes+1))
 197 	if len(data) > LogoMaxBytes {
 198 		return nil, ErrLogoTooLarge
 199 	}
 200 	if err != nil {
 201 		return nil, multipartError(err)
 202 	}
 203 	if _, err = reader.NextPart(); !errors.Is(err, io.EOF) {
 204 		if err == nil {
 205 			return nil, errMalformedMultipart
 206 		}
 207 		return nil, multipartError(err)
 208 	}
 209 	// NextPart ignores an epilogue. Still enforce the limit on the complete body.
 210 	if _, err = io.Copy(io.Discard, req.Body); err != nil {
 211 		return nil, multipartError(err)
 212 	}
 213 	if len(data) == 0 {
 214 		return nil, FieldErrors{"file": "required"}
 215 	}
 216 	return data, nil
 217 }
 218 func multipartError(err error) error {
 219 	var max *http.MaxBytesError
 220 	if errors.As(err, &max) {
 221 		return ErrLogoTooLarge
 222 	}
 223 	return errMalformedMultipart
 224 }

 228 func writeCompanyJSON(w http.ResponseWriter, value Tenant) {
 229 	body, err := json.Marshal(value)
 230 	if err != nil {
 231 		panic(err)
 232 	}
 233 	w.Header().Set("Content-Type", "application/json")
 234 	w.WriteHeader(http.StatusOK)
 235 	_, _ = w.Write(body)

 237 func writeCompanyError(w http.ResponseWriter, req *http.Request, err error, logger *slog.Logger) bool {
 238 	if err == nil {
 239 		return false
 240 	}
 241 	var fields FieldErrors
 242 	switch {
 243 	case errors.As(err, &fields):
 244 		httpx.ValidationError(w, req, fields)
 245 	case errors.Is(err, ErrMalformedUpdate), errors.Is(err, errMalformedMultipart):
 246 		httpx.WriteProblem(w, req, httpx.CodeMalformedRequest)
 247 	case errors.Is(err, ErrLogoTooLarge):
 248 		httpx.WriteProblem(w, req, httpx.CodePayloadTooLarge)
 249 	case errors.Is(err, ErrLogoUnsupportedType):
 250 		httpx.WriteProblem(w, req, httpx.CodeUnsupportedMediaType)
 251 	case errors.Is(err, ErrLogoInvalidImage):
 252 		httpx.ValidationError(w, req, map[string]string{"file": "invalid_value"})
 253 	case errors.Is(err, authz.ErrForbidden):
 254 		httpx.WriteProblem(w, req, httpx.CodeForbidden)
 255 	case errors.Is(err, ErrLogoNotFound), errors.Is(err, objectstore.ErrNotFound):
 256 		httpx.WriteProblem(w, req, httpx.CodeNotFound)
 257 	case errors.Is(err, objectstore.ErrUnavailable):
 258 		if req.Context().Err() != nil {
 259 			httpx.WriteDBError(w, req, db.MapError(req.Context().Err()), logger)
 260 		} else {
 261 			logger.WarnContext(req.Context(), "logo storage unavailable", "event", "logo_storage_unavailable", "error", err)
 262 			httpx.WriteProblem(w, req, httpx.CodeServiceUnavailable)
 263 		}
 264 	default:
 265 		httpx.WriteDBError(w, req, db.MapError(err), logger)
 266 	}
 267 	return true
