  35 func (r *uncertainLogoRunner) InTenantTx(ctx context.Context, id uuid.UUID, fn func(context.Context, db.Tx) error) error {
  36 	r.calls++
  37 	if r.calls > 1 {
  38 		if r.mode == "unknown" {
  39 			return db.ErrUnavailable
  40 		}
  41 		return r.TxRunner.InTenantTx(ctx, id, fn)
  42 	}
  43 	if r.mode == "before_function" {
  44 		r.cancel()
  45 		return db.ErrUnavailable
  46 	}
  47 	err := r.TxRunner.InTenantTx(ctx, id, func(ctx context.Context, tx db.Tx) error {
  48 		if err := fn(ctx, tx); err != nil {
  49 			return err
  50 		}
  51 		if r.mode == "rolled_back" {
  52 			return errForcedRollback
  53 		}
  54 		return nil
  55 	})
  56 	if err != nil && !errors.Is(err, errForcedRollback) {
  57 		return err
  58 	}
  59 	r.cancel()
  60 	if r.mode == "confirmed" {
  61 		return nil
  62 	}
  63 	return errUncertainCommit
  64 }

 121 type flyingLogoRunner struct {
 122 	db.TxRunner
 123 	pid      chan int32
 124 	release  chan struct{}
 125 	finished chan error
 126 	calls    int
 127 }

 129 func (r *flyingLogoRunner) InTenantTx(ctx context.Context, id uuid.UUID, fn func(context.Context, db.Tx) error) error {
 130 	r.calls++
 131 	if r.calls > 1 {
 132 		return r.TxRunner.InTenantTx(ctx, id, fn)
 133 	}
 134 	ready := make(chan error, 1)
 135 	go func() {
 136 		err := r.TxRunner.InTenantTx(ctx, id, func(ctx context.Context, tx db.Tx) error {
 137 			if err := fn(ctx, tx); err != nil {
 138 				ready <- err
 139 				return err
 140 			}
 141 			var pid int32
 142 			if err := tx.QueryRow(ctx, `SELECT pg_backend_pid()`).Scan(&pid); err != nil {
 143 				ready <- err
 144 				return err
 145 			}
 146 			r.pid <- pid
 147 			ready <- nil
 148 			<-r.release
 149 			return nil
 150 		})
 151 		r.finished <- err
 152 	}()
 153 	if err := <-ready; err != nil {
 154 		return err
 155 	}
 156 	return errUncertainCommit
 157 }
 158 func TestLogoCommitInFlightWaitsForLock(t *testing.T) {
 159 	_, _, real := newRegistrationServices(t)
 160 	f := newLogoStorage()
 161 	var logs bytes.Buffer
 162 	svc := logoService(t, real, audit.NewRecorder(), f, &logs)
 163 	p := companyAdmin(t, svc)
 164 	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
 165 	defer cancel()
 166 	r := &flyingLogoRunner{TxRunner: real, pid: make(chan int32, 1), release: make(chan struct{}), finished: make(chan error, 1)}
 167 	svc = logoService(t, r, audit.NewRecorder(), f, &logs)
 168 	done := make(chan error, 1)
 169 	go func() { _, err := svc.SetLogo(ctx, p, pngLogo(t, 10, 10, 0), identity.RequestMeta{}); done <- err }()
 170 	var once sync.Once
 171 	release := func() { once.Do(func() { close(r.release) }) }
 172 	defer release()
 173 	var pid int32
 174 	select {
 175 	case pid = <-r.pid:
 176 	case err := <-done:
 177 		t.Fatalf("DD-42: SetLogo returned before publishing the in-flight transaction PID: %v", err)
 178 	case <-ctx.Done():
 179 		t.Fatalf("DD-42: in-flight transaction did not publish its PID before the context deadline: %v", ctx.Err())
 180 	}
 181 	ticker := time.NewTicker(10 * time.Millisecond)
 182 	defer ticker.Stop()
 183 	var operationErr, errorBeforeLock error
 184 	completed := false
 185 	waiting := false
 186 	for !waiting && errorBeforeLock == nil {
 187 		select {
 188 		case operationErr = <-done:
 189 			completed = true
 190 			errorBeforeLock = fmt.Errorf("DD-42: SetLogo returned before reconciliation locked the in-flight row: %w", errors.Join(operationErr, errors.New("no waiter")))
 191 		case <-ticker.C:
 192 			if err := pgtest.AppPool(t).QueryRow(ctx, `SELECT EXISTS(SELECT pid FROM pg_stat_activity WHERE $1=ANY(pg_blocking_pids(pid)))`, pid).Scan(&waiting); err != nil {
 193 				errorBeforeLock = err
 194 			}
 195 		case <-ctx.Done():
 196 			errorBeforeLock = fmt.Errorf("DD-42: reconciliation never waited for in-flight COMMIT: %w", ctx.Err())
 197 		}
 198 	}
 199 	release()
 200 	commitErr := <-r.finished
 201 	if !completed {
 202 		operationErr = <-done
 203 	}
 204 	if errorBeforeLock != nil {
 205 		t.Fatal(errorBeforeLock)
 206 	}
 207 	if commitErr != nil || operationErr != nil {
 208 		t.Fatalf("commit=%v set=%v", commitErr, operationErr)
 209 	}
 210 	key, _ := companyRow(t, real, p)
 211 	if key == nil {
 212 		t.Fatal("no committed key")
 213 	}
 214 	if _, ok := f.objects[*key]; !ok {
 215 		t.Fatal("DD-42: committed object was deleted")
 216 	}
 217 	if !strings.Contains(logs.String(), "event=logo_commit_uncertain") || !strings.Contains(logs.String(), "outcome=committed") {
 218 		t.Fatal(logs.String())
 219 	}
