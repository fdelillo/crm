  48 func TestRegisterIsAtomicAndAuditsOnlyCatalogData(t *testing.T) {
  49 	svc, ident, runner := newRegistrationServices(t)
  50 	email := "  ANA-" + uuid.NewString() + "@example.com  "
  51 	ua := strings.Repeat("a", 600) + "\xff\x00"
  52 	meta := identity.RequestMeta{IP: netip.MustParseAddr("192.0.2.42"), UserAgent: ua}
  53 	input := tenant.Signup{Name: "Ana", Email: email, Password: "a safe password 123",
  54 		CompanyName: "Empresa", BaseCurrency: "ARS", IndustryTemplateCode: "generic"}
  55 	var result tenant.Registration
  56 	var err error
  57 	var requestID string
  58 	httpx.RequestID(http.HandlerFunc(func(_ http.ResponseWriter, r *http.Request) {
  59 		requestID = httpx.RequestIDFrom(r.Context())
  60 		result, err = svc.Register(r.Context(), input, meta)
  61 	})).ServeHTTP(httptest.NewRecorder(), httptest.NewRequest(http.MethodPost, "/", nil))
  62 	if err != nil {
  63 		t.Fatal(err)
  64 	}
  65 	if result.User.Email != strings.ToLower(strings.TrimSpace(email)) || result.Session.RawToken == "" || result.User.Role != "admin" {
  66 		t.Fatalf("unexpected registration result: %+v", result.User)
  67 	}
  68 	var dataText, auditAction, auditTargetType, auditIP, auditRequestID string
  69 	var auditActor, auditTarget uuid.UUID
  70 	var auditUA, sessionUA *string
  71 	var outboxPayload []byte
  72 	var storedSessionHash []byte
  73 	var storedEmail, passwordHash string
  74 	var tokenCount, messageCount, auditCount int
  75 	err = runner.InTenantTx(context.Background(), result.Tenant.ID, func(ctx context.Context, tx db.Tx) error {
  76 		if err := tx.QueryRow(ctx, `SELECT email, password_hash FROM app.users WHERE tenant_id=$1 AND id=$2`, result.Tenant.ID, result.User.ID).Scan(&storedEmail, &passwordHash); err != nil {
  77 			return err
  78 		}
  79 		if err := tx.QueryRow(ctx, `SELECT token_hash, user_agent FROM app.sessions WHERE tenant_id=$1 AND id=$2`, result.Tenant.ID, result.Session.Principal.SessionID).Scan(&storedSessionHash, &sessionUA); err != nil {
  80 			return err
  81 		}
  82 		if err := tx.QueryRow(ctx, `SELECT count(*) FROM app.user_tokens WHERE tenant_id=$1 AND user_id=$2 AND purpose='email_verification'`, result.Tenant.ID, result.User.ID).Scan(&tokenCount); err != nil {
  83 			return err
  84 		}
  85 		if err := tx.QueryRow(ctx, `SELECT count(*) FROM app.outbox_messages WHERE tenant_id=$1 AND template='email_verification'`, result.Tenant.ID).Scan(&messageCount); err != nil {
  86 			return err
  87 		}
  88 		if err := tx.QueryRow(ctx, `SELECT payload FROM app.outbox_messages WHERE tenant_id=$1 AND template='email_verification'`, result.Tenant.ID).Scan(&outboxPayload); err != nil {
  89 			return err
  90 		}
  91 		if err := tx.QueryRow(ctx, `SELECT count(*) FROM app.audit_log WHERE tenant_id=$1`, result.Tenant.ID).Scan(&auditCount); err != nil {
  92 			return err
  93 		}
  94 		return tx.QueryRow(ctx, `SELECT action, actor_user_id, target_type, target_id, data::text,
  95 			host(ip), user_agent, request_id FROM app.audit_log WHERE tenant_id=$1`, result.Tenant.ID).
  96 			Scan(&auditAction, &auditActor, &auditTargetType, &auditTarget, &dataText,
  97 				&auditIP, &auditUA, &auditRequestID)
  98 	})
  99 	if err != nil {
 100 		t.Fatal(err)
 101 	}
 102 	if storedEmail != result.User.Email || !strings.HasPrefix(passwordHash, "$argon2id$") || string(storedSessionHash) == result.Session.RawToken || string(storedSessionHash) != string(securetoken.Hash(result.Session.RawToken)) {
 103 		t.Fatal("stored credentials do not match the required hash/normalization")
 104 	}

 106 		t.Fatalf("token=%d message=%d audit=%d", tokenCount, messageCount, auditCount)
 107 	}

 111 	}

 113 		t.Fatalf("audit data=%v", data)
 114 	}
 115 	if auditAction != "tenant.registered" || auditActor != result.User.ID || auditTargetType != "tenant" ||
 116 		auditTarget != result.Tenant.ID || auditIP != meta.IP.String() || auditRequestID != requestID {
 117 		t.Fatalf("audit metadata: action=%s actor=%s target=%s/%s ip=%s request=%s",
 118 			auditAction, auditActor, auditTargetType, auditTarget, auditIP, auditRequestID)
 119 	}
 120 	var payload struct {
 121 		Token string `json:"token"`
 122 	}
 123 	if err := json.Unmarshal(outboxPayload, &payload); err != nil || payload.Token == "" {
 124 		t.Fatalf("verification payload missing token: %v", err)
 125 	}
 126 	for _, secret := range []string{result.Session.RawToken, payload.Token, input.Password, result.User.Email} {

 128 			t.Fatal("audit contains registration secret or email")
 129 		}
 130 	}
 131 	wantUA := httpx.NormalizeUserAgent(ua)
 132 	if auditUA == nil || sessionUA == nil || *auditUA != wantUA || *sessionUA != wantUA {
 133 		t.Fatalf("user agent not normalized identically")
 134 	}

 136 	if err != nil || principal.UserID != result.User.ID {
 137 		t.Fatalf("resolve session: %+v %v", principal, err)
 138 	}
 139 	_, err = svc.Register(context.Background(), input, meta)
 140 	if !errors.Is(err, identity.ErrEmailTaken) {
 141 		t.Fatalf("duplicate registration error=%v", err)
 142 	}

 223 func TestRegisterConcurrentDistinctAndDuplicateEmails(t *testing.T) {
 224 	svc, _, _ := newRegistrationServices(t)
 225 	ctx := context.Background()
 226 	var wg sync.WaitGroup
 227 	errorsByAttempt := make(chan error, 10)
 228 	for range 10 {
 229 		wg.Add(1)
 230 		go func() {
 231 			defer wg.Done()
 232 			_, err := svc.Register(ctx, signupWithEmail(uuid.NewString()+"@example.com"), identity.RequestMeta{})
 233 			errorsByAttempt <- err
 234 		}()
 235 	}
 236 	wg.Wait()
 237 	close(errorsByAttempt)
 238 	for err := range errorsByAttempt {
 239 		if err != nil {
 240 			t.Fatalf("distinct concurrent registration: %v", err)
 241 		}
 242 	}
 243 	email := uuid.NewString() + "@example.com"
 244 	errorsByAttempt = make(chan error, 2)
 245 	for range 2 {
 246 		wg.Add(1)
 247 		go func() {
 248 			defer wg.Done()
 249 			_, err := svc.Register(ctx, signupWithEmail(email), identity.RequestMeta{})
 250 			errorsByAttempt <- err
 251 		}()
 252 	}
 253 	wg.Wait()
 254 	close(errorsByAttempt)
 255 	var ok, taken int
 256 	for err := range errorsByAttempt {
 257 		switch {
 258 		case err == nil:
 259 			ok++
 260 		case errors.Is(err, identity.ErrEmailTaken):
 261 			taken++
 262 		default:
 263 			t.Fatalf("same-email concurrent registration: %v", err)
 264 		}
 265 	}
 266 	if ok != 1 || taken != 1 {
 267 		t.Fatalf("same-email outcomes: success=%d taken=%d", ok, taken)
 268 	}
 269 }

 271 func TestRegisterLockTimeoutIsUnavailableAndRetryWorks(t *testing.T) {
 272 	svc, _, _ := newRegistrationServices(t)
 273 	ctx := context.Background()
 274 	super := pgtest.SuperuserPool(t)
 275 	role := "crm_test_" + strings.ReplaceAll(uuid.NewString(), "-", "")
 276 	if _, err := super.Exec(ctx, `CREATE ROLE `+pgx.Identifier{role}.Sanitize()); err != nil {
 277 		t.Fatal(err)
 278 	}
 279 	t.Cleanup(func() { _, _ = super.Exec(context.Background(), `DROP ROLE `+pgx.Identifier{role}.Sanitize()) })
 280 	lock, err := super.Begin(ctx)
 281 	if err != nil {
 282 		t.Fatal(err)
 283 	}
 284 	defer func() { _ = lock.Rollback(context.Background()) }()
 285 	if _, err := lock.Exec(ctx, `GRANT crm_tenant TO `+pgx.Identifier{role}.Sanitize()); err != nil {
 286 		t.Fatal(err)
 287 	}
 288 	input := signupWithEmail(uuid.NewString() + "@example.com")
 289 	start := time.Now()
 290 	_, err = svc.Register(ctx, input, identity.RequestMeta{})
 291 	elapsed := time.Since(start)
 292 	if !errors.Is(err, db.ErrUnavailable) || elapsed < 2*time.Second || elapsed >= 4*time.Second {
 293 		t.Fatalf("lock timeout: err=%v elapsed=%s", err, elapsed)
 294 	}
 295 	if err := lock.Rollback(ctx); err != nil {
 296 		t.Fatal(err)
 297 	}
 298 	if _, err := svc.Register(ctx, input, identity.RequestMeta{}); err != nil {
 299 		t.Fatalf("retry after lock release: %v", err)
 300 	}
 301 }
