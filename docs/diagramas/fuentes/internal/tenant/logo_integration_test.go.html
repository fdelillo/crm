  42 	objects             map[string]logoObject

  49 func (f *logoStorage) Put(_ context.Context, key string, r io.Reader, size int64, ct string) error {
  50 	data, err := io.ReadAll(r)
  51 	if err != nil {
  52 		return err
  53 	}
  54 	if int64(len(data)) != size {
  55 		return errors.New("size mismatch")
  56 	}
  57 	if f.onPut != nil {
  58 		f.onPut(key)
  59 	}
  60 	f.mu.Lock()
  61 	defer f.mu.Unlock()
  62 	f.puts = append(f.puts, key)
  63 	if f.putErr != nil {
  64 		return f.putErr
  65 	}
  66 	f.objects[key] = logoObject{data, ct}
  67 	return nil
  68 }
  69 func (f *logoStorage) Get(_ context.Context, key string) (io.ReadCloser, objectstore.ObjectInfo, error) {
  70 	f.mu.Lock()
  71 	defer f.mu.Unlock()
  72 	f.gets = append(f.gets, key)
  73 	o, ok := f.objects[key]
  74 	if !ok {
  75 		return nil, objectstore.ObjectInfo{}, objectstore.ErrNotFound
  76 	}
  77 	return io.NopCloser(bytes.NewReader(o.data)), objectstore.ObjectInfo{Size: int64(len(o.data)), ContentType: o.contentType}, nil
  78 }
  79 func (f *logoStorage) Delete(_ context.Context, key string) error {
  80 	if f.onDelete != nil {
  81 		f.onDelete(key)
  82 	}
  83 	f.mu.Lock()
  84 	defer f.mu.Unlock()
  85 	f.deletes = append(f.deletes, key)
  86 	if f.deleteErr != nil {
  87 		return f.deleteErr
  88 	}
  89 	delete(f.objects, key)
  90 	return nil
  91 }

  97 func pngLogo(t *testing.T, w, h, size int) []byte {
  98 	t.Helper()
  99 	var b bytes.Buffer
 100 	if err := png.Encode(&b, image.NewRGBA(image.Rect(0, 0, w, h))); err != nil {
 101 		t.Fatal(err)
 102 	}
 103 	data := b.Bytes()
 104 	if size == 0 {
 105 		return data
 106 	}
 107 	padding := size - len(data) - 12
 108 	if padding < 0 {
 109 		t.Fatal("PNG fixture size too small")
 110 	}
 111 	chunk := make([]byte, padding+12)
 112 	binary.BigEndian.PutUint32(chunk, uint32(padding))
 113 	copy(chunk[4:8], "tEXt")
 114 	copy(chunk[8:], "padding\x00")
 115 	binary.BigEndian.PutUint32(chunk[len(chunk)-4:], crc32.ChecksumIEEE(chunk[4:len(chunk)-4]))
 116 	return append(append(append([]byte{}, data[:len(data)-12]...), chunk...), data[len(data)-12:]...)
 117 }
 118 func jpegLogo(t *testing.T, exif bool) []byte {
 119 	t.Helper()
 120 	var b bytes.Buffer
 121 	if err := jpeg.Encode(&b, image.NewRGBA(image.Rect(0, 0, 10, 10)), nil); err != nil {
 122 		t.Fatal(err)
 123 	}
 124 	if !exif {
 125 		return b.Bytes()
 126 	}
 127 	// Valid little-endian TIFF: orientation=6 and GPS IFD with fictitious latitude 1/1,2/1,3/1.
 128 	tiff := make([]byte, 92)
 129 	copy(tiff, "II")
 130 	binary.LittleEndian.PutUint16(tiff[2:], 42)
 131 	binary.LittleEndian.PutUint32(tiff[4:], 8)
 132 	binary.LittleEndian.PutUint16(tiff[8:], 2)
 133 	binary.LittleEndian.PutUint16(tiff[10:], 0x112)
 134 	binary.LittleEndian.PutUint16(tiff[12:], 3)
 135 	binary.LittleEndian.PutUint32(tiff[14:], 1)
 136 	binary.LittleEndian.PutUint16(tiff[18:], 6)
 137 	binary.LittleEndian.PutUint16(tiff[22:], 0x8825)
 138 	binary.LittleEndian.PutUint16(tiff[24:], 4)
 139 	binary.LittleEndian.PutUint32(tiff[26:], 1)
 140 	binary.LittleEndian.PutUint32(tiff[30:], 38)
 141 	binary.LittleEndian.PutUint16(tiff[38:], 2)
 142 	binary.LittleEndian.PutUint16(tiff[40:], 1)
 143 	binary.LittleEndian.PutUint16(tiff[42:], 2)
 144 	binary.LittleEndian.PutUint32(tiff[44:], 2)
 145 	copy(tiff[48:], "N\x00")
 146 	binary.LittleEndian.PutUint16(tiff[52:], 2)
 147 	binary.LittleEndian.PutUint16(tiff[54:], 5)
 148 	binary.LittleEndian.PutUint32(tiff[56:], 3)
 149 	binary.LittleEndian.PutUint32(tiff[60:], 68)
 150 	for i := range 3 {
 151 		binary.LittleEndian.PutUint32(tiff[68+i*8:], uint32(i+1))
 152 		binary.LittleEndian.PutUint32(tiff[72+i*8:], 1)
 153 	}
 154 	metadata := append([]byte("Exif\x00\x00"), tiff...)
 155 	segment := []byte{0xff, 0xe1, 0, 0}
 156 	binary.BigEndian.PutUint16(segment[2:], uint16(len(metadata)+2))
 157 	return append(append(append([]byte{}, b.Bytes()[:2]...), append(segment, metadata...)...), b.Bytes()[2:]...)
 158 }

 421 func (r *tenantWriteRunner) InTenantTx(ctx context.Context, id uuid.UUID, fn func(context.Context, db.Tx) error) error {
 422 	return r.TxRunner.InTenantTx(ctx, id, func(ctx context.Context, tx db.Tx) error { return fn(ctx, &tenantWriteTx{Tx: tx, runner: r}) })

 424 func (tx *tenantWriteTx) observe(sql string) {
 425 	for strings.HasPrefix(strings.TrimSpace(sql), "--") {
 426 		_, tail, ok := strings.Cut(sql, "\n")
 427 		if !ok {
 428 			return
 429 		}
 430 		sql = tail
 431 	}
 432 	upper := strings.ToUpper(strings.TrimSpace(sql))
 433 	if strings.HasPrefix(upper, "UPDATE ") || strings.HasPrefix(upper, "INSERT ") || strings.HasPrefix(upper, "DELETE ") {
 434 		tx.runner.writes = append(tx.runner.writes, upper)
 435 	}
 436 }

 438 	tx.observe(sql)
 439 	return tx.Tx.Exec(ctx, sql, args...)
 440 }
