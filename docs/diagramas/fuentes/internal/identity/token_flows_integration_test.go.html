  21 func deliveredToken(t *testing.T, runner db.TxRunner, tenantID uuid.UUID, recipient, template string) string {
  22 	t.Helper()
  23 	var raw []byte
  24 	err := runner.InTenantTx(context.Background(), tenantID, func(ctx context.Context, tx db.Tx) error {
  25 		return tx.QueryRow(ctx, `SELECT payload FROM app.outbox_messages WHERE tenant_id=$1 AND recipient=$2 AND template=$3 ORDER BY created_at DESC, id DESC LIMIT 1`, tenantID, recipient, template).Scan(&raw)
  26 	})
  27 	if err != nil {
  28 		t.Fatal(err)
  29 	}
  30 	var payload map[string]string
  31 	if err := json.Unmarshal(raw, &payload); err != nil {
  32 		t.Fatal(err)
  33 	}
  34 	if payload["token"] == "" {
  35 		t.Fatal("missing token")
  36 	}
  37 	return payload["token"]
  38 }

  40 func TestPasswordResetLifecycle(t *testing.T) {
  41 	ctx := context.Background()
  42 	hasher := password.NewHasher(2)
  43 	svc, runner, company, email, clock, _ := loginFixture(t, hasher)
  44 	original, err := hasher.Hash(ctx, "old-password")
  45 	if err != nil {
  46 		t.Fatal(err)
  47 	}
  48 	setLoginPassword(t, runner, company, original, "active")
  49 	session, err := svc.Login(ctx, email, "old-password", RequestMeta{})
  50 	if err != nil {
  51 		t.Fatal(err)
  52 	}
  53 	if err := svc.RequestPasswordReset(ctx, email, RequestMeta{}); err != nil {
  54 		t.Fatal(err)
  55 	}
  56 	first := deliveredToken(t, runner, company.ID, email, "password_reset")
  57 	if err := svc.RequestPasswordReset(ctx, email, RequestMeta{}); err != nil {
  58 		t.Fatal(err)
  59 	}
  60 	second := deliveredToken(t, runner, company.ID, email, "password_reset")
  61 	if first == second {
  62 		t.Fatal("token was not renewed")
  63 	}
  64 	if err := svc.ConfirmPasswordReset(ctx, first, "replacement-password", RequestMeta{}); !errors.Is(err, ErrTokenInvalid) {
  65 		t.Fatalf("old token: %v", err)
  66 	}
  67 	if err := svc.ConfirmPasswordReset(ctx, second, "short", RequestMeta{}); err == nil {
  68 		t.Fatal("accepted short password")
  69 	}
  70 	if err := svc.ConfirmPasswordReset(ctx, second, "replacement-password", RequestMeta{}); err != nil {
  71 		t.Fatal(err)
  72 	}
  73 	if err := svc.ConfirmPasswordReset(ctx, second, "another-password", RequestMeta{}); !errors.Is(err, ErrTokenInvalid) {
  74 		t.Fatalf("reused token: %v", err)
  75 	}
  76 	if _, err := svc.ResolveSession(ctx, session.RawToken); !errors.Is(err, ErrUnauthenticated) {
  77 		t.Fatalf("old session: %v", err)
  78 	}
  79 	if _, err := svc.Login(ctx, email, "old-password", RequestMeta{}); !errors.Is(err, ErrInvalidCredentials) {
  80 		t.Fatalf("old password: %v", err)
  81 	}
  82 	if _, err := svc.Login(ctx, email, "replacement-password", RequestMeta{}); err != nil {
  83 		t.Fatalf("new password: %v", err)
  84 	}
  85 	var used, revoked, audits int
  86 	err = runner.InTenantTx(ctx, company.ID, func(ctx context.Context, tx db.Tx) error {
  87 		return tx.QueryRow(ctx, `SELECT count(*) FILTER (WHERE used_at IS NOT NULL), count(*) FILTER (WHERE revoked_at IS NOT NULL),
  88             (SELECT count(*) FROM app.audit_log WHERE tenant_id=$1 AND action='auth.password_reset_completed' AND data->>'sessions_revoked'='2')
  89             FROM app.user_tokens WHERE tenant_id=$1 AND user_id=$2 AND purpose='password_reset'`, company.ID, company.UserID).Scan(&used, &revoked, &audits)
  90 	})
  91 	if err != nil || used != 1 || revoked < 1 || audits != 1 {
  92 		t.Fatalf("state: used=%d revoked=%d audits=%d err=%v", used, revoked, audits, err)
  93 	}
  94 	_ = clock
  95 }

 170 func TestResetExpiryAndConcurrentConsumption(t *testing.T) {
 171 	svc, runner, company, email, clock, _ := loginFixture(t, password.NewHasher(2))
 172 	ctx := context.Background()
 173 	if err := svc.RequestPasswordReset(ctx, email, RequestMeta{}); err != nil {
 174 		t.Fatal(err)
 175 	}
 176 	expired := deliveredToken(t, runner, company.ID, email, "password_reset")
 177 	clock.now = clock.now.Add(time.Hour + time.Second)
 178 	if err := svc.ConfirmPasswordReset(ctx, expired, "valid-password", RequestMeta{}); !errors.Is(err, ErrTokenInvalid) {
 179 		t.Fatalf("expired reset: %v", err)
 180 	}
 181 	if err := svc.RequestPasswordReset(ctx, email, RequestMeta{}); err != nil {
 182 		t.Fatal(err)
 183 	}
 184 	valid := deliveredToken(t, runner, company.ID, email, "password_reset")
 185 	var wg sync.WaitGroup
 186 	results := make(chan error, 2)
 187 	for range 2 {
 188 		wg.Add(1)
 189 		go func() {
 190 			defer wg.Done()
 191 			results <- svc.ConfirmPasswordReset(ctx, valid, "valid-password", RequestMeta{})
 192 		}()
 193 	}
 194 	wg.Wait()
 195 	close(results)
 196 	succeeded, invalid := 0, 0
 197 	for err := range results {
 198 		switch {
 199 		case err == nil:
 200 			succeeded++
 201 		case errors.Is(err, ErrTokenInvalid):
 202 			invalid++
 203 		default:
 204 			t.Fatalf("concurrent confirm: %v", err)
 205 		}
 206 	}
 207 	if succeeded != 1 || invalid != 1 {
 208 		t.Fatalf("concurrent outcomes: success=%d invalid=%d", succeeded, invalid)
 209 	}
 210 }

 243 // gatedLoginHasher holds the first Verify of the Slot it hands out (DD-45: the flows derive with the
 244 // Slot, never with the Hasher, so the gate sits there).
 245 type gatedLoginHasher struct {
 246 	password.Hasher
 247 	entered chan struct{}
 248 	release chan struct{}
 249 	once    sync.Once
 250 }
 251 
 252 func (h *gatedLoginHasher) Acquire(ctx context.Context) (password.Slot, error) {
 253 	slot, err := h.Hasher.Acquire(ctx)
 254 	if err != nil {
 255 		return nil, err
 256 	}
 257 	return &gatedLoginSlot{Slot: slot, h: h}, nil
 258 }
 259 
 260 type gatedLoginSlot struct {
 261 	password.Slot
 262 	h *gatedLoginHasher
 263 }
 264 
 265 func (s *gatedLoginSlot) Verify(plain, encoded string) (bool, bool, error) {
 266 	s.h.once.Do(func() {
 267 		close(s.h.entered)
 268 		<-s.h.release
 269 	})
 270 	return s.Slot.Verify(plain, encoded)
 271 }

 280 func (r resetRaceRunner) InSystemTx(ctx context.Context, role db.SystemRole, fn func(context.Context, db.Tx) error) error {
 281 	return r.TxRunner.InSystemTx(ctx, role, func(ctx context.Context, tx db.Tx) error {
 282 		if ctx.Value(resetRaceContextKey{}) != nil {
 283 			var pid int32
 284 			if err := tx.QueryRow(ctx, `SELECT pg_backend_pid()`).Scan(&pid); err != nil {
 285 				return err
 286 			}
 287 			r.pid <- pid
 288 		}
 289 		return fn(ctx, tx)
 290 	})
 291 }

 293 // Login holds the throttle lock while verifying its password. The reset must wait there,
 294 // without locking the user first; otherwise login's session FK and reset's throttle DELETE
 295 // form a deadlock. The user-row probe makes that ordering assertion deterministic.
 296 func TestConcurrentLoginAndPasswordResetUseSameLockOrder(t *testing.T) {
 297 	baseHasher := password.NewHasher(2)
 298 	svc, runner, company, email, _, _ := loginFixture(t, baseHasher)
 299 	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
 300 	defer cancel()
 301 	oldHash, err := baseHasher.Hash(ctx, "old-password")
 302 	if err != nil {
 303 		t.Fatal(err)
 304 	}
 305 	setLoginPassword(t, runner, company, oldHash, "active")
 306 	if err := svc.RequestPasswordReset(ctx, email, RequestMeta{}); err != nil {
 307 		t.Fatal(err)
 308 	}
 309 	token := deliveredToken(t, runner, company.ID, email, "password_reset")
 310 	gate := &gatedLoginHasher{Hasher: baseHasher, entered: make(chan struct{}), release: make(chan struct{})}
 311 	svc.hasher = gate
 312 	pids := make(chan int32, 1)
 313 	svc.runner = resetRaceRunner{TxRunner: runner, pid: pids}
 314 	loginDone := make(chan struct {
 315 		session SessionResult
 316 		err     error
 317 	}, 1)
 318 	go func() {
 319 		session, err := svc.Login(ctx, email, "old-password", RequestMeta{})
 320 		loginDone <- struct {
 321 			session SessionResult
 322 			err     error
 323 		}{session, err}
 324 	}()
 325 	select {
 326 	case <-gate.entered:
 327 	case <-ctx.Done():
 328 		t.Fatal("login did not reach password verification")
 329 	}
 330 	var release sync.Once
 331 	defer release.Do(func() { close(gate.release) })
 332 	resetDone := make(chan error, 1)
 333 	go func() {
 334 		resetCtx := context.WithValue(ctx, resetRaceContextKey{}, true)
 335 		resetDone <- svc.ConfirmPasswordReset(resetCtx, token, "new-password", RequestMeta{})
 336 	}()
 337 	var resetPID int32
 338 	select {
 339 	case resetPID = <-pids:
 340 	case <-ctx.Done():
 341 		t.Fatal("reset did not start")
 342 	}
 343 	// Wait until PostgreSQL confirms the reset is blocked by login's throttle lock.
 344 	ticker := time.NewTicker(10 * time.Millisecond)
 345 	defer ticker.Stop()
 346 	for {
 347 		var wait string
 348 		if err := pgtest.AppPool(t).QueryRow(ctx, `SELECT coalesce(wait_event_type, '') FROM pg_stat_activity WHERE pid=$1`, resetPID).Scan(&wait); err != nil {
 349 			t.Fatal(err)
 350 		}
 351 		if wait == "Lock" {
 352 			break
 353 		}
 354 		select {
 355 		case <-ticker.C:
 356 		case <-ctx.Done():
 357 			t.Fatal("reset did not wait on login's lock")
 358 		}
 359 	}
 360 	probeCtx, probeCancel := context.WithTimeout(ctx, time.Second)
 361 	defer probeCancel()
 362 	err = runner.InTenantTx(probeCtx, company.ID, func(ctx context.Context, tx db.Tx) error {
 363 		var id uuid.UUID
 364 		return tx.QueryRow(ctx, `SELECT id FROM app.users WHERE tenant_id=$1 AND id=$2 FOR KEY SHARE`, company.ID, company.UserID).Scan(&id)
 365 	})
 366 	if err != nil {
 367 		t.Fatalf("reset held the user lock while waiting for login throttle: %v", err)
 368 	}
 369 	release.Do(func() { close(gate.release) })
 370 	login := <-loginDone
 371 	if login.err != nil {
 372 		t.Fatalf("concurrent login: %v", login.err)
 373 	}
 374 	if err := <-resetDone; err != nil {
 375 		t.Fatalf("concurrent reset: %v", err)
 376 	}
 377 	if _, err := svc.ResolveSession(ctx, login.session.RawToken); !errors.Is(err, ErrUnauthenticated) {
 378 		t.Fatalf("login session survived reset: %v", err)
 379 	}
 380 	var newHash string
 381 	if err := runner.InTenantTx(ctx, company.ID, func(ctx context.Context, tx db.Tx) error {
 382 		return tx.QueryRow(ctx, `SELECT password_hash FROM app.users WHERE tenant_id=$1 AND id=$2`, company.ID, company.UserID).Scan(&newHash)
 383 	}); err != nil {
 384 		t.Fatal(err)
 385 	}
 386 	ok, _, err := baseHasher.Verify(ctx, "new-password", newHash)
 387 	if err != nil || !ok {
 388 		t.Fatalf("new password was not saved: ok=%t err=%v", ok, err)
 389 	}
 390 }
