  68 func TestLoginServiceSuccessFailureLockAndLogout(t *testing.T) {
  69 	hasher := password.NewHasher(2)
  70 	svc, runner, company, email, clock, _ := loginFixture(t, hasher)
  71 	hash, err := hasher.Hash(context.Background(), "correct-password")
  72 	if err != nil {
  73 		t.Fatal(err)
  74 	}
  75 	setLoginPassword(t, runner, company, hash, "active")
  76 	for i := 0; i < 5; i++ {
  77 		_, err := svc.Login(context.Background(), email, "wrong", RequestMeta{})
  78 		if !errors.Is(err, ErrInvalidCredentials) {
  79 			t.Fatalf("failure %d: %v", i+1, err)
  80 		}
  81 	}
  82 	_, err = svc.Login(context.Background(), email, "correct-password", RequestMeta{})
  83 	var locked *LockedError
  84 	if !errors.As(err, &locked) || locked.RetryAfter != 15*time.Minute {
  85 		t.Fatalf("locked: %v", err)
  86 	}
  87 	clock.now = clock.now.Add(15*time.Minute + time.Second)
  88 	session, err := svc.Login(context.Background(), "  "+strings.ToUpper(email)+"  ", "correct-password", RequestMeta{})
  89 	if err != nil || session.Principal.UserID != company.UserID || session.ExpiresAt != clock.now.Add(7*24*time.Hour) {
  90 		t.Fatalf("success: %+v %v", session, err)
  91 	}
  92 	if err := svc.Logout(context.Background(), session.RawToken, RequestMeta{}); err != nil {
  93 		t.Fatal(err)
  94 	}
  95 	if err := svc.Logout(context.Background(), session.RawToken, RequestMeta{}); err != nil {
  96 		t.Fatalf("idempotent logout: %v", err)
  97 	}
  98 	if _, err := svc.ResolveSession(context.Background(), session.RawToken); !errors.Is(err, ErrUnauthenticated) {
  99 		t.Fatalf("revoked session: %v", err)
 100 	}
 101 	if _, err := svc.Login(context.Background(), email, "wrong", RequestMeta{}); !errors.Is(err, ErrInvalidCredentials) {
 102 		t.Fatalf("counter not cleared: %v", err)
 103 	}
 104 	var failed, lockedAudits, succeeded, logouts int
 105 	err = runner.InTenantTx(context.Background(), company.ID, func(ctx context.Context, tx db.Tx) error {
 106 		return tx.QueryRow(ctx, `SELECT count(*) FILTER (WHERE action='auth.login_failed'), count(*) FILTER (WHERE action='auth.login_locked'), count(*) FILTER (WHERE action='auth.login_succeeded'), count(*) FILTER (WHERE action='auth.logout') FROM app.audit_log WHERE tenant_id=$1`, company.ID).Scan(&failed, &lockedAudits, &succeeded, &logouts)
 107 	})
 108 	if err != nil || failed != 6 || lockedAudits != 1 || succeeded != 1 || logouts != 1 {
 109 		t.Fatalf("audit: %d %d %d %d: %v", failed, lockedAudits, succeeded, logouts, err)
 110 	}
 111 	var correctlyShaped int
 112 	err = runner.InTenantTx(context.Background(), company.ID, func(ctx context.Context, tx db.Tx) error {
 113 		return tx.QueryRow(ctx, `SELECT count(*) FROM app.audit_log WHERE tenant_id=$1 AND
 114 			(action='auth.login_failed' AND actor_user_id IS NULL AND target_type='user' AND target_id=$2 AND data->>'reason'='bad_password'
 115 			 OR action='auth.login_locked' AND actor_user_id IS NULL AND target_type='user' AND target_id=$2 AND data='{}'::jsonb
 116 			 OR action='auth.login_succeeded' AND actor_user_id=$2 AND target_type='session' AND target_id=$3 AND data='{}'::jsonb
 117 			 OR action='auth.logout' AND actor_user_id=$2 AND target_type='session' AND target_id=$3 AND data='{}'::jsonb)`,
 118 			company.ID, company.UserID, session.Principal.SessionID).Scan(&correctlyShaped)
 119 	})
 120 	if err != nil || correctlyShaped != 9 {
 121 		t.Fatalf("audit fields: rows=%d err=%v", correctlyShaped, err)
 122 	}

 162 func TestLoginSecuritySignalsIncludeUnknownEmailsWithoutLeakingThem(t *testing.T) {
 163 	svc, _, company, email, _, logs := loginFixture(t, &countingHasher{})
 164 	missing := uuid.NewString() + "@example.com"
 165 	meta := RequestMeta{IP: netip.MustParseAddr("192.0.2.42"), RequestID: "request-42"}
 166 	failedBefore, lockedBefore := loginFailedTotal.Value(), loginLockedTotal.Value()
 167 	for _, candidate := range []string{email, missing} {
 168 		for i := 0; i < 5; i++ {
 169 			if _, err := svc.Login(context.Background(), candidate, "wrong", meta); !errors.Is(err, ErrInvalidCredentials) {
 170 				t.Fatalf("%s failure %d: %v", candidate, i+1, err)
 171 			}
 172 		}
 173 		var locked *LockedError
 174 		if _, err := svc.Login(context.Background(), candidate, "wrong", meta); !errors.As(err, &locked) {
 175 			t.Fatalf("%s locked attempt: %v", candidate, err)
 176 		}
 177 	}
 178 	if got := loginFailedTotal.Value() - failedBefore; got != 10 {
 179 		t.Fatalf("login_failed_total delta=%d want 10", got)
 180 	}
 181 	if got := loginLockedTotal.Value() - lockedBefore; got != 2 {
 182 		t.Fatalf("login_locked_total delta=%d want 2", got)
 183 	}
 184 	if strings.Contains(logs.String(), email) || strings.Contains(logs.String(), missing) {
 185 		t.Fatal("security log contains a plaintext email")
 186 	}
 187 	knownKey := hex.EncodeToString(svc.emailHMAC(email))
 188 	unknownKey := hex.EncodeToString(svc.emailHMAC(missing))
 189 	counts := map[string]map[string]int{}
 190 	for _, line := range bytes.Split(bytes.TrimSpace(logs.Bytes()), []byte("\n")) {
 191 		var entry map[string]any
 192 		if err := json.Unmarshal(line, &entry); err != nil {
 193 			t.Fatal(err)
 194 		}
 195 		key, _ := entry["email_hmac"].(string)
 196 		if key != knownKey && key != unknownKey {
 197 			t.Fatalf("unexpected email_hmac %q", key)
 198 		}
 199 		if entry["ip"] != "192.0.2.42" || entry["request_id"] != "request-42" {
 200 			t.Fatalf("missing IP or request ID: %v", entry)
 201 		}
 202 		// An already-locked attempt stops before the user lookup, so its tenant is unknown
 203 		// even if the HMAC corresponds to an existing account.
 204 		blockedBeforeLookup := entry["security_event"] == "login_locked" && entry["lock_started"] != true
 205 		if key == knownKey && !blockedBeforeLookup && entry["tenant_id"] != company.ID.String() {
 206 			t.Fatalf("resolved account lacks tenant_id: %v", entry)
 207 		}
 208 		if key == unknownKey {
 209 			if _, hasTenant := entry["tenant_id"]; hasTenant {
 210 				t.Fatalf("unknown account was assigned a tenant: %v", entry)
 211 			}
 212 		}
 213 		event, _ := entry["security_event"].(string)
 214 		if counts[key] == nil {
 215 			counts[key] = map[string]int{}
 216 		}
 217 		counts[key][event]++
 218 	}
 219 	for _, key := range []string{knownKey, unknownKey} {
 220 		if counts[key]["login_failed"] != 5 || counts[key]["login_locked"] != 2 {
 221 			t.Fatalf("events for %s: %v", key, counts[key])
 222 		}
 223 	}
 224 }

 267 func TestConcurrentLoginVerifiesOnlyFivePasswords(t *testing.T) {
 268 	h := &countingHasher{}
 269 	svc, _, _, email, _, _ := loginFixture(t, h)
 270 	var wg sync.WaitGroup
 271 	results := make(chan error, 10)
 272 	for i := 0; i < 10; i++ {
 273 		wg.Add(1)
 274 		go func() {
 275 			defer wg.Done()
 276 			_, err := svc.Login(context.Background(), email, "wrong", RequestMeta{})
 277 			results <- err
 278 		}()
 279 	}
 280 	wg.Wait()
 281 	close(results)
 282 	invalid, locked := 0, 0
 283 	for err := range results {
 284 		var lock *LockedError
 285 		switch {
 286 		case errors.Is(err, ErrInvalidCredentials):
 287 			invalid++
 288 		case errors.As(err, &lock):
 289 			locked++
 290 		default:
 291 			t.Fatalf("unexpected concurrent error: %v", err)
 292 		}
 293 	}
 294 	if invalid != 5 || locked != 5 || h.verifies.Load() != 5 {
 295 		t.Fatalf("invalid=%d locked=%d verifies=%d", invalid, locked, h.verifies.Load())
 296 	}
 297 }
