  65 func TestInviteRequiresVerifiedEmail(t *testing.T) {
  66 	ctx := context.Background()
  67 	meta := RequestMeta{}
  68 	t.Run("V1 an unverified administrator cannot invite", func(t *testing.T) {
  69 		s, r, p, _ := unverifiedFixture(t)
  70 		before := tenantFootprint(t, r, p.TenantID)
  71 		if _, _, err := s.Invite(ctx, p, "new-"+uuid.NewString()+"@example.com", authz.RoleOperator, meta); !errors.Is(err, ErrEmailNotVerified) {
  72 			t.Fatalf("got %v, want ErrEmailNotVerified", err)
  73 		}
  74 		if after := tenantFootprint(t, r, p.TenantID); after != before {
  75 			t.Fatalf("the refused invitation left rows: %v -> %v", before, after)
  76 		}
  77 	})
  78 	t.Run("V2 the check comes before looking up the email", func(t *testing.T) {
  79 		s, _, p, _ := unverifiedFixture(t)
  80 		other := fixture.NewCompany(t, pgtest.AppPool(t))
  81 		var email string
  82 		if err := db.NewTxRunner(pgtest.AppPool(t)).InTenantTx(ctx, other.ID, func(ctx context.Context, tx db.Tx) error {
  83 			return tx.QueryRow(ctx, `SELECT email FROM app.users WHERE tenant_id=$1 AND id=$2`, other.ID, other.UserID).Scan(&email)
  84 		}); err != nil {
  85 			t.Fatal(err)
  86 		}
  87 		if _, _, err := s.Invite(ctx, p, email, authz.RoleOperator, meta); !errors.Is(err, ErrEmailNotVerified) {
  88 			t.Fatalf("got %v, want ErrEmailNotVerified and not ErrEmailTaken", err)
  89 		}
  90 	})
  91 	t.Run("V3 after confirming the email it can invite", func(t *testing.T) {
  92 		s, r, p, _ := unverifiedFixture(t)
  93 		if err := s.ResendEmailVerification(ctx, p); err != nil {
  94 			t.Fatal(err)
  95 		}
  96 		token := deliveredToken(t, r, p.TenantID, emailOf(t, r, p), "email_verification")
  97 		if err := s.ConfirmEmailVerification(ctx, token, meta); err != nil {
  98 			t.Fatal(err)
  99 		}
 100 		if _, _, err := s.Invite(ctx, p, "new-"+uuid.NewString()+"@example.com", authz.RoleOperator, meta); err != nil {
 101 			t.Fatalf("verified administrator: %v", err)
 102 		}
 103 	})
 104 	t.Run("V4 an administrator that arrived by an accepted invitation invites with no further step", func(t *testing.T) {
 105 		s, r, p, _ := usersFixture(t)
 106 		invited := inviteForTest(t, s, p, authz.RoleAdmin)
 107 		accepted, err := s.AcceptInvitation(ctx, deliveredToken(t, r, p.TenantID, invited.Email, "invitation"), "B", "valid-password", meta)
 108 		if err != nil {
 109 			t.Fatal(err)
 110 		}
 111 		if _, _, err := s.Invite(ctx, accepted.Principal, "new-"+uuid.NewString()+"@example.com", authz.RoleOperator, meta); err != nil {
 112 			t.Fatalf("accepted administrator: %v", err)
 113 		}
 114 	})
 115 	t.Run("V9 an unverified administrator has nobody to reactivate", func(t *testing.T) {
 116 		s, _, p, _ := unverifiedFixture(t)
 117 		users, err := s.ListUsers(ctx, p)
 118 		if err != nil || len(users) != 1 || users[0].ID != p.UserID {
 119 			t.Fatalf("users of a fresh company: %+v %v", users, err)
 120 		}
 121 		other := fixture.NewCompany(t, pgtest.AppPool(t))
 122 		for _, id := range []uuid.UUID{uuid.New(), other.UserID} {
 123 			if _, err := s.Reactivate(ctx, p, id, meta); !errors.Is(err, ErrUserNotFound) {

 130 func TestPasswordResetVerifiesTheEmail(t *testing.T) {
 131 	ctx := context.Background()
 132 	meta := RequestMeta{}
 133 	setup := func(t *testing.T, verified *time.Time) (*Service, db.TxRunner, fixture.Company, string, *loginClock) {
 134 		hasher := password.NewHasher(2)
 135 		s, r, c, email, clock, _ := loginFixture(t, hasher)
 136 		hash, err := hasher.Hash(ctx, "old-password")
 137 		if err != nil {
 138 			t.Fatal(err)
 139 		}
 140 		setLoginPassword(t, r, c, hash, "active")
 141 		if err := r.InTenantTx(ctx, c.ID, func(ctx context.Context, tx db.Tx) error {
 142 			_, err := tx.Exec(ctx, `UPDATE app.users SET email_verified_at=$3 WHERE tenant_id=$1 AND id=$2`, c.ID, c.UserID, verified)
 143 			return err
 144 		}); err != nil {
 145 			t.Fatal(err)
 146 		}
 147 		if err := s.RequestPasswordReset(ctx, email, meta); err != nil {
 148 			t.Fatal(err)
 149 		}
 150 		return s, r, c, email, clock
 151 	}
 152 	t.Run("V5 confirming a reset verifies an unverified email", func(t *testing.T) {
 153 		s, r, c, email, clock := setup(t, nil)
 154 		if err := s.ConfirmPasswordReset(ctx, deliveredToken(t, r, c.ID, email, "password_reset"), "new-password", meta); err != nil {
 155 			t.Fatal(err)
 156 		}
 157 		p := authz.Principal{TenantID: c.ID, UserID: c.UserID}
 158 		if at := verifiedAt(t, r, p); at == nil || !at.Equal(clock.now) {
 159 			t.Fatalf("email_verified_at=%v, want the clock %v", at, clock.now)
 160 		}
 161 		session, err := s.Login(ctx, email, "new-password", meta)
 162 		if err != nil {
 163 			t.Fatal(err)
 164 		}
 165 		me, err := s.Me(ctx, session.Principal)
 166 		if err != nil || !me.EmailVerified {
 167 			t.Fatalf("/me: %+v %v", me, err)
 168 		}
 169 		// The audit entry keeps the keys it had (the fixture's session is the one it revokes).
 170 		var raw []byte
 171 		if err := r.InTenantTx(ctx, c.ID, func(ctx context.Context, tx db.Tx) error {
 172 			return tx.QueryRow(ctx, `SELECT data FROM app.audit_log WHERE tenant_id=$1 AND action='auth.password_reset_completed' AND target_id=$2`, c.ID, c.UserID).Scan(&raw)
 173 		}); err != nil {
 174 			t.Fatal(err)
 175 		}
 176 		var data map[string]any
 177 		if err := json.Unmarshal(raw, &data); err != nil || !reflect.DeepEqual(data, map[string]any{"sessions_revoked": float64(1)}) {
 178 			t.Fatalf("audit data: %s (%v)", raw, err)
 179 		}
 180 	})
 181 	t.Run("V6 confirming a reset keeps an existing verification date", func(t *testing.T) {
 182 		before := time.Date(2026, 9, 1, 8, 0, 0, 0, time.UTC)
 183 		s, r, c, email, _ := setup(t, &before)
 184 		if err := s.ConfirmPasswordReset(ctx, deliveredToken(t, r, c.ID, email, "password_reset"), "new-password", meta); err != nil {
 185 			t.Fatal(err)
 186 		}
 187 		if at := verifiedAt(t, r, authz.Principal{TenantID: c.ID, UserID: c.UserID}); at == nil || !at.Equal(before) {
 188 			t.Fatalf("email_verified_at=%v, want %v", at, before)
 189 		}
 190 	})
