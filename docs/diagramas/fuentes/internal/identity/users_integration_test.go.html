 426 				t.Fatalf("token fixture=%d want=%d", len(tokens), expectedTokens)
 427 			}
 428 			disabled, err := s.Deactivate(ctx, p, u.ID, RequestMeta{})
 429 			if err != nil || disabled.Status != "disabled" || disabled.InvitationExpiresAt != nil {
 430 				t.Fatalf("disabled: %+v %v", disabled, err)
 431 			}
 432 
 433 			if err := runner.InTenantTx(ctx, p.TenantID, func(ctx context.Context, tx db.Tx) error {
 434 				for id, before := range tokens {
 435 					var used, revoked *time.Time
 436 					if err := tx.QueryRow(ctx, `SELECT used_at,revoked_at FROM app.user_tokens WHERE tenant_id=$1 AND id=$2`, p.TenantID, id).Scan(&used, &revoked); err != nil {
 437 						return err
 438 					}
 439 					expectedRevoked := before.revoked
 440 					if before.revoke {
 441 						expectedRevoked = &c.now
 442 					}
 443 					if !sameTokenTime(used, before.used) || !sameTokenTime(revoked, expectedRevoked) {
 444 						t.Errorf("INV-11 token %s: used=%v revoked=%v want used=%v revoked=%v", id, used, revoked, before.used, expectedRevoked)
 445 					}
 446 				}
 447 				return nil
 448 			}); err != nil {
 449 				t.Fatal(err)
 450 			}
 451 			count := float64(0)
 452 			if accepted {
 453 				count = 2
 454 			}
 455 			requireUserAudit(t, runner, p, u.ID, "user.deactivated", p.UserID, map[string]any{"sessions_revoked": count})
 456 			for _, session := range sessions {
 457 				if _, err := s.ResolveSession(ctx, session.RawToken); !errors.Is(err, ErrUnauthenticated) {
 458 					t.Fatal(err)
 459 				}
 460 			}
 461 			if _, err := s.PreviewInvitation(ctx, raw); !errors.Is(err, ErrTokenInvalid) {
 462 				t.Fatal(err)
 463 			}
 464 			before := userAuditCount(t, runner, p)
 465 			if _, err := s.Deactivate(ctx, p, u.ID, RequestMeta{}); !errors.Is(err, ErrInvalidTransition) {
 466 				t.Fatal(err)
 467 			}
 468 			if userAuditCount(t, runner, p) != before {
 469 				t.Fatal("failed deactivation audited")
 470 			}
 471 			if !accepted {
 472 				err := runner.InTenantTx(ctx, p.TenantID, func(ctx context.Context, tx db.Tx) error {
 473 					_, err := tx.Exec(ctx, `UPDATE app.users SET status='active' WHERE tenant_id=$1 AND id=$2`, p.TenantID, u.ID)
 474 					return db.MapError(err)
 475 				})
 476 				var constraint *pgconn.PgError
 477 				if !errors.As(err, &constraint) || constraint.Code != "23514" || constraint.ConstraintName != "users_active_complete_chk" {
 478 					t.Fatalf("constraint: %v", err)
 479 				}
 480 			}
 481 			got, err := s.Reactivate(ctx, p, u.ID, RequestMeta{})
 482 			want := "invited"
 483 			delta := 2
 484 			if accepted {
 485 				want = "active"
 486 				delta = 1
 487 			}
 488 			if err != nil || got.Status != want {
 489 				t.Fatalf("reactivate: %+v %v", got, err)
 490 			}
 491 			requireUserAudit(t, runner, p, u.ID, "user.reactivated", p.UserID, map[string]any{"to_status": want})
 492 			if userAuditCount(t, runner, p) != before+delta {
 493 				t.Fatal("reactivation audit count")
 494 			}
 495 			for _, session := range sessions {
 496 				if _, err := s.ResolveSession(ctx, session.RawToken); !errors.Is(err, ErrUnauthenticated) {
 497 					t.Fatal("reopened session")
 498 				}
 499 			}
 500 			if accepted {
 501 				if _, err := s.Login(ctx, u.Email, "valid-password", RequestMeta{}); err != nil {
 502 					t.Fatal(err)
 503 				}
 504 				var revoked int
 505 				if err := runner.InTenantTx(ctx, p.TenantID, func(ctx context.Context, tx db.Tx) error {
 506 					return tx.QueryRow(ctx, `SELECT count(*) FROM app.sessions WHERE tenant_id=$1 AND user_id=$2 AND revoked_reason='user_disabled'`, p.TenantID, u.ID).Scan(&revoked)
 507 				}); err != nil || revoked != 2 {
 508 					t.Fatalf("revoked=%d %v", revoked, err)
 509 				}
 510 			} else {
 511 				requireUserAudit(t, runner, p, u.ID, "user.invitation_reissued", p.UserID, map[string]any{"role": "operator", "trigger": "reactivation"})
 512 				if got.InvitationExpiresAt == nil || !got.InvitationExpiresAt.Equal(c.now.Add(7*24*time.Hour)) {
 513 					t.Fatalf("expiry: %+v", got)
 514 				}
 515 				newer := deliveredToken(t, runner, p.TenantID, u.Email, "invitation")
 516 				if newer == raw {
 517 					t.Fatal("reused invitation")
 518 				}
 519 				if _, err := s.PreviewInvitation(ctx, newer); err != nil {
 520 					t.Fatal(err)
 521 				}
 522 			}
 523 			before = userAuditCount(t, runner, p)
 524 			if _, err := s.Reactivate(ctx, p, u.ID, RequestMeta{}); !errors.Is(err, ErrInvalidTransition) {
 525 				t.Fatal(err)
 526 			}
 527 			if userAuditCount(t, runner, p) != before {
 528 				t.Fatal("failed reactivation audited")
 529 			}
 530 		})
 531 	}

 533 func concurrentUsers(t *testing.T, a, b func() error) []error {
 534 	t.Helper()
 535 	start := make(chan struct{})
 536 	results := make(chan error, 2)
 537 	var wg sync.WaitGroup
 538 	for _, fn := range []func() error{a, b} {
 539 		wg.Add(1)
 540 		go func() { defer wg.Done(); <-start; results <- fn() }()
 541 	}
 542 	close(start)
 543 	wg.Wait()
 544 	close(results)
 545 	var out []error
 546 	for err := range results {
 547 		out = append(out, err)
 548 	}
 549 	return out
 550 }
