  51 // nextThrottle only changes state after an evaluated password attempt. A locked attempt never
  52 // extends the deadline and must not verify a password.
  53 func nextThrottle(state throttle, success bool, now time.Time) (throttle, bool, time.Duration) {
  54 	if state.LockedUntil != nil && now.Before(*state.LockedUntil) {
  55 		return state, true, state.LockedUntil.Sub(now)
  56 	}
  57 	if success {
  58 		return throttle{}, false, 0
  59 	}
  60 	if state.LockedUntil != nil {
  61 		state = throttle{}
  62 	}
  63 	if state.FailedCount == 0 {
  64 		state.FirstFailedAt = now
  65 	}
  66 	state.FailedCount++
  67 	state.LastFailedAt = now
  68 	if state.FailedCount >= 5 {
  69 		until := now.Add(15 * time.Minute)
  70 		state.LockedUntil = &until
  71 	}
  72 	return state, false, 0
  73 }

  75 // WithAuthentication supplies the login dependencies; the key is copied and never logged.
  76 func WithAuthentication(hasher password.Hasher, recorder audit.Recorder, hmacKey []byte, logger *slog.Logger) ServiceOption {
  77 	if hasher == nil || recorder == nil || len(hmacKey) < 32 || logger == nil {
  78 		panic("identity: invalid authentication dependencies")
  79 	}
  80 	key := append([]byte(nil), hmacKey...)
  81 	return func(s *Service) {
  82 		s.hasher, s.audit, s.hmacKey, s.logger = hasher, recorder, key, logger
  83 		s.resetEmailLimiter = ratelimit.NewLimiter(rate.Every(time.Hour/3), 3, s.clock, time.Hour)
  84 	}
  85 }

  87 func (s *Service) emailHMAC(email string) []byte {
  88 	mac := hmac.New(sha256.New, s.hmacKey)
  89 	_, _ = mac.Write([]byte(email))
  90 	return mac.Sum(nil)
  91 }

  93 // Login serializes attempts for one normalized email by locking its HMAC row. Domain errors are
  94 // returned after commit, so failed attempts and their audit entries are not rolled back.
  95 func (s *Service) Login(ctx context.Context, email, plain string, meta RequestMeta) (SessionResult, error) {
  96 	if s.hasher == nil {
  97 		return SessionResult{}, errors.New("identity: authentication not configured")
  98 	}
  99 	email = strings.ToLower(strings.TrimSpace(email))
 100 	key := s.emailHMAC(email)
 101 	// DD-45: the seat of the hasher is reserved before the transaction takes a connection of the
 102 	// pool, so waiting for it never retains one (INV-36). Order: seat, connection, row locks.
 103 	slot, err := s.hasher.Acquire(ctx)
 104 	if err != nil {
 105 		return SessionResult{}, err
 106 	}
 107 	defer slot.Release()
 108 	var result SessionResult
 109 	var outcome error
 110 	var signal loginSignal
 111 	err = s.runner.InSystemTx(ctx, db.RoleAuth, func(ctx context.Context, tx db.Tx) error {
 112 		q := store.New(tx)
 113 		now := s.clock.Now()
 114 		if err := q.EnsureLoginThrottle(ctx, store.EnsureLoginThrottleParams{EmailHmac: key, Now: now}); err != nil {
 115 			return fmt.Errorf("identity: ensure throttle: %w", db.MapError(err))
 116 		}
 117 		row, err := q.LockLoginThrottle(ctx, key)
 118 		if err != nil {
 119 			return fmt.Errorf("identity: lock throttle: %w", db.MapError(err))
 120 		}
 121 		state := throttle{FailedCount: row.FailedCount, FirstFailedAt: row.FirstFailedAt,
 122 			LastFailedAt: row.LastFailedAt, LockedUntil: row.LockedUntil}
 123 		if _, locked, retry := nextThrottle(state, false, now); locked {
 124 			outcome = &LockedError{RetryAfter: retry}
 125 			signal.event = "login_locked"
 126 			return nil
 127 		}
 128 		route, err := q.LookupUserByEmail(ctx, email)
 129 		if errors.Is(db.MapError(err), db.ErrNotFound) {
 130 			slot.VerifyDummy(plain)
 131 			return s.failedLogin(ctx, tx, key, state, now, uuid.Nil, "", meta, &outcome, &signal)
 132 		}
 133 		if err != nil {
 134 			return fmt.Errorf("identity: locate user: %w", db.MapError(err))
 135 		}
 136 		if err := tx.AsTenant(ctx, route.TenantID); err != nil {
 137 			return err
 138 		}
 139 		user, err := q.GetLoginUser(ctx, store.GetLoginUserParams{TenantID: route.TenantID, UserID: route.ID})
 140 		if err != nil {
 141 			return fmt.Errorf("identity: read login user: %w", db.MapError(err))
 142 		}
 143 		valid := false
 144 		needsRehash := false
 145 		if user.PasswordHash.Valid && user.Status != "invited" {
 146 			valid, needsRehash, err = slot.Verify(plain, user.PasswordHash.String)
 147 			if err != nil {
 148 				return fmt.Errorf("identity: verify password: %w", err)
 149 			}
 150 		} else {
 151 			slot.VerifyDummy(plain)
 152 		}
 153 		if !valid || user.Status == "invited" {
 154 			reason := "bad_password"
 155 			if user.Status == "invited" {
 156 				reason = "not_active"
 157 			}
 158 			signal.tenantID = route.TenantID
 159 			return s.failedLogin(ctx, tx, key, state, now, route.ID, reason, meta, &outcome, &signal)
 160 		}
 161 		if user.Status == "disabled" {
 162 			if err := s.recordLogin(ctx, tx, route.TenantID, route.ID, uuid.Nil, "auth.login_rejected_disabled", "", meta); err != nil {
 163 				return err
 164 			}
 165 			outcome = ErrAccountDisabled
 166 			signal = loginSignal{event: "login_disabled", tenantID: route.TenantID}
 167 			return nil
 168 		}
 169 		if needsRehash {
 170 			hash, err := slot.Hash(plain)
 171 			if err != nil {
 172 				return fmt.Errorf("identity: rehash password: %w", err)
 173 			}
 174 			if err := q.UpdatePasswordHash(ctx, store.UpdatePasswordHashParams{TenantID: route.TenantID,
 175 				UserID: route.ID, PasswordHash: pgtype.Text{String: hash, Valid: true}, Now: now}); err != nil {
 176 				return fmt.Errorf("identity: update password hash: %w", db.MapError(err))
 177 			}
 178 		}
 179 		result, err = s.CreateSession(ctx, tx, authz.Principal{TenantID: route.TenantID,
 180 			UserID: route.ID, Role: authz.Role(user.Role)}, meta)
 181 		if err != nil {
 182 			return err
 183 		}
 184 		if err := s.recordLogin(ctx, tx, route.TenantID, route.ID, result.Principal.SessionID, "auth.login_succeeded", "", meta); err != nil {
 185 			return err
 186 		}
 187 		if err := tx.AsSystem(ctx, db.RoleAuth); err != nil {
 188 			return err
 189 		}
 190 		if err := q.DeleteLoginThrottle(ctx, key); err != nil {
 191 			return fmt.Errorf("identity: clear throttle: %w", db.MapError(err))
 192 		}
 193 		return nil
 194 	})
 195 	if err != nil {
 196 		return SessionResult{}, err
 197 	}
 198 	s.emitLoginSignal(ctx, signal, key, meta)
 199 	return result, outcome
 200 }

 202 func (s *Service) failedLogin(ctx context.Context, tx db.Tx, key []byte, state throttle, now time.Time,
 203 	userID uuid.UUID, reason string, meta RequestMeta, outcome *error, signal *loginSignal) error {
 204 	state, _, _ = nextThrottle(state, false, now)
 205 	if userID != uuid.Nil {
 206 		if err := s.recordLogin(ctx, tx, txTenantID(tx), userID, uuid.Nil, "auth.login_failed", reason, meta); err != nil {
 207 			return err
 208 		}
 209 		if state.LockedUntil != nil {
 210 			if err := s.recordLogin(ctx, tx, txTenantID(tx), userID, uuid.Nil, "auth.login_locked", "", meta); err != nil {
 211 				return err
 212 			}
 213 		}
 214 		if err := tx.AsSystem(ctx, db.RoleAuth); err != nil {
 215 			return err
 216 		}
 217 	}
 218 	if err := store.New(tx).SaveLoginThrottle(ctx, store.SaveLoginThrottleParams{EmailHmac: key,
 219 		FailedCount: state.FailedCount, FirstFailedAt: state.FirstFailedAt, LastFailedAt: state.LastFailedAt,
 220 		LockedUntil: state.LockedUntil}); err != nil {
 221 		return fmt.Errorf("identity: save throttle: %w", db.MapError(err))
 222 	}
 223 	*outcome = ErrInvalidCredentials
 224 	signal.event = "login_failed"
 225 	signal.lockStarted = state.LockedUntil != nil
 226 	return nil
 227 }

 229 // emitLoginSignal runs only after commit: rolled-back attempts must not increment metrics or
 230 // claim a security event. The HMAC is hex-encoded for logs; plaintext email never leaves Login.
 231 func (s *Service) emitLoginSignal(ctx context.Context, signal loginSignal, key []byte, meta RequestMeta) {
 232 	if signal.event == "" {
 233 		return
 234 	}
 235 	switch signal.event {
 236 	case "login_failed":
 237 		loginFailedTotal.Add(1) // evaluated failure, including the fifth and unknown emails
 238 	case "login_locked":
 239 		loginLockedTotal.Add(1) // a rejected attempt while the 15-minute lock is active
 240 	}
 241 	ip := ""
 242 	if meta.IP.IsValid() {
 243 		ip = meta.IP.Unmap().String()
 244 	}
 245 	fields := []any{"security_event", signal.event, "email_hmac", hex.EncodeToString(key), "ip", ip}
 246 	if signal.tenantID != uuid.Nil {
 247 		fields = append(fields, "tenant_id", signal.tenantID.String())
 248 	}
 249 	if meta.RequestID != "" {
 250 		fields = append(fields, "request_id", meta.RequestID)
 251 	}
 252 	if signal.lockStarted {
 253 		fields = append(fields, "lock_started", true)
 254 	}
 255 	s.logger.WarnContext(ctx, "login security event", fields...)
 256 	if signal.lockStarted {
 257 		// The fifth bad password starts the lock but still returns 401. Log the transition;
 258 		// login_locked_total counts only attempts actually rejected with 429.
 259 		lockedFields := append([]any(nil), fields...)
 260 		lockedFields[1] = "login_locked"
 261 		s.logger.WarnContext(ctx, "login lock started", lockedFields...)
 262 	}
 263 }

 267 func (s *Service) recordLogin(ctx context.Context, tx db.Tx, tenantID, userID, sessionID uuid.UUID, action, reason string, meta RequestMeta) error {
 268 	entry := audit.Entry{TenantID: tenantID, Action: action, IP: meta.IP, UserAgent: meta.UserAgent}
 269 	if action == "auth.login_succeeded" {
 270 		entry.ActorUserID = &userID
 271 		entry.TargetType, entry.TargetID = "session", &sessionID
 272 	} else {
 273 		entry.TargetType, entry.TargetID = "user", &userID
 274 		if reason != "" {
 275 			entry.Data = map[string]any{"reason": reason}
 276 		}
 277 	}
 278 	return s.audit.Record(ctx, tx, entry)
 279 }

 281 // Logout revokes only the presented session. Missing, expired and previously revoked tokens are
 282 // idempotent. The cookie is cleared by the HTTP handler even when there is no valid session.
 283 func (s *Service) Logout(ctx context.Context, raw string, meta RequestMeta) error {
 284 	if strings.TrimSpace(raw) == "" {
 285 		return nil
 286 	}
 287 	return s.runner.InSystemTx(ctx, db.RoleAuth, func(ctx context.Context, tx db.Tx) error {
 288 		q := store.New(tx)
 289 		route, err := q.LookupSessionByHash(ctx, securetoken.Hash(raw))
 290 		if errors.Is(db.MapError(err), db.ErrNotFound) {
 291 			return nil
 292 		}
 293 		if err != nil {
 294 			return fmt.Errorf("identity: locate logout session: %w", db.MapError(err))
 295 		}
 296 		if err := tx.AsTenant(ctx, route.TenantID); err != nil {
 297 			return err
 298 		}
 299 		now := s.clock.Now()
 300 		userID, err := q.RevokeSession(ctx, store.RevokeSessionParams{TenantID: route.TenantID,
 301 			SessionID: route.ID, Now: &now, IdleCutoff: now.Add(-s.idle)})
 302 		if errors.Is(db.MapError(err), db.ErrNotFound) {
 303 			return nil
 304 		}
 305 		if err != nil {
 306 			return fmt.Errorf("identity: revoke session: %w", db.MapError(err))
 307 		}
 308 		return s.audit.Record(ctx, tx, audit.Entry{TenantID: route.TenantID, ActorUserID: &userID,
 309 			Action: "auth.logout", TargetType: "session", TargetID: &route.ID,
 310 			IP: meta.IP, UserAgent: meta.UserAgent})
 311 	})
