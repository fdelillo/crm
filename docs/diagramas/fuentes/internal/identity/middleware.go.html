  14 const SessionCookieName = "__Host-crm_session"
  15 
  16 // Authenticate resolves the cookie on every protected request and injects the current Principal.
  17 func Authenticate(service *Service, logger *slog.Logger) func(http.Handler) http.Handler {
  18 	return func(next http.Handler) http.Handler {
  19 		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
  20 			cookie, err := r.Cookie(SessionCookieName)
  21 			if err != nil {
  22 				httpx.WriteProblem(w, r, httpx.CodeUnauthenticated)
  23 				return
  24 			}
  25 			principal, err := service.ResolveSession(r.Context(), cookie.Value)
  26 			if errors.Is(err, ErrUnauthenticated) {
  27 				ClearSessionCookie(w)
  28 				httpx.WriteProblem(w, r, httpx.CodeUnauthenticated)
  29 				return
  30 			}
  31 			if err != nil {
  32 				httpx.WriteDBError(w, r, err, logger)
  33 				return
  34 			}
  35 			next.ServeHTTP(w, r.WithContext(authz.WithPrincipal(r.Context(), principal)))
  36 		})
  37 	}
  38 }

  40 // SetSessionCookie keeps the browser's lifetime aligned with the persisted session expiry.
  41 func SetSessionCookie(w http.ResponseWriter, session SessionResult) {
  42 	remaining := time.Until(session.ExpiresAt)
  43 	maxAge := -1
  44 	if remaining > 0 {
  45 		maxAge = int(math.Ceil(remaining.Seconds()))
  46 	}
  47 	http.SetCookie(w, &http.Cookie{Name: SessionCookieName, Value: session.RawToken, Path: "/",
  48 		MaxAge: maxAge, Expires: session.ExpiresAt, HttpOnly: true, Secure: true, SameSite: http.SameSiteLaxMode})
  49 }
  50 
  51 func ClearSessionCookie(w http.ResponseWriter) {
  52 	http.SetCookie(w, &http.Cookie{Name: SessionCookieName, Path: "/", MaxAge: -1,
  53 		HttpOnly: true, Secure: true, SameSite: http.SameSiteLaxMode})
  54 }
