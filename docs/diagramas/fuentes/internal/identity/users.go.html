  42 // Invite serializes administrator writes through INV-10, then locks an existing
  43 // invitee before rechecking its state (acceptance does not take the tenant lock).
  44 func (s *Service) Invite(ctx context.Context, p authz.Principal, email string, role authz.Role, meta RequestMeta) (User, bool, error) {
  45 	email = strings.ToLower(strings.TrimSpace(email))
  46 	fields := map[string]string{}
  47 	if !ValidEmail(email) {
  48 		fields["email"] = "invalid_format"
  49 	}
  50 	if !validUserRole(role) {
  51 		fields["role"] = "invalid_value"
  52 	}
  53 	if len(fields) > 0 {
  54 		return User{}, false, &UserValidationError{Fields: fields}
  55 	}
  56 	var result User
  57 	var reissued bool
  58 	err := s.runner.InTenantTx(ctx, p.TenantID, func(ctx context.Context, tx db.Tx) error {
  59 		q := store.New(tx)
  60 		if _, err := q.LockUsersTenant(ctx, p.TenantID); err != nil {
  61 			return db.MapError(err)
  62 		}
  63 		actor, err := s.requireActor(ctx, q, p)
  64 		if err != nil {
  65 			return err
  66 		}
  67 		// DD-46, INV-37: an administrator that has not proven its email cannot invite. It comes
  68 		// after the revalidation (INV-35) and before the email is looked up, so the answer does not
  69 		// depend on who owns that email.
  70 		if actor.EmailVerifiedAt == nil {
  71 			return ErrEmailNotVerified
  72 		}
  73 		existing, err := q.FindInvitedEmail(ctx, store.FindInvitedEmailParams{TenantID: p.TenantID, Email: email})
  74 		now := s.clock.Now()
  75 		var id uuid.UUID
  76 		switch {
  77 		case errors.Is(db.MapError(err), db.ErrNotFound):
  78 			id, err = q.InsertInvitedUser(ctx, store.InsertInvitedUserParams{TenantID: p.TenantID, Email: email, Role: string(role), Now: now})
  79 			if err != nil {
  80 				return db.MapError(err)
  81 			}
  82 			company, err := q.GetTokenFlowTenantName(ctx, p.TenantID)
  83 			if err != nil {
  84 				return db.MapError(err)
  85 			}
  86 			if err := s.issueToken(ctx, tx, p.TenantID, id, email, "invitation", 7*24*time.Hour, map[string]string{"company_name": company, "role": string(role)}, &p.UserID); err != nil {
  87 				return err
  88 			}
  89 			if err := s.recordUser(ctx, tx, p, id, "user.invited", map[string]any{"role": string(role)}, meta); err != nil {
  90 				return err
  91 			}
  92 		case err != nil:
  93 			return db.MapError(err)
  94 		default:
  95 			id = existing.ID
  96 			user, err := q.LockManagedUser(ctx, store.LockManagedUserParams{TenantID: p.TenantID, UserID: id})
  97 			if err != nil {
  98 				return db.MapError(err)
  99 			}
 100 			if _, err := nextUserStatus(user.Status, reinvite, user.PasswordHash.Valid); err != nil {
 101 				return ErrEmailTaken
 102 			}
 103 			if err := s.setUserRole(ctx, tx, p, user, role, meta); err != nil {
 104 				return err
 105 			}
 106 			if err := s.reissueInvitation(ctx, tx, p.TenantID, id, email, string(role), now, "admin", meta, &p.UserID); err != nil {
 107 				return err
 108 			}
 109 			reissued = true
 110 		}
 111 		row, err := q.GetManagedUser(ctx, store.GetManagedUserParams{TenantID: p.TenantID, UserID: id})
 112 		result = managedUser(row)
 113 		return db.MapError(err)
 114 	})
 115 	var constraint *db.ConstraintError
 116 	if errors.As(err, &constraint) && constraint.Constraint == "users_email_key" {
 117 		err = ErrEmailTaken
 118 	}
 119 	return result, reissued, err
 120 }

 122 // requireActor rereads the session and the actor right after the tenant lock (DD-44, INV-35):
 123 // the Principal was resolved before the handler, and every demotion or deactivation takes this
 124 // same lock, so what is read here is the authority the actor still has. It is a plain SELECT.
 125 func (s *Service) requireActor(ctx context.Context, q *store.Queries, p authz.Principal) (store.GetSessionUserRow, error) {
 126 	reject := func(reason string, err error) (store.GetSessionUserRow, error) {
 127 		if s.logger != nil {
 128 			s.logger.WarnContext(ctx, "actor revalidation failed", "security_event", "actor_revalidation_failed",
 129 				"tenant_id", p.TenantID, "user_id", p.UserID, "reason", reason)
 130 		}
 131 		return store.GetSessionUserRow{}, err
 132 	}
 133 	if p.SessionID == uuid.Nil {
 134 		return reject("session", ErrUnauthenticated)
 135 	}
 136 	row, err := q.GetSessionUser(ctx, store.GetSessionUserParams{TenantID: p.TenantID, SessionID: p.SessionID})
 137 	if errors.Is(db.MapError(err), db.ErrNotFound) {
 138 		return reject("session", ErrUnauthenticated)
 139 	}
 140 	if err != nil {
 141 		return store.GetSessionUserRow{}, db.MapError(err)
 142 	}
 143 	if row.UserID != p.UserID {
 144 		return reject("session", ErrUnauthenticated)
 145 	}
 146 	if !s.sessionValid(row, s.clock.Now()) {
 147 		reason := "session"
 148 		if row.Status != "active" {
 149 			reason = "status"
 150 		}
 151 		return reject(reason, ErrUnauthenticated)
 152 	}
 153 	if !authz.Can(authz.Role(row.Role), authz.SettingsManage) {
 154 		return reject("permission", authz.ErrForbidden)
 155 	}
 156 	return row, nil
 157 }

 162 func (s *Service) recordUser(ctx context.Context, tx db.Tx, p authz.Principal, id uuid.UUID, action string, data map[string]any, meta RequestMeta) error {

 165 func (s *Service) protectLastAdmin(ctx context.Context, q *store.Queries, p authz.Principal, user store.LockManagedUserRow) error {
 166 	if user.Status != "active" || user.Role != "admin" {
 167 		return nil
 168 	}
 169 	n, err := q.CountActiveAdmins(ctx, p.TenantID)
 170 	if err != nil {
 171 		return db.MapError(err)
 172 	}
 173 	if n <= 1 {
 174 		return ErrLastAdmin
 175 	}
 176 	return nil
 177 }
 178 func (s *Service) setUserRole(ctx context.Context, tx db.Tx, p authz.Principal, user store.LockManagedUserRow, role authz.Role, meta RequestMeta) error {
 179 	if user.Role == string(role) {
 180 		return nil
 181 	}
 182 	q := store.New(tx)
 183 	if role == authz.RoleOperator {
 184 		if err := s.protectLastAdmin(ctx, q, p, user); err != nil {
 185 			return err
 186 		}
 187 	}
 188 	if err := q.SetManagedUserRole(ctx, store.SetManagedUserRoleParams{TenantID: p.TenantID, UserID: user.ID, Role: string(role), Now: s.clock.Now()}); err != nil {
 189 		return db.MapError(err)
 190 	}
 191 	return s.recordUser(ctx, tx, p, user.ID, "user.role_changed", map[string]any{"from": user.Role, "to": string(role), "status": user.Status}, meta)
 192 }

 194 // withManagedUser always acquires the tenant before the target. Every successful
 195 // mutation, its outbox messages and its audit entries share this transaction.
 196 func (s *Service) withManagedUser(ctx context.Context, p authz.Principal, id uuid.UUID, action userAction, fn func(context.Context, db.Tx, store.LockManagedUserRow, string) error) (User, error) {
 197 	var result User
 198 	err := s.runner.InTenantTx(ctx, p.TenantID, func(ctx context.Context, tx db.Tx) error {
 199 		q := store.New(tx)
 200 		if _, err := q.LockUsersTenant(ctx, p.TenantID); err != nil {
 201 			return db.MapError(err)
 202 		}
 203 		if _, err := s.requireActor(ctx, q, p); err != nil {
 204 			return err
 205 		}
 206 		user, err := q.LockManagedUser(ctx, store.LockManagedUserParams{TenantID: p.TenantID, UserID: id})
 207 		if errors.Is(db.MapError(err), db.ErrNotFound) {
 208 			return ErrUserNotFound
 209 		}
 210 		if err != nil {
 211 			return db.MapError(err)
 212 		}
 213 		next, err := nextUserStatus(user.Status, action, user.PasswordHash.Valid)
 214 		if err != nil {
 215 			return err
 216 		}
 217 		if err := fn(ctx, tx, user, next); err != nil {
 218 			return err
 219 		}
 220 		row, err := q.GetManagedUser(ctx, store.GetManagedUserParams{TenantID: p.TenantID, UserID: id})
 221 		result = managedUser(row)
 222 		return db.MapError(err)
 223 	})
 224 	return result, err
 225 }
 226 func (s *Service) ListUsers(ctx context.Context, p authz.Principal) ([]User, error) {
 227 	users := make([]User, 0)

 229 		rows, err := store.New(tx).ListManagedUsers(ctx, p.TenantID)
 230 		if err != nil {

 233 		for _, row := range rows {

 235 		}

 258 func (s *Service) Deactivate(ctx context.Context, p authz.Principal, userID uuid.UUID, meta RequestMeta) (User, error) {
 259 	return s.withManagedUser(ctx, p, userID, deactivate, func(ctx context.Context, tx db.Tx, user store.LockManagedUserRow, next string) error {
 260 		q := store.New(tx)
 261 		if err := s.protectLastAdmin(ctx, q, p, user); err != nil {
 262 			return err
 263 		}
 264 		now := s.clock.Now()
 265 		if err := q.SetManagedUserStatus(ctx, store.SetManagedUserStatusParams{TenantID: p.TenantID, UserID: userID, Status: next, Now: now}); err != nil {
 266 			return db.MapError(err)
 267 		}
 268 		count, err := q.RevokeDisabledUserSessions(ctx, store.RevokeDisabledUserSessionsParams{TenantID: p.TenantID, UserID: userID, Now: &now})
 269 		if err != nil {
 270 			return db.MapError(err)
 271 		}
 272 		if err := q.RevokeDisabledUserTokens(ctx, store.RevokeDisabledUserTokensParams{TenantID: p.TenantID, UserID: userID, Now: &now}); err != nil {
 273 			return db.MapError(err)
 274 		}
 275 		return s.recordUser(ctx, tx, p, userID, "user.deactivated", map[string]any{"sessions_revoked": count}, meta)
 276 	})
 277 }
 278 func (s *Service) Reactivate(ctx context.Context, p authz.Principal, userID uuid.UUID, meta RequestMeta) (User, error) {
 279 	return s.withManagedUser(ctx, p, userID, reactivate, func(ctx context.Context, tx db.Tx, user store.LockManagedUserRow, next string) error {
 280 		now := s.clock.Now()
 281 		q := store.New(tx)
 282 		if err := q.SetManagedUserStatus(ctx, store.SetManagedUserStatusParams{TenantID: p.TenantID, UserID: userID, Status: next, Now: now}); err != nil {
 283 			return db.MapError(err)
 284 		}
 285 		if next == "invited" {
 286 			if err := s.reissueInvitation(ctx, tx, p.TenantID, userID, user.Email, user.Role, now, "reactivation", meta, &p.UserID); err != nil {
 287 				return err
 288 			}
 289 		}
 290 		return s.recordUser(ctx, tx, p, userID, "user.reactivated", map[string]any{"to_status": next}, meta)
 291 	})
 292 }

 306 // Acceptance only increases the active administrator count; DD-40 permits it to
 307 // acquire user -> token without the tenant lock used for administrator writes.
 308 func (s *Service) withInvitation(ctx context.Context, raw string, fn func(context.Context, db.Tx, store.LookupTokenByHashRow, uuid.UUID, store.GetTokenFlowUserRow, store.GetTokenForUpdateRow) error) error {
 309 	if raw == "" {
 310 		return ErrTokenInvalid
 311 	}
 312 	return s.runner.InSystemTx(ctx, db.RoleAuth, func(ctx context.Context, tx db.Tx) error {
 313 		q := store.New(tx)
 314 		route, err := q.LookupTokenByHash(ctx, securetoken.Hash(raw))
 315 		if errors.Is(db.MapError(err), db.ErrNotFound) {
 316 			return ErrTokenInvalid
 317 		}
 318 		if err != nil {
 319 			return db.MapError(err)
 320 		}
 321 		if route.Purpose != "invitation" {
 322 			return ErrTokenInvalid
 323 		}
 324 		if err := tx.AsTenant(ctx, route.TenantID); err != nil {
 325 			return err
 326 		}
 327 		id, err := q.GetTokenUser(ctx, store.GetTokenUserParams{TenantID: route.TenantID, TokenID: route.ID})
 328 		if errors.Is(db.MapError(err), db.ErrNotFound) {
 329 			return ErrTokenInvalid
 330 		}
 331 		if err != nil {
 332 			return db.MapError(err)
 333 		}
 334 		user, err := q.GetTokenFlowUser(ctx, store.GetTokenFlowUserParams{TenantID: route.TenantID, UserID: id})
 335 		if errors.Is(db.MapError(err), db.ErrNotFound) {
 336 			return ErrTokenInvalid
 337 		}
 338 		if err != nil {
 339 			return db.MapError(err)
 340 		}
 341 		token, err := q.GetTokenForUpdate(ctx, store.GetTokenForUpdateParams{TenantID: route.TenantID, TokenID: route.ID})
 342 		if errors.Is(db.MapError(err), db.ErrNotFound) {
 343 			return ErrTokenInvalid
 344 		}
 345 		if err != nil {
 346 			return db.MapError(err)
 347 		}
 348 		_, transitionErr := nextUserStatus(user.Status, acceptInvitation, false)
 349 		if token.UserID != id || token.Purpose != "invitation" || token.UsedAt != nil || token.RevokedAt != nil || !s.clock.Now().Before(token.ExpiresAt) || transitionErr != nil {
 350 			return ErrTokenInvalid
 351 		}
 352 		return fn(ctx, tx, route, id, user, token)
 353 	})
 354 }
 355 func (s *Service) AcceptInvitation(ctx context.Context, raw, name, plain string, meta RequestMeta) (SessionResult, error) {
 356 	if raw == "" {
 357 		return SessionResult{}, ErrTokenInvalid
 358 	}
 359 	// DD-45: reserve the seat before the transaction opens (INV-36).
 360 	slot, err := s.hasher.Acquire(ctx)
 361 	if err != nil {
 362 		return SessionResult{}, err
 363 	}
 364 	defer slot.Release()
 365 	var session SessionResult
 366 	err = s.withInvitation(ctx, raw, func(ctx context.Context, tx db.Tx, route store.LookupTokenByHashRow, id uuid.UUID, user store.GetTokenFlowUserRow, _ store.GetTokenForUpdateRow) error {
 367 		name = strings.TrimSpace(name)
 368 		if name == "" || utf8.RuneCountInString(name) > 120 || strings.ContainsRune(name, '\x00') {
 369 			return &UserValidationError{Fields: map[string]string{"name": "invalid_value"}}
 370 		}
 371 		if code := password.Validate(plain, user.Email); code != "" {
 372 			return &PasswordValidationError{Code: code}
 373 		}
 374 		hash, err := slot.Hash(plain)
 375 		if err != nil {
 376 			return err
 377 		}
 378 		q := store.New(tx)
 379 		now := s.clock.Now()
 380 		used, err := q.UseUserToken(ctx, store.UseUserTokenParams{TenantID: route.TenantID, TokenID: route.ID, Purpose: "invitation", Now: &now})
 381 		if err != nil {
 382 			return db.MapError(err)
 383 		}
 384 		if used != 1 {
 385 			return ErrTokenInvalid
 386 		}
 387 		if err := q.ActivateInvitedUser(ctx, store.ActivateInvitedUserParams{TenantID: route.TenantID, UserID: id, Name: pgtype.Text{String: name, Valid: true}, PasswordHash: pgtype.Text{String: hash, Valid: true}, Now: &now}); err != nil {
 388 			return db.MapError(err)
 389 		}
 390 		principal := authz.Principal{TenantID: route.TenantID, UserID: id, Role: authz.Role(user.Role)}
 391 		session, err = s.CreateSession(ctx, tx, principal, meta)
 392 		if err != nil {
 393 			return err
 394 		}
 395 		return s.recordUser(ctx, tx, principal, id, "user.invitation_accepted", nil, meta)
 396 	})
 397 	return session, err
