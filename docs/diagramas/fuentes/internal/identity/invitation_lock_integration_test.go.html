  21 // Mechanism E deliberately holds the tenant before either Invite starts. Both
  22 // must wait before deciding whether the email is new (T-B601, INV-10, DD-40).
  23 // PostgreSQL queues tuple locks: the second waiter can wait behind the first.
  24 // Count the direct/transitive blocking chain, approved by the user on 2026-10-06.
  25 func TestConcurrentNewInvitationsLockTenantBeforeEmailLookup(t *testing.T) {
  26 	s, r, p, _ := usersFixture(t)
  27 	const waitingCalls = 2
  28 	if maxConns := pgtest.AppPool(t).Config().MaxConns; waitingCalls+1 > int(maxConns) {
  29 		t.Fatalf("app pool has %d connections: insufficient for T_admin plus %d waiting calls (need %d)", maxConns, waitingCalls, waitingCalls+1)
  30 	}
  31 	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
  32 	defer cancel()
  33 	// Observe outside the app pool so T_admin and Invite waiters cannot starve the probe.
  34 	monitor, err := pgx.Connect(ctx, pgtest.AppURL(t))
  35 	if err != nil {
  36 		t.Fatal(err)
  37 	}
  38 	defer func() {
  39 		closeCtx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 5*time.Second)
  40 		defer cancel()
  41 		_ = monitor.Close(closeCtx)
  42 	}()
  43 	email := uuid.NewString() + "@example.com"
  44 	type outcome struct {
  45 		call     int
  46 		user     User
  47 		reissued bool
  48 		err      error
  49 	}
  50 	done := make(chan outcome, waitingCalls)
  51 	var outcomes []outcome
  52 	rollback := errors.New("test: release tenant through rollback")
  53 	err = r.InTenantTx(ctx, p.TenantID, func(ctx context.Context, tx db.Tx) error {
  54 		if _, err := store.New(tx).LockUsersTenant(ctx, p.TenantID); err != nil {
  55 			return err
  56 		}
  57 		var pid int32
  58 		if err := tx.QueryRow(ctx, `SELECT pg_backend_pid()`).Scan(&pid); err != nil {
  59 			return err
  60 		}
  61 		for i := range waitingCalls {
  62 			go func() {
  63 				u, reissued, err := s.Invite(ctx, p, email, authz.RoleOperator, RequestMeta{})
  64 				done <- outcome{call: i, user: u, reissued: reissued, err: err}
  65 			}()
  66 		}
  67 		ticker := time.NewTicker(10 * time.Millisecond)
  68 		defer ticker.Stop()
  69 		for {
  70 			select {
  71 			case result := <-done:
  72 				outcomes = append(outcomes, result)
  73 				if result.err != nil {
  74 					return fmt.Errorf("INV-10/DD-40: Invite %d returned before acquiring the tenant lock: %w", result.call, result.err)
  75 				}
  76 				return fmt.Errorf("INV-10/DD-40: Invite %d returned before acquiring the tenant lock", result.call)
  77 			case <-ticker.C:
  78 				var blocked int
  79 				if err := monitor.QueryRow(ctx, `WITH RECURSIVE waiting(pid) AS (
  80      SELECT pid FROM pg_stat_activity WHERE $1 = ANY(pg_blocking_pids(pid))
  81      UNION
  82      SELECT a.pid FROM pg_stat_activity a JOIN waiting w ON w.pid = ANY(pg_blocking_pids(a.pid))
  83     ) SELECT count(*) FROM waiting`, pid).Scan(&blocked); err != nil {
  84 					return err
  85 				}
  86 
  87 				if blocked == waitingCalls {
  88 					return rollback
  89 				}
  90 			case <-ctx.Done():
  91 				return fmt.Errorf("INV-10/DD-40: two Invite backends did not block on tenant: %w", ctx.Err())
  92 			}
  93 		}
  94 	})
  95 	// Always roll back the held tenant and join both calls before failing, including
  96 	// the mutation that removes its lock. The buffered channel cannot strand a sender.
  97 	for len(outcomes) < waitingCalls {
  98 		outcomes = append(outcomes, <-done)
  99 	}
 100 	if !errors.Is(err, rollback) {
 101 		t.Fatal(err)
 102 	}
 103 	var created, reissued int
 104 	for _, result := range outcomes {
 105 		if result.err != nil {
 106 			t.Fatalf("INV-10/DD-40: Invite %d must succeed after rollback (lookup must follow tenant lock): %v", result.call, result.err)
 107 		}
 108 		if result.reissued {
 109 			reissued++
 110 		} else {
 111 			created++
 112 		}
 113 		if result.user.Email != email || result.user.Role != authz.RoleOperator || result.user.Status != "invited" {
 114 			t.Fatal(result.user)
 115 		}
 116 	}
 117 	if created != 1 || reissued != 1 || outcomes[0].user.ID != outcomes[1].user.ID {
 118 		t.Fatalf("creation/reissue outcomes: %+v", outcomes)
 119 	}
 120 	id := outcomes[0].user.ID
 121 	requireUserAudit(t, r, p, id, "user.invited", p.UserID, map[string]any{"role": "operator"})
 122 	requireUserAudit(t, r, p, id, "user.invitation_reissued", p.UserID, map[string]any{"role": "operator", "trigger": "admin"})
 123 	if err := r.InTenantTx(ctx, p.TenantID, func(ctx context.Context, tx db.Tx) error {
 124 		var users, mails, open, revoked int
 125 		if err := tx.QueryRow(ctx, `SELECT
 126    (SELECT count(*) FROM app.users WHERE tenant_id=$1 AND email=$2),
 127    (SELECT count(*) FROM app.outbox_messages WHERE tenant_id=$1 AND recipient=$2 AND template='invitation' AND status='pending'),
 128    (SELECT count(*) FROM app.user_tokens WHERE tenant_id=$1 AND user_id=$3 AND purpose='invitation' AND used_at IS NULL AND revoked_at IS NULL),
 129    (SELECT count(*) FROM app.user_tokens WHERE tenant_id=$1 AND user_id=$3 AND purpose='invitation' AND revoked_at IS NOT NULL)`, p.TenantID, email, id).Scan(&users, &mails, &open, &revoked); err != nil {
 130 			return err
 131 		}
 132 		if users != 1 || mails != 2 || open != 1 || revoked != 1 {
 133 			t.Errorf("users=%d mails=%d open=%d revoked=%d", users, mails, open, revoked)
 134 		}
 135 		var actions []string
 136 		rows, err := tx.Query(ctx, `SELECT action FROM app.audit_log WHERE tenant_id=$1 AND target_id=$2 AND action LIKE 'user.%' ORDER BY action`, p.TenantID, id)
 137 		if err != nil {
 138 			return err
 139 		}
 140 		for rows.Next() {
 141 			var action string
 142 			if err := rows.Scan(&action); err != nil {
 143 				rows.Close()
 144 				return err
 145 			}
 146 			actions = append(actions, action)
 147 		}
 148 		rows.Close()
 149 		if err := rows.Err(); err != nil {
 150 			return err
 151 		}
 152 		if !reflect.DeepEqual(actions, []string{"user.invitation_reissued", "user.invited"}) {
 153 			t.Errorf("actions=%v", actions)
 154 		}
 155 		var openXID, reissueXID string
 156 		if err := tx.QueryRow(ctx, `SELECT xmin::text FROM app.user_tokens WHERE tenant_id=$1 AND user_id=$2 AND purpose='invitation' AND used_at IS NULL AND revoked_at IS NULL`, p.TenantID, id).Scan(&openXID); err != nil {
 157 			return err
 158 		}
 159 		if err := tx.QueryRow(ctx, `SELECT xmin::text FROM app.audit_log WHERE tenant_id=$1 AND target_id=$2 AND action='user.invitation_reissued'`, p.TenantID, id).Scan(&reissueXID); err != nil {
 160 			return err
 161 		}
 162 		if openXID != reissueXID {
 163 			t.Error("open invitation must belong to reissue transaction")
 164 		}
 165 		return nil
 166 	}); err != nil {
 167 		t.Fatal(err)
 168 	}
