  13 type pendingSample struct {
  14 	count  int64
  15 	oldest time.Time
  16 }
  17 
  18 // Stats stores an immutable queue sample. Scrapes read memory and the injected clock only.
  19 type Stats struct {
  20 	runner db.TxRunner
  21 	clock  clock.Clock
  22 	sample atomic.Pointer[pendingSample]
  23 }
  24 
  25 var publishedStats atomic.Pointer[Stats]
  26 var failedTotal = expvar.NewInt("outbox_failed_total")
  27 var deferredTotal = expvar.NewInt("outbox_deferred_total")
  28 var deliveryErrors = expvar.NewMap("outbox_delivery_errors_total")
  29 
  30 func init() {
  31 	for _, cause := range []Cause{CauseNetwork, CauseTransient, CauseConfig, CauseRecipient, CauseBug} {
  32 		deliveryErrors.Add(string(cause), 0)
  33 	}
  34 	expvar.Publish("outbox_pending", expvar.Func(func() any {
  35 		stats := publishedStats.Load()
  36 		if stats == nil {
  37 			return int64(-1)
  38 		}
  39 		sample := stats.sample.Load()
  40 		if sample == nil {
  41 			return int64(-1)
  42 		}
  43 		return sample.count
  44 	}))
  45 	expvar.Publish("outbox_oldest_pending_seconds", expvar.Func(func() any {
  46 		stats := publishedStats.Load()
  47 		if stats == nil {
  48 			return float64(-1)
  49 		}
  50 		sample := stats.sample.Load()
  51 		if sample == nil {
  52 			return float64(-1)
  53 		}
  54 		if sample.count == 0 {
  55 			return float64(0)
  56 		}
  57 		return max(0, stats.clock.Now().Sub(sample.oldest).Seconds())
  58 	}))
  59 }
  60 func NewStats(runner db.TxRunner, clock clock.Clock) *Stats {
  61 	task := &Stats{runner: runner, clock: clock}
  62 	publishedStats.Store(task)
  63 	return task
  64 }
  65 func (*Stats) Name() string         { return "outbox_stats" }
  66 func (*Stats) Every() time.Duration { return time.Minute }
  67 func (s *Stats) Run(ctx context.Context) error {
  68 	var sample pendingSample
  69 	err := s.runner.InSystemTx(ctx, db.RoleWorker, func(ctx context.Context, tx db.Tx) error {
  70 		row, err := store.New(tx).PendingStats(ctx)
  71 		sample = pendingSample{row.Pending, row.Oldest}
  72 		return db.MapError(err)
  73 	})
  74 	if err != nil {
  75 		s.sample.Store(nil)
  76 		return err
  77 	}
  78 	s.sample.Store(&sample)
  79 	return nil
  80 }

  82 // Delivery counters are published only after the delivery transaction commits.
  83 type deliveryMetric struct {
  84 	cause  Cause
  85 	failed bool
  86 }

  88 func (m deliveryMetric) publish() {
  89 	if m.cause != "" {
  90 		deliveryErrors.Add(string(m.cause), 1)
  91 	}
  92 	if m.failed {
  93 		failedTotal.Add(1)
