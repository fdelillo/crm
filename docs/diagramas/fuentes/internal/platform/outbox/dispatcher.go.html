  27 // retryDelay is the fixed backoff of plan §9.4: 1, 5, 15, 60 minutes, then 6 hours for every
  28 // attempt after that (the 8th exhausts it and the message becomes failed).
  29 func retryDelay(attempt int) time.Duration {
  30 	switch attempt {
  31 	case 1:
  32 		return time.Minute
  33 	case 2:
  34 		return 5 * time.Minute
  35 	case 3:
  36 		return 15 * time.Minute
  37 	case 4:
  38 		return time.Hour
  39 	default:
  40 		return 6 * time.Hour
  41 	}
  42 }

  44 // deferDelay is the pure backoff of the Dispatcher's own failures (AsTenant, GetMessage or the
  45 // mark; ADR-024 §6): clamp(age, 10s, 15m), with age = now - created_at. It needs no state besides
  46 // the message's own created_at, so it doubles naturally on successive deferrals (10s, 20s, 40s…)
  47 // without a counter column. A negative age (a clock skew, or created_at in the future) is treated
  48 // as 0: the minimum of 10s still applies.
  49 func deferDelay(age time.Duration) time.Duration {
  50 	switch {
  51 	case age < 10*time.Second:
  52 		return 10 * time.Second
  53 	case age > 15*time.Minute:
  54 		return 15 * time.Minute
  55 	default:
  56 		return age
  57 	}
  58 }

  60 // classifyHandleResult turns the result of Handler.Handle into either a cancellation or a
  61 // *DeliveryError with a cause and a phase (ADR-024 §1). cycleAlive is whether the cycle's own
  62 // context (not the per-send one SendBudget wraps) was still alive when Handle returned: it tells
  63 // "the SendBudget ran out" (network, recoverable) from "the cycle itself was cancelled or timed
  64 // out" (no classification, the message is left untouched). unclassified marks a bare error the
  65 // Handler returned that is neither a *DeliveryError nor a budget timeout: ADR-024 treats it as our
  66 // own bug (config/unknown) and the caller must log its %T, never its text (it could hold an
  67 // address).
  68 func classifyHandleResult(err error, cycleAlive bool) (de *DeliveryError, canceled, unclassified bool) {
  69 	// *DeliveryError is checked first (N1 of the second PR #8 review): a dial/TLS timeout wraps
  70 	// context.DeadlineExceeded inside a connection-phase *DeliveryError (classifyConnectionError in
  71 	// platform/mailer), and errors.Is traverses Unwrap. Checking the bare DeadlineExceeded branch
  72 	// first would replace that phase with "unknown" and the cycle would not stop (ADR-024 §4,
  73 	// INV-31). This is safe: once SendBudget itself runs out, the adapter returns ctx.Err()
  74 	// unwrapped (platform/mailer/smtp.go), never inside a *DeliveryError.
  75 	var existing *DeliveryError
  76 	if errors.As(err, &existing) {
  77 		return existing, false, false
  78 	}
  79 	if errors.Is(err, context.DeadlineExceeded) && cycleAlive {
  80 		return &DeliveryError{Cause: CauseNetwork, Phase: PhaseUnknown, Detail: "timeout"}, false, false
  81 	}
  82 	if errors.Is(err, context.Canceled) || errors.Is(err, context.DeadlineExceeded) || errors.Is(err, db.ErrCanceled) {
  83 		return nil, true, false
  84 	}
  85 	return &DeliveryError{Cause: CauseConfig, Phase: PhaseUnknown, Detail: "unclassified handler error", Err: err}, false, true
  86 }

  97 // stepFailure tags an error from one of the per-message steps that run under the company's role
  98 // (AsTenant, GetMessage, or marking the outcome) once the transaction that attempted it has
  99 // already rolled back: processOne uses step to log and to pick the "mark" record it sends to
 100 // DeferMessage. A failure without this wrapper (LockDueMessage, DeferMessage itself) always ends
 101 // the cycle; one with it is isolated to its own message when it is not db.ErrUnavailable
 102 // (ADR-024 §6).

 104 	step deliveryStep
 105 	err  error
 106 }

 111 // Dispatcher polls app.outbox_messages and delivers pending messages (ADR-010). Each message runs
 112 // in its own InSystemTx(crm_worker) -> AsTenant transaction, so SKIP LOCKED divides work across
 113 // dispatchers and a failure never holds another company's messages.
 114 type Dispatcher struct {
 115 	runner  db.TxRunner
 116 	handler Handler
 117 	clock   clock.Clock
 118 	logger  *slog.Logger
 119 	tasks   []PeriodicTask
 120 	lastRun map[string]time.Time
 121 
 122 	// Test hooks (same package only). Each runs after the real call it shadows succeeded, and may
 123 	// turn that success into a failure, so a test can simulate db.ErrUnavailable or any other
 124 	// error without a broken connection or a missing role (ADR-024 §6).
 125 	onTenant       func(context.Context, db.Tx) // invoked after AsTenant, before reading the payload
 126 	failAsTenant   func() error
 127 	failGetMessage func() error
 128 	failMarkSent   func() error
 129 }

 131 // NewDispatcher returns a Dispatcher. c defaults to clock.Real{} and logger to slog.Default() when nil.
 132 func NewDispatcher(runner db.TxRunner, handler Handler, c clock.Clock, logger *slog.Logger, tasks ...PeriodicTask) *Dispatcher {
 133 	if c == nil {
 134 		c = clock.Real{}
 135 	}
 136 	if logger == nil {
 137 		logger = slog.Default()
 138 	}
 139 	return &Dispatcher{runner: runner, handler: handler, clock: c, logger: logger,
 140 		tasks: append([]PeriodicTask{terminalCleanup{runner, logger}}, tasks...), lastRun: map[string]time.Time{}}
 141 }

 143 // Run polls immediately and then every two seconds until ctx is done. A cycle's own failures are
 144 // already logged where they happen (processOne, RunOnce); Run only drives the loop and the
 145 // periodic tasks (T-B902).
 146 func (d *Dispatcher) Run(ctx context.Context) error {
 147 	ticker := time.NewTicker(pollInterval)
 148 	defer ticker.Stop()
 149 	for {
 150 		if ctx.Err() != nil {
 151 			return nil
 152 		}
 153 		_ = d.RunOnce(ctx)
 154 		d.runTasks(ctx)
 155 		select {
 156 		case <-ctx.Done():
 157 			return nil
 158 		case <-ticker.C:
 159 		}
 160 	}
 161 }

 163 // runTasks shares Run's scheduler with deterministic clock-driven tests.
 164 func (d *Dispatcher) runTasks(ctx context.Context) {
 165 	now := d.clock.Now()
 166 	for _, task := range d.tasks {
 167 		if ctx.Err() != nil {
 168 			return
 169 		}
 170 		last, ran := d.lastRun[task.Name()]
 171 		if ran && now.Sub(last) < task.Every() {
 172 			continue
 173 		}
 174 		if err := task.Run(ctx); err != nil {
 175 			if errors.Is(err, context.Canceled) || errors.Is(err, db.ErrCanceled) {
 176 				d.logger.InfoContext(ctx, "periodic task canceled", "task", task.Name(), "outcome", "canceled")
 177 			} else {
 178 				d.logger.ErrorContext(ctx, "periodic task failed", "task", task.Name(), "err", err)
 179 			}
 180 		}
 181 		d.lastRun[task.Name()] = now
 182 	}
 183 }

 185 // RunOnce processes at most ten due messages, checking ctx before taking each one (DD-35): an
 186 // already-cancelled ctx takes none. It stops early, without error, on cancellation and on a
 187 // connection-phase delivery failure (ADR-024 §4); any other failure is logged here, at the point
 188 // it ends the cycle, with the event every caller (including Run) relies on.
 189 func (d *Dispatcher) RunOnce(ctx context.Context) error {
 190 	for i := 0; i < batchSize; i++ {
 191 		if ctx.Err() != nil {
 192 			return nil
 193 		}
 194 		handled, err := d.processOne(ctx)
 195 		if err != nil {
 196 			if errors.Is(err, context.Canceled) || errors.Is(err, db.ErrCanceled) {
 197 				d.logger.InfoContext(ctx, "outbox cycle canceled", "event", "outbox_delivery", "outcome", "canceled")
 198 				return nil
 199 			}
 200 			if errors.Is(err, errCycleStop) {
 201 				return nil
 202 			}
 203 			d.logger.ErrorContext(ctx, "outbox cycle failed", "event", "outbox_cycle_failed", "err", err)
 204 			return err
 205 		}
 206 		if !handled {
 207 			return nil
 208 		}
 209 	}
 210 	return nil
 211 }

 213 // processOne claims at most one due message and resolves it. The transaction runs with
 214 // context.WithoutCancel(ctx) (DD-35): its statements are already bounded by statement_timeout and
 215 // idle_in_transaction_session_timeout, and letting the cycle's own cancellation reach it would
 216 // risk a half-finished mark. Only Handler.Handle receives ctx directly, wrapped with SendBudget,
 217 // so an apagado can still cut a send short.
 218 func (d *Dispatcher) processOne(ctx context.Context) (bool, error) {
 219 	var item store.LockDueMessageRow
 220 	handled := false
 221 	// stopCycle is set by resolveDelivery, inside fn, when the failure was in the connection phase
 222 	// (ADR-024 §4): it is read only after the transaction below has committed, so a stop never
 223 	// rolls back the mark that was just made.
 224 	stopCycle := false
 225 	var outcome deliveryMetric
 226 	err := d.runner.InSystemTx(context.WithoutCancel(ctx), db.RoleWorker, func(fnCtx context.Context, tx db.Tx) error {
 227 		q := store.New(tx)
 228 		row, lockErr := q.LockDueMessage(fnCtx, d.clock.Now())
 229 		if errors.Is(lockErr, pgx.ErrNoRows) {
 230 			return nil
 231 		}
 232 		if lockErr != nil {
 233 			return fmt.Errorf("lock due message: %w", db.MapError(lockErr))
 234 		}
 235 		item = row
 236 		handled = true
 237 
 238 		asTenantErr := tx.AsTenant(fnCtx, item.TenantID)
 239 		if asTenantErr == nil && d.failAsTenant != nil {
 240 			asTenantErr = d.failAsTenant()
 241 		}
 242 		if asTenantErr != nil {
 243 			return &stepFailure{step: stepAsTenant, err: db.MapError(asTenantErr)}
 244 		}
 245 		if d.onTenant != nil {
 246 			d.onTenant(fnCtx, tx)
 247 		}
 248 
 249 		message, getErr := q.GetMessage(fnCtx, store.GetMessageParams{TenantID: item.TenantID, ID: item.ID})
 250 		if getErr == nil && d.failGetMessage != nil {
 251 			getErr = d.failGetMessage()
 252 		}
 253 		if getErr != nil {
 254 			return &stepFailure{step: stepGetMessage, err: db.MapError(getErr)}
 255 		}
 256 
 257 		var payload map[string]string
 258 		if jsonErr := json.Unmarshal(message.Payload, &payload); jsonErr != nil {
 259 			var resolveErr error
 260 			stopCycle, resolveErr = d.resolveDelivery(fnCtx, q, item, message.Attempts,
 261 				&DeliveryError{Cause: CauseBug, Phase: PhaseCompose, Detail: "invalid payload"}, true, &outcome)
 262 			return resolveErr
 263 		}
 264 
 265 		m := Message{TenantID: message.TenantID, Kind: message.Kind, Template: message.Template,
 266 			Recipient: message.Recipient, Payload: payload}
 267 		sendCtx, cancel := context.WithTimeout(ctx, SendBudget)
 268 		handleErr := d.handler.Handle(sendCtx, m)
 269 		cycleAlive := ctx.Err() == nil
 270 		cancel()
 271 		var resolveErr error
 272 		stopCycle, resolveErr = d.resolveDelivery(fnCtx, q, item, message.Attempts, handleErr, cycleAlive, &outcome)
 273 		return resolveErr
 274 	})
 275 
 276 	switch {
 277 	case err == nil:
 278 		outcome.publish()
 279 		if stopCycle {
 280 			return handled, errCycleStop
 281 		}
 282 		return handled, nil
 283 	case errors.Is(err, context.Canceled), errors.Is(err, db.ErrCanceled):
 284 		return handled, context.Canceled
 285 	}
 286 
 287 	var sf *stepFailure
 288 	if errors.As(err, &sf) {
 289 		if errors.Is(sf.err, db.ErrUnavailable) {
 290 			return handled, fmt.Errorf("%s: %w", sf.step, sf.err)
 291 		}
 292 		if deferErr := d.deferMessage(ctx, item, sf.step, sf.err); deferErr != nil {
 293 			return handled, deferErr
 294 		}
 295 		return handled, nil
 296 	}
 297 	return handled, err
 298 }

 300 // resolveDelivery marks the outcome of one delivery attempt and logs it (plan §9.4, ADR-024
 301 // §1-§3). handleErr is nil on success, or whatever Handler.Handle returned. A failure to mark is
 302 // returned as a *stepFailure so processOne can defer the message instead (ADR-024 §6): the
 303 // message is not lost, it is retried once the transient condition clears. stop is true only for a
 304 // connection-phase failure, and only once err is nil: the mark must still commit (ADR-024 §4 ends
 305 // the cycle only *after* the failure is recorded), so processOne reads stop once this transaction
 306 // has succeeded.
 307 func (d *Dispatcher) resolveDelivery(ctx context.Context, q *store.Queries, item store.LockDueMessageRow, attempts int32, handleErr error, cycleAlive bool, outcome *deliveryMetric) (stop bool, err error) {
 308 	if handleErr == nil {
 309 		now := d.clock.Now()
 310 		markErr := q.MarkSent(ctx, store.MarkSentParams{TenantID: item.TenantID, ID: item.ID, SentAt: &now})
 311 		if markErr == nil && d.failMarkSent != nil {
 312 			markErr = d.failMarkSent()
 313 		}
 314 		if markErr != nil {
 315 			return false, &stepFailure{step: stepMark, err: db.MapError(markErr)}
 316 		}
 317 		d.logger.InfoContext(ctx, "outbox delivery sent", "event", "outbox_delivery", "outcome", "sent",
 318 			"tenant_id", item.TenantID, "message_id", item.ID)
 319 		return false, nil
 320 	}
 321 
 322 	de, canceled, unclassified := classifyHandleResult(handleErr, cycleAlive)
 323 	if canceled {
 324 		d.logger.InfoContext(ctx, "outbox delivery canceled", "event", "outbox_delivery", "outcome", "canceled",
 325 			"tenant_id", item.TenantID, "message_id", item.ID)
 326 		return false, context.Canceled
 327 	}
 328 
 329 	outcome.cause = de.Cause
 330 	last := de.LastError()
 331 	fields := []any{"event", "outbox_delivery", "tenant_id", item.TenantID, "message_id", item.ID,
 332 		"error_cause", string(de.Cause), "smtp_phase", string(de.Phase), "last_error", last}
 333 	if de.SMTPCode != 0 {
 334 		fields = append(fields, "smtp_code", de.SMTPCode)
 335 	}
 336 	if unclassified {
 337 		fields = append(fields, "err_type", fmt.Sprintf("%T", handleErr))
 338 	}
 339 
 340 	now := d.clock.Now()
 341 	if de.Cause.Permanent() || attempts+1 >= maxAttempts {
 342 		if err := q.MarkFailed(ctx, store.MarkFailedParams{TenantID: item.TenantID, ID: item.ID,
 343 			FailedAt: &now, LastError: pgtype.Text{String: last, Valid: true}}); err != nil {
 344 			return false, &stepFailure{step: stepMark, err: db.MapError(err)}
 345 		}
 346 		outcome.failed = true
 347 		level := de.Cause.LogLevel()
 348 		outcomeFields := []any{"outcome", "failed"}
 349 		if !de.Cause.Permanent() {
 350 			outcomeFields = append(outcomeFields, "reason", "max_attempts")
 351 			level = slog.LevelError
 352 		}
 353 		d.logger.Log(ctx, level, "outbox delivery failed", append(fields, outcomeFields...)...)
 354 	} else {
 355 		next := now.Add(retryDelay(int(attempts) + 1))
 356 		if err := q.MarkRecoverable(ctx, store.MarkRecoverableParams{TenantID: item.TenantID, ID: item.ID,
 357 			NextAttemptAt: next, LastError: pgtype.Text{String: last, Valid: true}}); err != nil {
 358 			return false, &stepFailure{step: stepMark, err: db.MapError(err)}
 359 		}
 360 		d.logger.Log(ctx, de.Cause.LogLevel(), "outbox delivery will retry",
 361 			append(fields, "outcome", "retry", "next_attempt_at", next)...)
 362 	}
 363 
 364 	return de.Phase == PhaseConnection, nil
 365 }

 367 // deferMessage isolates a failure of AsTenant, GetMessage or the mark to its own message
 368 // (ADR-024 §6): the message's transaction has already rolled back, so this runs in a new one, as
 369 // crm_worker, moving only next_attempt_at. A failure here (unlike a deferral itself) always ends
 370 // the cycle: retrying it inside the same batch risks taking the same message again. cause is
 371 // included in that failure (nit of the second PR #8 review): otherwise the step that originally
 372 // failed, and a possible security_event=rls_violation, would never reach a log line.
 373 func (d *Dispatcher) deferMessage(ctx context.Context, item store.LockDueMessageRow, step deliveryStep, cause error) error {
 374 	now := d.clock.Now()
 375 	delay := deferDelay(now.Sub(item.CreatedAt))
 376 	next := now.Add(delay)
 377 	var affected int64
 378 	err := d.runner.InSystemTx(context.WithoutCancel(ctx), db.RoleWorker, func(fnCtx context.Context, tx db.Tx) error {
 379 		rows, err := store.New(tx).DeferMessage(fnCtx, store.DeferMessageParams{
 380 			NextAttemptAt: next, ID: item.ID, ExpectedNextAttemptAt: item.NextAttemptAt,
 381 		})
 382 		affected = rows
 383 		return err
 384 	})
 385 	if err != nil {
 386 		return fmt.Errorf("defer message %s after %s failure (%w): %w", item.ID, step, cause, db.MapError(err))
 387 	}
 388 	// 0 rows affected means another worker already claimed and resolved this message (DeferMessage's
 389 	// own doc comment): nothing was deferred, so there is nothing to log.
 390 	if affected == 0 {
 391 		return nil
 392 	}
 393 	deferredTotal.Add(1)
 394 	fields := []any{"event", "outbox_delivery", "outcome", "deferred", "step", string(step),
 395 		"message_id", item.ID, "tenant_id", item.TenantID, "defer_seconds", int(delay.Seconds()), "err", cause}
 396 	if errors.Is(cause, db.ErrPrivilege) {
 397 		fields = append(fields, "security_event", "rls_violation")
 398 	}
 399 	d.logger.ErrorContext(ctx, "outbox message deferred", fields...)
 400 	return nil
