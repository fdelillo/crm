   1 -- name: LockDueMessage :one
   2 -- next_attempt_at and created_at are returned too: the Dispatcher needs them to defer the message
   3 -- (ADR-024 §6) without a second read, and crm_worker already has SELECT on both columns.
   4 SELECT id, tenant_id, next_attempt_at, created_at FROM app.outbox_messages
   5 WHERE status = 'pending' AND next_attempt_at <= @now
   6 ORDER BY next_attempt_at, id
   7 FOR UPDATE SKIP LOCKED
   8 LIMIT 1;

  10 -- name: DeferMessage :execrows
  11 -- Isolates a per-message failure of the Dispatcher (AsTenant, GetMessage or the mark) from the
  12 -- rest of the batch (ADR-024 §6, INV-31): it moves only next_attempt_at, as crm_worker, with the
  13 -- UPDATE (next_attempt_at) privilege and the worker_lock policy that already exist for
  14 -- LockDueMessage (no new privilege). The condition on next_attempt_at is the optimistic check
  15 -- against another worker having already claimed and resolved the same message: 0 rows affected
  16 -- means that happened, and it is not an error. attempts and last_error are untouched: no delivery
  17 -- was attempted.
  18 UPDATE app.outbox_messages
  19 SET next_attempt_at = @next_attempt_at
  20 WHERE id = @id AND status = 'pending' AND next_attempt_at = @expected_next_attempt_at;
  21 
  22 -- name: DeleteTerminalMessages :execrows
  23 DELETE FROM app.outbox_messages
  24 WHERE status <> 'pending' AND created_at < now() - interval '30 days';

  26 -- name: PendingStats :one
  27 SELECT count(*)::bigint AS pending,
  28        coalesce(min(created_at), 'epoch'::timestamptz)::timestamptz AS oldest
  29 FROM app.outbox_messages WHERE status = 'pending';
