  21 // Message is one queued delivery: Enqueue writes it in the caller's transaction, the Dispatcher

  31 // Enqueuer writes a Message in the caller's transaction (ADR-010): it commits or rolls back with
  32 // whatever operation caused it, so a message is never visible unless its triggering change is.
  33 type Enqueuer interface {
  34 	Enqueue(ctx context.Context, tx db.Tx, m Message) error
  35 }

  37 // enqueuer fixes created_at and next_attempt_at with its own clock (DD-18) instead of the column
  38 // defaults: deferDelay (ADR-024 §6) needs created_at to agree with the clock the rest of the
  39 // outbox uses, in production and in tests alike (M6).
  40 type enqueuer struct{ clock clock.Clock }

  42 // NewEnqueuer returns the Enqueuer. c defaults to clock.Real{} when nil.
  43 func NewEnqueuer(c clock.Clock) Enqueuer {
  44 	if c == nil {
  45 		c = clock.Real{}
  46 	}
  47 	return enqueuer{clock: c}

  50 func (e enqueuer) Enqueue(ctx context.Context, tx db.Tx, m Message) error {
  51 	if tx == nil {
  52 		return errors.New("outbox: transaction required")
  53 	}
  54 	tenant, bound := tx.TenantID()
  55 	if !bound || tenant != m.TenantID {
  56 		return errors.New("outbox: transaction tenant mismatch")
  57 	}
  58 	payload, err := json.Marshal(m.Payload)
  59 	if err != nil {
  60 		return fmt.Errorf("outbox: encode payload: %w", err)
  61 	}
  62 	now := e.clock.Now()
  63 	if err := store.New(tx).InsertMessage(ctx, store.InsertMessageParams{
  64 		TenantID: m.TenantID, Kind: m.Kind, Template: m.Template, Recipient: m.Recipient, Payload: payload,
  65 		CreatedAt: now, NextAttemptAt: now,
  66 	}); err != nil {
  67 		return fmt.Errorf("outbox: enqueue: %w", db.MapError(err))
  68 	}
  69 	return nil
  70 }

  89 // Cause is why a delivery failed (ADR-024 §1). It alone decides whether the Dispatcher may retry
  90 // the message and at what log level, so every DeliveryError carries exactly one.
  91 type Cause string
  92 
  93 const (
  94 	// CauseNetwork: a timeout (of a stage or of SendBudget), a rejected, reset or closed
  95 	// connection, or a DNS failure. Always recoverable, logged at WARN.
  96 	CauseNetwork Cause = "network"
  97 	// CauseTransient: any SMTP 4xx response, in any phase. Always recoverable, logged at WARN.
  98 	CauseTransient Cause = "transient"
  99 	// CauseConfig: our configuration or the provider account, affecting every message (any 5xx of
 100 	// the connection phase or of MAIL FROM, a 5xx of RCPT TO/DATA that does not identify the
 101 	// recipient, TLS/AUTH unavailable, or anything the adapter cannot classify). Recoverable, but
 102 	// logged at ERROR: it needs an operator, not a retry (ADR-024 §4).
 103 	CauseConfig Cause = "config"
 104 	// CauseRecipient: the recipient is invalid or does not exist (ADR-024 §1's "recipient rule").
 105 	// Permanent, logged at WARN: it is routine, not an incident.
 106 	CauseRecipient Cause = "recipient"
 107 	// CauseBug: an unknown template, an unreadable or incomplete payload, or an invalid subject —
 108 	// a programming error that a retry cannot fix. Permanent, logged at ERROR.
 109 	CauseBug Cause = "bug"
 110 )
 111 
 112 // Permanent reports whether c makes a delivery failure definitive (CauseRecipient, CauseBug):
 113 // every other cause is retried with backoff until MarkFailed's attempt limit (ADR-024 §1).
 114 func (c Cause) Permanent() bool {
 115 	return c == CauseRecipient || c == CauseBug
 116 }
 117 
 118 // LogLevel is the level each delivery attempt of cause c is logged at (ADR-024 §1): ERROR for
 119 // CauseConfig and CauseBug, the two that need an operator's attention; WARN for the rest, which
 120 // are either routine (recipient) or expected to clear on retry (network, transient).
 121 func (c Cause) LogLevel() slog.Level {
 122 	switch c {
 123 	case CauseConfig, CauseBug:
 124 		return slog.LevelError
 125 	default:
 126 		return slog.LevelWarn
 127 	}
 128 }

 130 // Phase is the stage of the SMTP conversation a delivery failed in (ADR-024 §2). The connection
 131 // phase is special: any failure in it is attributable to us or the provider, never to the
 132 // message, so the Dispatcher ends the cycle after handling it (ADR-024 §4, §6).
 133 type Phase string
 134 
 135 const (
 136 	PhaseCompose    Phase = "compose"    // before connecting: template, payload, addresses, subject
 137 	PhaseConnection Phase = "connection" // TCP, TLS, greeting, EHLO, STARTTLS, AUTH
 138 	PhaseMailFrom   Phase = "mail_from"
 139 	PhaseRcptTo     Phase = "rcpt_to"
 140 	PhaseData       Phase = "data" // DATA, content and the end-of-data response
 141 	PhaseUnknown    Phase = "unknown"

 144 // maxLastErrorRunes is the CHECK (char_length(last_error) <= 1000) of migration 00006
 145 // (data-model.md §2.5): LastError truncates to it, counting runes so a multi-byte provider
 146 // message is not cut mid-character.
 147 const maxLastErrorRunes = 1000

 149 // DeliveryError describes one failed delivery attempt (ADR-024): the Mailer and the template
 150 // Handler return it instead of a bare error so the Dispatcher knows the Cause, the Phase and,
 151 // when there was an SMTP response, its code. It is a struct, not a sentinel: errors.Is only says
 152 // *that* something failed, and the Dispatcher needs to know *how* to classify and log it.
 153 type DeliveryError struct {
 154 	Cause    Cause
 155 	Phase    Phase
 156 	SMTPCode int    // the SMTP reply code, or 0 when there was none
 157 	Enhanced string // the RFC 3463 extended code ("d.d.d"), or "" when the server did not report one
 158 	Detail   string // the provider's text or a local description; LastError sanitizes it
 159 	// Err is the original error. It is never logged nor persisted (ADR-024 §3, INV-30): go-mail's
 160 	// own error text includes the recipient address.
 161 	Err error
 162 }
 163 
 164 // Error returns LastError(): even a caller that logs the bare error by mistake never leaks an
 165 // address, because this is already the sanitized text.
 166 func (e *DeliveryError) Error() string { return e.LastError() }

 168 // Unwrap exposes Err for tests (errors.Is/As); production code must never log or persist it.
 169 func (e *DeliveryError) Unwrap() error { return e.Err }

 171 // responseTextOmitted is the fixed detail of a data-phase failure that has an SMTP response
 172 // (ADR-025 §1).
 173 const responseTextOmitted = "response text omitted"

 175 // longSecretRun matches 20 or more consecutive characters of the base64, base64url and hex
 176 // alphabets: a securetoken (43), a JWT segment, a key or a long identifier (ADR-025 §2).
 177 var longSecretRun = regexp.MustCompile(`[A-Za-z0-9+/=_-]{20,}`)
 178 
 179 // LastError renders "<cause> <phase>[ <SMTP code>[ <extended code>]]: <detail>" (ADR-024 §3),
 180 // with Detail sanitized (every space-separated field containing '@', "://" or a run of 20+
 181 // characters of [A-Za-z0-9+/=_-] replaced by "[redacted]", control characters turned to spaces,
 182 // repeated spaces collapsed, trimmed; ADR-025 §2) and the whole result truncated to
 183 // maxLastErrorRunes runes, ending in "…" if it was cut. In the data phase with an SMTP code the
 184 // detail is the fixed "response text omitted" (ADR-025 §1): that is the only phase in which the
 185 // server already holds the message content, hence the link and its token, so omitting the text
 186 // is a structural guarantee and not a guess about how a provider quotes a URL. It is the only
 187 // place that builds this text: outbox_messages.last_error and every delivery log use it, so
 188 // neither can leak a recipient, a sender, a message id or a token by forgetting to sanitize
 189 // (ADR-024 §3, ADR-025, INV-30).
 190 func (e *DeliveryError) LastError() string {
 191 	head := string(e.Cause) + " " + string(e.Phase)
 192 	if e.SMTPCode != 0 {
 193 		head += " " + strconv.Itoa(e.SMTPCode)
 194 		if e.Enhanced != "" {
 195 			head += " " + e.Enhanced
 196 		}
 197 	}
 198 	detail := responseTextOmitted
 199 	if e.Phase != PhaseData || e.SMTPCode == 0 {
 200 		detail = sanitizeDetail(e.Detail)
 201 	}
 202 	return truncateRunes(head+": "+detail, maxLastErrorRunes)
 203 }
 204 
 205 // sanitizeDetail is the redaction (ADR-024 §3 (1), widened by ADR-025 §2) and whitespace
 206 // normalization step of LastError. Redacting one field too many costs nothing; keeping a secret
 207 // in a column that outlives the payload does.
 208 func sanitizeDetail(raw string) string {
 209 	noControl := strings.Map(func(r rune) rune {
 210 		if unicode.IsControl(r) {
 211 			return ' '
 212 		}
 213 		return r
 214 	}, raw)
 215 	fields := strings.Fields(noControl)
 216 	for i, f := range fields {
 217 		if strings.Contains(f, "@") || strings.Contains(f, "://") || longSecretRun.MatchString(f) {
 218 			fields[i] = "[redacted]"
 219 		}
 220 	}
 221 	return strings.Join(fields, " ")
 222 }

 224 // truncateRunes returns s unchanged if it has at most max runes, or its first max-1 runes plus
 225 // "…" (itself one rune) otherwise, so the result never exceeds max runes.
 226 func truncateRunes(s string, max int) string {
 227 	r := []rune(s)
 228 	if len(r) <= max {
 229 		return s
 230 	}
 231 	return string(r[:max-1]) + "…"
 232 }

 234 // SendBudget bounds one Handler.Handle call (DD-35, INV-29): the Dispatcher wraps ctx with it

 236 // crm_app (30s, db/bootstrap/001_roles_and_database.sql): PostgreSQL must never cut the worker's
 237 // transaction while a send is in flight. A test on each side of that inequality (here and in the
 238 // bootstrap) keeps the two from drifting apart (ADR-024 §5).
 239 const SendBudget = 20 * time.Second
