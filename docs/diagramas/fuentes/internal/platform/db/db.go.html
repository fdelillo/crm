  35 // Tx is a database transaction bound, as much as one, to a company. Only this package changes the
  36 // role of a transaction, and always with SET LOCAL, so it is undone at COMMIT or ROLLBACK and a
  37 // connection returns to the pool as crm_app with no privileges (INV-02, INV-03).
  38 type Tx interface {
  39 	DBTX
  40 	// TenantID reports the company the transaction is bound to, if any.
  41 	TenantID() (uuid.UUID, bool)
  42 	// AsTenant switches to the role of company tenantID. It returns ErrTenantAlreadyBound if the
  43 	// transaction was already bound to another company; nothing runs as the other one.
  44 	AsTenant(ctx context.Context, tenantID uuid.UUID) error
  45 	// AsSystem switches to a system role (crm_auth, crm_worker, crm_signup). The binding to a
  46 	// company, if any, is kept: the transaction may go back to that company later.
  47 	AsSystem(ctx context.Context, role SystemRole) error
  48 }

  50 // TxRunner opens the transactions of the application. Business code never touches the pool.
  51 type TxRunner interface {
  52 	// InTenantTx runs fn as the role of the company tenantID. It commits if fn returns nil and rolls
  53 	// back if fn returns an error or panics (the panic continues). fn runs at most once: if PostgreSQL
  54 	// refuses the first SET LOCAL ROLE (DD-34) the whole transaction is started again, once, before fn
  55 	// has done anything; after fn started, nothing is retried.
  56 	InTenantTx(ctx context.Context, tenantID uuid.UUID, fn func(ctx context.Context, tx Tx) error) error
  57 	// InSystemTx runs fn as a system role; fn may then move to one company with AsTenant.
  58 	InSystemTx(ctx context.Context, role SystemRole, fn func(ctx context.Context, tx Tx) error) error
  59 }

  61 // TenantRoleName is the PostgreSQL role of a company: "crm_t_" + the 32 hex digits of its UUID.
  62 // It is the only place in Go that derives it (the SQL side is app.current_tenant_id()).
  63 func TenantRoleName(tenantID uuid.UUID) string {
  64 	return "crm_t_" + hex.EncodeToString(tenantID[:])
  65 }

  94 // Outcomes of the retry of DD-34, the values of the label `outcome` of set_role_retry_total and of the log.
  95 const (
  96 	// RetryRecovered: the second SET LOCAL ROLE was accepted; the stale role list was the cause.
  97 	RetryRecovered = "recovered"
  98 	// RetryFailed: the second attempt was refused too, or the company could not be entered for another reason.
  99 	RetryFailed = "failed"
 100 )
 101 
 102 var (
 103 	retriesRecovered atomic.Int64
 104 	retriesFailed    atomic.Int64
 105 )
 106 
 107 // SetRoleRetryCount is the number of entries into a company that were retried after PostgreSQL refused the
 108 // first SET LOCAL ROLE (see InTenantTx), by outcome (RetryRecovered or RetryFailed), since the process
 109 // started. It is the value of set_role_retry_total{outcome}, exported with expvar in T-B904: a value that
 110 // grows in production means the stale role-list race of PostgreSQL is happening and the first defence
 111 // (the pg_auth_members read in setRole) is not enough. An unknown outcome counts 0.
 112 func SetRoleRetryCount(outcome string) int64 {
 113 	switch outcome {
 114 	case RetryRecovered:
 115 		return retriesRecovered.Load()
 116 	case RetryFailed:
 117 		return retriesFailed.Load()
 118 	}
 119 	return 0
 120 }

 122 func (r *runner) InTenantTx(ctx context.Context, tenantID uuid.UUID, fn func(ctx context.Context, tx Tx) error) error {
 123 	enter := func(ctx context.Context, t *tx) error { return t.AsTenant(ctx, tenantID) }
 124 	fnStarted, err := r.run(ctx, enter, fn)
 125 	if err == nil || fnStarted || !isSetRoleDenied(err) {
 126 		return err
 127 	}
 128 	// Second line of defence against the stale role list (see setRole): PostgreSQL refused the very first
 129 	// SET LOCAL ROLE of the transaction, so fn has not done anything and the whole transaction, with its
 130 	// BEGIN, is started again, once. Never after fn ran, never for system roles, never for AsTenant in the
 131 	// middle of a transaction (there fn may already have written).
 132 	fnStarted, err = r.run(ctx, enter, fn)
 133 	if !fnStarted && errors.Is(err, ErrCanceled) {
 134 		return err // the client left between the attempts: the outcome of the retry is unknown, nothing to report
 135 	}
 136 	outcome := RetryRecovered
 137 	counter := &retriesRecovered
 138 	if !fnStarted && err != nil { // the second entry was refused too (or failed otherwise): fn never ran
 139 		outcome, counter = RetryFailed, &retriesFailed
 140 	}
 141 	counter.Add(1)
 142 	r.logger.WarnContext(ctx, "retried the entry into a company after SET ROLE was refused",
 143 		"event", "set_role_retry", "tenant_id", tenantID.String(), "outcome", outcome)
 144 	return err
 145 }

 152 // isSetRoleDenied recognises PostgreSQL's refusal of SET ROLE: SQLSTATE 42501 "permission denied to set role".
 153 func isSetRoleDenied(err error) bool {
 154 	var pgErr *pgconn.PgError
 155 	return errors.As(err, &pgErr) && pgErr.Code == "42501" && strings.HasPrefix(pgErr.Message, "permission denied to set role")
 156 }

 158 // run executes one transaction. fnStarted reports whether fn was called, so callers know whether
 159 // starting over is safe.
 160 func (r *runner) run(ctx context.Context, enter func(context.Context, *tx) error, fn func(context.Context, Tx) error) (fnStarted bool, err error) {
 161 	pt, err := r.pool.Begin(ctx)
 162 	if err != nil {
 163 		return false, fmt.Errorf("db: begin: %w", MapError(err))
 164 	}
 165 	t := &tx{inner: pt, fault: r.fault, logger: r.logger}
 166 	committed := false
 167 	defer func() {
 168 		if committed {
 169 			return
 170 		}
 171 		// Also runs while a panic unwinds: roll back, then let the panic continue. The caller's
 172 		// context may be the reason we are here, so the ROLLBACK gets its own.
 173 		rbCtx, cancel := context.WithTimeout(context.WithoutCancel(ctx), rollbackTimeout)
 174 		defer cancel()
 175 		_ = pt.Rollback(rbCtx)
 176 	}()
 177 
 178 	if err := enter(ctx, t); err != nil {
 179 		return false, err
 180 	}
 181 	if err := fn(ctx, t); err != nil {
 182 		return true, err
 183 	}
 184 	if err := pt.Commit(ctx); err != nil {
 185 		return true, fmt.Errorf("db: commit: %w", MapError(err))
 186 	}
 187 	committed = true
 188 	return true, nil
 189 }

 216 func (t *tx) AsTenant(ctx context.Context, tenantID uuid.UUID) error {
 217 	if t.bound && t.tenant != tenantID {
 218 		return ErrTenantAlreadyBound
 219 	}
 220 	role := TenantRoleName(tenantID)
 221 	if err := t.setRole(ctx, role); err != nil {
 222 		// MapError so that "permission denied to set role" is ErrPrivilege (INV-19); the message keeps the step
 223 		// and the role (INV-27: it tells a refused role switch from an RLS violation inside a query). The hint
 224 		// only fits a role that does not exist (22023 in PostgreSQL 18); with 42501 the role exists.
 225 		hint := ""
 226 		var pgErr *pgconn.PgError
 227 		if errors.As(err, &pgErr) && pgErr.Code == "22023" {
 228 			hint = " (was the company provisioned?)"
 229 			if t.logger != nil {
 230 				t.logger.ErrorContext(ctx, "company role missing", "security_event", "privilege_error", "step", "set_role", "role", role, "sqlstate", pgErr.Code)
 231 			}
 232 		}
 233 		return MapError(fmt.Errorf("set role %s%s: %w", role, hint, err))
 234 	}
 235 	t.tenant, t.bound = tenantID, true
 236 	return nil
 237 }

 239 func (t *tx) AsSystem(ctx context.Context, role SystemRole) error {
 240 	switch role {
 241 	case RoleAuth, RoleWorker, RoleSignup:
 242 	default:
 243 		return fmt.Errorf("db: %q is not a system role", role)
 244 	}
 245 	if err := t.setRole(ctx, string(role)); err != nil {
 246 		return MapError(fmt.Errorf("set role %s: %w", role, err))
 247 	}
 248 	return nil
 249 }

 251 // setRole is the single place that changes a transaction's role (INV-03). SET LOCAL lasts until the
 252 // end of the transaction; the identifier is quoted, and comes from a UUID or a closed set anyway.
 253 //
 254 // The SET is preceded by a read of pg_auth_members, in the same round trip. Without it, a backend that
 255 // has not used SET ROLE since another connection created and granted a new company role can answer
 256 // "permission denied to set role" for a role that exists and was granted: the per-backend list of
 257 // SET-able roles (roles_is_member_of in PostgreSQL's acl.c) stays stale, and is rebuilt only when the
 258 // next role invalidation arrives. The read makes the backend refresh its role membership state first.
 259 // Measured on PostgreSQL 18.6 with 2000 roles and 16 concurrent registrations: about 1.4% of first
 260 // uses failed without it and none in 4800 attempts with it
 261 // (TestTxRunner_NewCompanyIsUsableOnAnyConnectionRightAfterProvisioning reproduces it in seconds).
 262 // pg_auth_members is readable by everybody; LIMIT 0 keeps it free.
 263 func (t *tx) setRole(ctx context.Context, role string) error {
 264 	if t.fault != nil {
 265 		if err := t.fault(role); err != nil {
 266 			return err
 267 		}
 268 	}
 269 	_, err := t.Exec(ctx, "SELECT 1 FROM pg_catalog.pg_auth_members LIMIT 0; SET LOCAL ROLE "+pgx.Identifier{role}.Sanitize())
 270 	return err
 271 }
