  23 // Config is the typed configuration of the binary (plan §10.5, §11.1).
  24 //
  25 // There is deliberately no field that could weaken the session cookie: it is always Secure
  26 // (INV-23, DD-24).
  27 type Config struct {
  28 	DatabaseURL          string // DATABASE_URL (role crm_app). Secret.
  29 	DatabaseMigrationURL string // DATABASE_MIGRATION_URL (role crm_owner); only `crm migrate` needs it. Secret.
  30 	AuthHMACKey          []byte // AUTH_HMAC_KEY, standard base64, at least 32 bytes decoded. Secret.
  31 	AppBaseURL           *url.URL
  32 	AppLinkReset         string
  33 	AppLinkVerify        string
  34 	AppLinkInvitation    string
  35 	SessionIdle          time.Duration
  36 	SessionAbsolute      time.Duration
  37 	SMTPHost             string
  38 	SMTPPort             int
  39 	SMTPUsername         string
  40 	SMTPPassword         string
  41 	SMTPFrom             string
  42 	S3Endpoint           string
  43 	S3Bucket             string
  44 	S3AccessKey          string
  45 	S3SecretKey          string
  46 	S3UseSSL             bool
  47 	HTTPAddr             string
  48 	MetricsAddr          string
  49 	TLS                  *TLSFiles // nil except in local mode with TLS_* set
  50 	// TrustedProxies are the proxies whose X-Forwarded-For is believed (TRUSTED_PROXIES, DD-32).
  51 	// Empty (the default) trusts nobody: the client IP is then always RemoteAddr.

  55 // Load reads the environment through getenv and validates the rules of plan §10.5.
  56 // It never reads files. Errors name the offending variable and never include the value of a
  57 // secret one.
  58 func Load(getenv func(string) string) (Config, error) {
  59 	var c Config
  60 	var err error
  61 
  62 	if c.DatabaseURL, err = required(getenv, "DATABASE_URL"); err != nil {
  63 		return Config{}, err
  64 	}
  65 	c.DatabaseMigrationURL = getenv("DATABASE_MIGRATION_URL")
  66 	if c.AuthHMACKey, err = hmacKey(getenv("AUTH_HMAC_KEY")); err != nil {
  67 		return Config{}, err
  68 	}
  69 	if c.AppBaseURL, err = baseURL(getenv("APP_BASE_URL")); err != nil {
  70 		return Config{}, err
  71 	}
  72 	if c.TLS, err = tlsFiles(getenv, c.AppBaseURL); err != nil {
  73 		return Config{}, err
  74 	}
  75 	links := []struct {
  76 		dst      *string
  77 		name     string
  78 		fallback string
  79 	}{
  80 		{&c.AppLinkReset, "APP_LINK_RESET", "/reset-password"},
  81 		{&c.AppLinkVerify, "APP_LINK_VERIFY", "/verify-email"},
  82 		{&c.AppLinkInvitation, "APP_LINK_INVITATION", "/accept-invitation"},
  83 	}
  84 	for _, l := range links {
  85 		if *l.dst, err = appLink(l.name, getenv(l.name), l.fallback); err != nil {
  86 			return Config{}, err
  87 		}
  88 	}
  89 	if c.SessionIdle, err = duration(getenv, "SESSION_IDLE", 24*time.Hour); err != nil {
  90 		return Config{}, err
  91 	}
  92 	if c.SessionAbsolute, err = duration(getenv, "SESSION_ABSOLUTE", 168*time.Hour); err != nil {
  93 		return Config{}, err
  94 	}
  95 	if c.SessionIdle > c.SessionAbsolute {
  96 		return Config{}, errors.New("config: SESSION_IDLE must not be greater than SESSION_ABSOLUTE")
  97 	}
  98 	c.SMTPHost = getenv("SMTP_HOST")
  99 	c.SMTPFrom = getenv("SMTP_FROM")
 100 	if c.IsLocal() {
 101 		c.SMTPHost = orDefault(c.SMTPHost, "localhost")
 102 		c.SMTPFrom = orDefault(c.SMTPFrom, "no-reply@crm.local")
 103 	}
 104 	if rawPort := getenv("SMTP_PORT"); rawPort != "" {
 105 		c.SMTPPort, err = strconv.Atoi(rawPort)
 106 		if err != nil || c.SMTPPort < 1 || c.SMTPPort > 65535 {
 107 			return Config{}, errors.New("config: SMTP_PORT must be between 1 and 65535")
 108 		}
 109 	} else if c.IsLocal() {
 110 		c.SMTPPort = 1025
 111 	}
 112 	c.SMTPUsername, c.SMTPPassword = getenv("SMTP_USERNAME"), getenv("SMTP_PASSWORD")
 113 	c.S3Endpoint, c.S3Bucket = getenv("S3_ENDPOINT"), getenv("S3_BUCKET")
 114 	c.S3AccessKey, c.S3SecretKey = getenv("S3_ACCESS_KEY"), getenv("S3_SECRET_KEY")
 115 	if rawSSL := getenv("S3_USE_SSL"); rawSSL != "" {
 116 		c.S3UseSSL, err = strconv.ParseBool(rawSSL)
 117 		if err != nil {
 118 			return Config{}, errors.New("config: S3_USE_SSL must be a boolean")
 119 		}
 120 	}
 121 	if c.IsLocal() {
 122 		c.S3Endpoint = orDefault(c.S3Endpoint, "localhost:9000")
 123 		c.S3Bucket = orDefault(c.S3Bucket, "crm-dev")
 124 		c.S3AccessKey = orDefault(c.S3AccessKey, "minio_dev")
 125 		c.S3SecretKey = orDefault(c.S3SecretKey, "minio_dev_secret")
 126 	}
 127 	if c.TrustedProxies, err = trustedProxies(getenv("TRUSTED_PROXIES")); err != nil {
 128 		return Config{}, err
 129 	}
 130 	c.HTTPAddr = orDefault(getenv("HTTP_ADDR"), ":8080")
 131 	c.MetricsAddr = orDefault(getenv("METRICS_ADDR"), "127.0.0.1:9090")
 132 	return c, nil
 133 }

 135 // IsLocal reports whether APP_BASE_URL points at this machine: the host is exactly localhost
 136 // or 127.0.0.1 (any port). It is the only place that decides "local mode", used for TLS and
 137 // HSTS (DD-24).
 138 func (c Config) IsLocal() bool {
 139 	if c.AppBaseURL == nil {
 140 		return false
 141 	}
 142 	switch strings.ToLower(c.AppBaseURL.Hostname()) {
 143 	case "localhost", "127.0.0.1":
 144 		return true
 145 	}
 146 	return false
 147 }

 164 // hmacKey decodes AUTH_HMAC_KEY. The error never includes the value.
 165 func hmacKey(raw string) ([]byte, error) {
 166 	if raw == "" {
 167 		return nil, errors.New("config: AUTH_HMAC_KEY is required")
 168 	}
 169 	key, err := base64.StdEncoding.DecodeString(raw)
 170 	if err != nil {
 171 		return nil, errors.New("config: AUTH_HMAC_KEY must be standard base64")
 172 	}
 173 	if len(key) < minHMACKeyBytes {
 174 		return nil, fmt.Errorf("config: AUTH_HMAC_KEY must decode to at least %d bytes", minHMACKeyBytes)
 175 	}
 176 	return key, nil
 177 }

 179 // baseURL validates APP_BASE_URL: always https (H-10, DD-24), origin only (email links are
 180 // APP_BASE_URL + APP_LINK_*), and normalized without a trailing slash.
 181 func baseURL(raw string) (*url.URL, error) {
 182 	if raw == "" {
 183 		return nil, errors.New("config: APP_BASE_URL is required")
 184 	}
 185 	u, err := url.Parse(raw)
 186 	if err != nil || u.Scheme != "https" || u.Hostname() == "" {
 187 		return nil, errors.New("config: APP_BASE_URL must be an absolute https:// URL")
 188 	}
 189 	if u.User != nil || u.RawQuery != "" || u.ForceQuery || u.Fragment != "" ||
 190 		strings.ContainsAny(raw, "?#") || (u.Path != "" && u.Path != "/") {
 191 		return nil, errors.New("config: APP_BASE_URL must be only scheme, host and optional port (no user info, path, query or fragment)")
 192 	}
 193 	u.Path, u.RawPath = "", ""
 194 	return u, nil
 195 }

 197 // tlsFiles validates TLS_CERT_FILE / TLS_KEY_FILE: both or none, and only in local mode.
 198 func tlsFiles(getenv func(string) string, base *url.URL) (*TLSFiles, error) {
 199 	cert, key := getenv("TLS_CERT_FILE"), getenv("TLS_KEY_FILE")
 200 	if cert == "" && key == "" {
 201 		return nil, nil
 202 	}
 203 	if !(Config{AppBaseURL: base}).IsLocal() {
 204 		return nil, errors.New("config: TLS_CERT_FILE/TLS_KEY_FILE are only accepted in local mode (APP_BASE_URL host localhost or 127.0.0.1): in production TLS is terminated by the hosting")
 205 	}
 206 	if cert == "" {
 207 		return nil, errors.New("config: TLS_CERT_FILE is required when TLS_KEY_FILE is set")
 208 	}
 209 	if key == "" {
 210 		return nil, errors.New("config: TLS_KEY_FILE is required when TLS_CERT_FILE is set")
 211 	}
 212 	return &TLSFiles{CertFile: cert, KeyFile: key}, nil
 213 }

 215 // appLink validates a path used to build email links: it starts with "/" and has no query or
 216 // fragment (the token goes in the fragment, appended later, DD-14).
 217 func appLink(name, raw, fallback string) (string, error) {
 218 	v := orDefault(raw, fallback)
 219 	if !strings.HasPrefix(v, "/") || strings.ContainsAny(v, "#?") {
 220 		return "", fmt.Errorf("config: %s must start with '/' and contain no '#' or '?'", name)
 221 	}
 222 	return v, nil
 223 }

 225 func duration(getenv func(string) string, name string, fallback time.Duration) (time.Duration, error) {
 226 	raw := getenv(name)
 227 	if raw == "" {
 228 		return fallback, nil
 229 	}
 230 	d, err := time.ParseDuration(raw)
 231 	if err != nil || d <= 0 {
 232 		return 0, fmt.Errorf("config: %s must be a positive duration such as 24h", name)
 233 	}
 234 	return d, nil
 235 }

 237 // trustedProxies parses TRUSTED_PROXIES: CIDRs or bare addresses separated by commas (blanks around
 238 // them are fine; empty items are skipped). A bare address is a /32 or /128. Trusting every address
 239 // (0.0.0.0/0, ::/0) would let any client forge its IP, so it is a startup error, and so is anything
 240 // that does not parse; the error names the variable and the item's position, not more.
 241 func trustedProxies(raw string) ([]netip.Prefix, error) {
 242 	var out []netip.Prefix
 243 	pos := 0
 244 	for _, item := range strings.Split(raw, ",") {
 245 		item = strings.TrimSpace(item)
 246 		if item == "" {
 247 			continue
 248 		}
 249 		pos++
 250 		p, err := netip.ParsePrefix(item)
 251 		if err != nil {
 252 			addr, addrErr := netip.ParseAddr(item)
 253 			if addrErr != nil {
 254 				return nil, fmt.Errorf("config: TRUSTED_PROXIES item #%d is not a CIDR or an IP address", pos)
 255 			}
 256 			p = netip.PrefixFrom(addr, addr.BitLen())
 257 		}
 258 		if p.Addr().Is4In6() {
 259 			return nil, mappedProxyError(pos, p)
 260 		}
 261 		if p.Bits() == 0 {
 262 			return nil, fmt.Errorf("config: TRUSTED_PROXIES item #%d (%s) would trust every address: the client IP could be forged", pos, p)
 263 		}
 264 		out = append(out, p.Masked())
 265 	}
 266 	return out, nil
 267 }

 269 // mappedProxyError rejects an IPv4-mapped IPv6 prefix (::ffff:a.b.c.d/n). A client address is compared
 270 // unmapped, so such an entry would not mean what it says, and ::ffff:0.0.0.0/96 would be a disguised
 271 // 0.0.0.0/0. The error suggests the IPv4 form when there is one.
 272 func mappedProxyError(pos int, p netip.Prefix) error {
 273 	if p.Bits() < 96 {
 274 		return fmt.Errorf("config: TRUSTED_PROXIES item #%d (%s) is an IPv4-mapped IPv6 prefix: write the IPv4 directly", pos, p)
 275 	}
 276 	v4 := netip.PrefixFrom(p.Addr().Unmap(), p.Bits()-96)
 277 	if v4.Bits() == 0 {
 278 		return fmt.Errorf("config: TRUSTED_PROXIES item #%d (%s) is an IPv4-mapped IPv6 prefix that would trust every address: the client IP could be forged", pos, p)
 279 	}
 280 	return fmt.Errorf("config: TRUSTED_PROXIES item #%d (%s) is an IPv4-mapped IPv6 prefix: write the IPv4 directly (%s)", pos, p, v4.Masked())
 281 }
