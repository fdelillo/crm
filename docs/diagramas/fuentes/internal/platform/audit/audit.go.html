  23 // Entry is one audit_log row to write (data-model.md §2.6, plan FR-008): who (ActorUserID, IP,
  24 // UserAgent), when (the database's own now(), not set here), and what (Action, Target*, Data).
  25 type Entry struct {
  26 	TenantID    uuid.UUID
  27 	ActorUserID *uuid.UUID
  28 	Action      string
  29 	TargetType  string
  30 	TargetID    *uuid.UUID
  31 	Data        map[string]any
  32 	IP          netip.Addr
  33 	UserAgent   string
  34 }

  36 // Recorder writes one Entry to audit_log in the caller's own transaction (tx), so the audit trail
  37 // commits or rolls back with the operation it documents (plan FR-008).
  38 type Recorder interface {
  39 	Record(ctx context.Context, tx db.Tx, e Entry) error
  40 }

  42 // Sentinels of Record (DD-37, INV-32). Errors wrap them with the path of the offending key, never
  43 // with the value.
  44 var (
  45 	ErrSecretInData    = errors.New("audit: data contains a credential")
  46 	ErrUnsupportedData = errors.New("audit: data contains an unsupported type")
  47 	ErrTxRequired      = errors.New("audit: transaction is required")
  48 )

  55 // Record validates Data (DD-37), normalizes UserAgent (DD-38) and inserts the row in tx.
  56 func (recorder) Record(ctx context.Context, tx db.Tx, e Entry) error {
  57 	// Validation runs before looking at tx: a credential must be reported as such even when the
  58 	// caller also forgot the transaction (DD-37).
  59 	if err := validateData("", e.Data); err != nil {
  60 		return err
  61 	}
  62 	if tx == nil {
  63 		return ErrTxRequired
  64 	}
  65 	data := e.Data
  66 	if data == nil {
  67 		data = map[string]any{}
  68 	}
  69 	encoded, err := json.Marshal(data)
  70 	if err != nil {
  71 		return fmt.Errorf("audit: encode data: %w", err)
  72 	}
  73 	var ip *netip.Addr
  74 	if e.IP.IsValid() {
  75 		value := e.IP.Unmap()
  76 		ip = &value
  77 	}
  78 	userAgent := httpx.NormalizeUserAgent(e.UserAgent) // DD-38, INV-33: never fail the audited operation
  79 	args := store.InsertAuditParams{
  80 		TenantID:   e.TenantID,
  81 		Action:     e.Action,
  82 		Data:       encoded,
  83 		Ip:         ip,
  84 		TargetType: pgtype.Text{String: e.TargetType, Valid: e.TargetType != ""},
  85 		UserAgent:  pgtype.Text{String: userAgent, Valid: userAgent != ""},
  86 	}
  87 	if e.ActorUserID != nil {
  88 		args.ActorUserID = uuid.NullUUID{UUID: *e.ActorUserID, Valid: true}
  89 	}
  90 	if e.TargetID != nil {
  91 		args.TargetID = uuid.NullUUID{UUID: *e.TargetID, Valid: true}
  92 	}
  93 	if id := httpx.RequestIDFrom(ctx); id != "" {
  94 		args.RequestID = pgtype.Text{String: id, Valid: true}
  95 	}
  96 	if err := store.New(tx).InsertAudit(ctx, args); err != nil {
  97 		return fmt.Errorf("audit: insert: %w", db.MapError(err))
  98 	}
  99 	return nil
 100 }

 102 // credentialKeyParts and credentialKeyExact are the key denylist of DD-37, compared against the
 103 // normalized key (lower case, letters and digits only: "Set-Cookie" -> "setcookie", "X-API-Key" ->
 104 // "xapikey"). "session" is exact because the catalog has "sessions_revoked".
 105 var (
 106 	credentialKeyParts = []string{"password", "passwd", "passphrase", "secret", "token", "authorization",
 107 		"cookie", "apikey", "privatekey", "credential", "signature", "csrf", "xsrf"}
 108 	credentialKeyExact = []string{"auth", "session", "sessionid", "sid", "otp", "pin"}
 109 )

 111 func isCredentialKey(key string) bool {
 112 	var b strings.Builder
 113 	for _, r := range key {
 114 		if unicode.IsLetter(r) || unicode.IsDigit(r) {
 115 			b.WriteRune(unicode.ToLower(r))
 116 		}
 117 	}
 118 	normalized := b.String()
 119 	for _, part := range credentialKeyParts {
 120 		if strings.Contains(normalized, part) {
 121 			return true
 122 		}
 123 	}
 124 	return slices.Contains(credentialKeyExact, normalized)
 125 }

 127 // isCredentialValue recognizes values that look like a credential whatever their key (DD-37 (3)):
 128 // an Authorization-style scheme, a JWT, or an embedded credential assignment or securetoken raw.
 129 func isCredentialValue(value string) bool {
 130 	trimmed := strings.TrimSpace(value)
 131 	lower := strings.ToLower(trimmed)
 132 	for _, scheme := range []string{"bearer ", "basic ", "digest "} {
 133 		if strings.HasPrefix(lower, scheme) {
 134 			return true
 135 		}
 136 	}
 137 	if strings.HasPrefix(trimmed, "eyJ") {
 138 		segments := strings.Split(trimmed, ".")
 139 		if len(segments) == 3 && slices.IndexFunc(segments, func(s string) bool { return !isBase64URL(s) }) < 0 {
 140 			return true
 141 		}
 142 	}
 143 	return containsEmbeddedCredential(value)
 144 }

 146 // containsEmbeddedCredential scans maximal ASCII base64url runs. A run of exactly 43 bytes is
 147 // securetoken.New's raw form; a run followed by optional whitespace and '=' or ':' is checked
 148 // against the same credential-key policy used for JSON keys (DD-37 (4)). No part of the input is
 149 // returned in an error, since it may itself be the secret.
 150 func containsEmbeddedCredential(value string) bool {
 151 	for i := 0; i < len(value); {
 152 		if !isBase64URLByte(value[i]) {
 153 			_, width := utf8.DecodeRuneInString(value[i:])
 154 			i += width
 155 			continue
 156 		}
 157 		start := i
 158 		for i < len(value) && isBase64URLByte(value[i]) {
 159 			i++
 160 		}
 161 		if i-start == 43 {
 162 			return true
 163 		}
 164 		j := i
 165 		for j < len(value) {
 166 			r, width := utf8.DecodeRuneInString(value[j:])
 167 			if !unicode.IsSpace(r) {
 168 				break
 169 			}
 170 			j += width
 171 		}
 172 		if j < len(value) && (value[j] == '=' || value[j] == ':') && isCredentialKey(value[start:i]) {
 173 			return true
 174 		}
 175 	}
 176 	return false
 177 }

 179 func isBase64URLByte(b byte) bool {
 180 	return b >= 'a' && b <= 'z' || b >= 'A' && b <= 'Z' || b >= '0' && b <= '9' || b == '-' || b == '_'
 181 }

 183 func isBase64URL(s string) bool {
 184 	if s == "" {
 185 		return false
 186 	}
 187 	for _, r := range s {
 188 		if r > 127 || !isBase64URLByte(byte(r)) {
 189 			return false
 190 		}
 191 	}
 192 	return true
 193 }

 195 // validateData walks data at every depth and rejects what DD-37 forbids. It closes the set of
 196 // admitted types on purpose: a struct, a pointer or a []byte would be serialized without being
 197 // inspected, which would make the credential check incomplete. Errors carry the path of the
 198 // offending key and never the value, so the error itself cannot leak a secret (INV-32).
 199 func validateData(path string, value any) error {
 200 	switch v := value.(type) {
 201 	case nil, bool, int, int32, int64, uuid.UUID, time.Time:
 202 		return nil
 203 	case string:
 204 		if isCredentialValue(v) {
 205 			return fmt.Errorf("audit: data.%s: %w", path, ErrSecretInData)
 206 		}
 207 		return nil
 208 	case map[string]any:
 209 		for _, key := range slices.Sorted(maps.Keys(v)) {
 210 			if err := validateEntry(path, key, v[key]); err != nil {
 211 				return err
 212 			}
 213 		}
 214 		return nil
 215 	case map[string]string:
 216 		for _, key := range slices.Sorted(maps.Keys(v)) {
 217 			if err := validateEntry(path, key, v[key]); err != nil {
 218 				return err
 219 			}
 220 		}
 221 		return nil
 222 	case []any:
 223 		for i, item := range v {
 224 			if err := validateData(fmt.Sprintf("%s[%d]", path, i), item); err != nil {
 225 				return err
 226 			}
 227 		}
 228 		return nil
 229 	case []string:
 230 		for i, item := range v {
 231 			if err := validateData(fmt.Sprintf("%s[%d]", path, i), item); err != nil {
 232 				return err
 233 			}
 234 		}
 235 		return nil
 236 	default:
 237 		return fmt.Errorf("audit: data.%s: %T: %w", path, value, ErrUnsupportedData)
 238 	}
 239 }

 241 func validateEntry(parent, key string, value any) error {
 242 	path := key
 243 	if parent != "" {
 244 		path = parent + "." + key
 245 	}
 246 	if isCredentialKey(key) {
 247 		return fmt.Errorf("audit: data.%s: %w", path, ErrSecretInData)
 248 	}
 249 	return validateData(path, value)
 250 }
