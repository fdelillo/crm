  35 // cluster is the PostgreSQL 18 shared by every test of a package: roles are cluster-wide, so one
  36 // container per package (not per test) with each test creating its own companies (ADR-012).
  37 type cluster struct {
  38 	container *postgres.PostgresContainer
  39 	ownerURL  string
  40 	appURL    string
  41 	app       *pgxpool.Pool
  42 	owner     *pgxpool.Pool
  43 	super     *pgxpool.Pool
  44 }

  46 // provider starts the cluster once and remembers the outcome, so every test of the package sees the
  47 // same result (a start failure fails all of them). start is a field so the failure path can be tested
  48 // without breaking Docker.
  49 type provider struct {
  50 	once  sync.Once
  51 	c     *cluster
  52 	err   error
  53 	start func(context.Context) (*cluster, error)
  54 }

  58 // Main is for TestMain: it runs the package's tests and then removes the container.
  59 //
  60 //	func TestMain(m *testing.M) { os.Exit(pgtest.Main(m)) }
  61 //
  62 // The container is started lazily by the first test that needs it, so a run that selects no
  63 // database test never pays for it. If TestMain does not use Main the testcontainers reaper still
  64 // removes the container when the process ends.
  65 func Main(m *testing.M) int {
  66 	code := m.Run()
  67 	stop()
  68 	return code
  69 }

  98 // ApplyBootstrap runs db/bootstrap/ again inside the container, as a DBA would, to check it is
  99 // idempotent. It runs the same file the container executed at startup.
 100 func ApplyBootstrap(t testing.TB) {
 101 	t.Helper()
 102 	c := get(t)
 103 	ctx, cancel := context.WithTimeout(context.Background(), time.Minute)
 104 	defer cancel()
 105 	code, out, err := c.container.Exec(ctx, []string{
 106 		"psql", "-v", "ON_ERROR_STOP=1", "-U", "postgres", "-d", "postgres",
 107 		"-f", "/docker-entrypoint-initdb.d/" + bootstrapFile,
 108 	}, exec.Multiplexed())
 109 	if err != nil {
 110 		t.Fatalf("pgtest: running the bootstrap again: %v", err)
 111 	}
 112 	text, _ := io.ReadAll(out)
 113 	if code != 0 {
 114 		t.Fatalf("pgtest: bootstrap exited with %d:\n%s", code, text)
 115 	}
 116 }

 123 // get fails t (never skips) when the cluster cannot start: without Docker an integration test must
 124 // not pass silently in CI (ADR-012).
 125 func (p *provider) get(t testing.TB) *cluster {
 126 	t.Helper()
 127 	p.once.Do(func() {
 128 		ctx, cancel := context.WithTimeout(context.Background(), startupDeadline)
 129 		defer cancel()
 130 		p.c, p.err = p.start(ctx)
 131 	})
 132 	if p.err != nil {
 133 		t.Fatalf("pgtest: Docker is required to run integration tests (make test-int) and PostgreSQL 18 could not start: %v", p.err)
 134 	}
 135 	return p.c
 136 }

 138 func startCluster(ctx context.Context) (*cluster, error) {
 139 	root, err := repoRoot()
 140 	if err != nil {
 141 		return nil, err
 142 	}
 143 	superPassword, ownerPassword, appPassword := randomPassword(), randomPassword(), randomPassword()
 144 
 145 	ctr, err := postgres.Run(ctx, image,
 146 		postgres.WithUsername("postgres"),
 147 		postgres.WithPassword(superPassword),
 148 		// The bootstrap reads these through psql \getenv (never stored in the repository).
 149 		testcontainers.WithEnv(map[string]string{
 150 			"CRM_OWNER_PASSWORD": ownerPassword,
 151 			"CRM_APP_PASSWORD":   appPassword,
 152 		}),
 153 		postgres.WithInitScripts(filepath.Join(root, "db", "bootstrap", bootstrapFile)),
 154 		postgres.BasicWaitStrategies(),
 155 	)
 156 	if err != nil {
 157 		if ctr != nil {
 158 			_ = testcontainers.TerminateContainer(ctr)
 159 		}
 160 		return nil, fmt.Errorf("starting %s: %w", image, err)
 161 	}
 162 
 163 	c := &cluster{container: ctr}
 164 	host, err := ctr.Host(ctx)
 165 	if err != nil {
 166 		c.close()
 167 		return nil, err
 168 	}
 169 	port, err := ctr.MappedPort(ctx, "5432/tcp")
 170 	if err != nil {
 171 		c.close()
 172 		return nil, err
 173 	}
 174 	dsn := func(user, password string) string {
 175 		u := url.URL{
 176 			Scheme:   "postgres",
 177 			User:     url.UserPassword(user, password),
 178 			Host:     net.JoinHostPort(host, port.Port()),
 179 			Path:     "/crm",
 180 			RawQuery: "sslmode=disable",
 181 		}
 182 		return u.String()
 183 	}
 184 	c.ownerURL = dsn("crm_owner", ownerPassword)
 185 	c.appURL = dsn("crm_app", appPassword)
 186 
 187 	if err := migrate(ctx, c.ownerURL); err != nil {
 188 		c.close()
 189 		return nil, err
 190 	}
 191 	for _, p := range []struct {
 192 		dst  **pgxpool.Pool
 193 		user string
 194 		pass string
 195 	}{{&c.app, "crm_app", appPassword}, {&c.owner, "crm_owner", ownerPassword}, {&c.super, "postgres", superPassword}} {
 196 		pool, err := pgxpool.New(ctx, dsn(p.user, p.pass))
 197 		if err != nil {
 198 			c.close()
 199 			return nil, fmt.Errorf("opening pool as %s: %w", p.user, err)
 200 		}
 201 		*p.dst = pool
 202 	}
 203 	return c, nil
 204 }

 206 // migrate applies the embedded migrations as crm_owner, exactly as `crm migrate up` does.
 207 func migrate(ctx context.Context, ownerURL string) error {
 208 	db, err := sql.Open("pgx", ownerURL)
 209 	if err != nil {
 210 		return fmt.Errorf("opening migration connection: %w", err)
 211 	}
 212 	defer func() { _ = db.Close() }() // the work is done by now: a close error changes nothing
 213 	provider, err := goose.NewProvider(goose.DialectPostgres, db, migrations.FS)
 214 	if err != nil {
 215 		return fmt.Errorf("creating goose provider: %w", err)
 216 	}
 217 	if _, err := provider.Up(ctx); err != nil {
 218 		return fmt.Errorf("running migrations as crm_owner: %w", err)
 219 	}
 220 	return nil
 221 }

 223 func (c *cluster) close() {
 224 	for _, p := range []*pgxpool.Pool{c.app, c.owner, c.super} {
 225 		if p == nil {
 226 			continue
 227 		}
 228 		// Close waits for every acquired connection to come back; a test that leaked one must not
 229 		// hang the whole package (and leave the container running).
 230 		done := make(chan struct{})
 231 		go func() { p.Close(); close(done) }()
 232 		select {
 233 		case <-done:
 234 		case <-time.After(10 * time.Second):
 235 			fmt.Fprintln(os.Stderr, "pgtest: a pool did not close in 10 s: some test left a connection or transaction unreleased")
 236 		}
 237 	}
 238 	if c.container != nil {
 239 		_ = testcontainers.TerminateContainer(c.container)
 240 	}
 241 }

 249 // repoRoot finds the directory holding go.mod, walking up from the package under test.
 250 func repoRoot() (string, error) {
 251 	dir, err := os.Getwd()
 252 	if err != nil {
 253 		return "", err
 254 	}
 255 	for {
 256 		if _, err := os.Stat(filepath.Join(dir, "go.mod")); err == nil {
 257 			return dir, nil
 258 		}
 259 		parent := filepath.Dir(dir)
 260 		if parent == dir {
 261 			return "", errors.New("go.mod not found above the working directory")
 262 		}
 263 		dir = parent
 264 	}
 265 }

 267 func randomPassword() string {
 268 	b := make([]byte, 16)
 269 	if _, err := rand.Read(b); err != nil {
 270 		panic(err) // crypto/rand never fails on supported platforms
 271 	}
 272 	return hex.EncodeToString(b)
 273 }
