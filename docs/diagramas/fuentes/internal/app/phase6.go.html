  18 // RegisterUserRoutes shares one IP quota between preview and accept (DD-9).
  19 // Management permission is checked before parsing or searching any target ID (INV-12). Inviting
  20 // and reactivating share a quota per IP and per company (DD-48); both run after Authenticate and
  21 // RequirePermission and before the body is decoded, and they count successes and rejections.
  22 func RegisterUserRoutes(r chi.Router, users *identity.Service, companies *tenant.Service, invitationsLimiter, inviteTenantLimiter, inviteIPLimiter *ratelimit.Limiter, logger *slog.Logger) {
  23 	r.With(ratelimit.Middleware(invitationsLimiter)).Post("/api/v1/auth/invitations/preview", func(w http.ResponseWriter, req *http.Request) {
  24 		var input struct {
  25 			Token string `json:"token"`
  26 		}
  27 		if err := httpx.DecodeJSON(w, req, &input); err != nil {
  28 			return
  29 		}
  30 		preview, err := users.PreviewInvitation(req.Context(), input.Token)
  31 		if writeUserError(w, req, err, logger) {
  32 			return
  33 		}
  34 		writeUsersJSON(w, http.StatusOK, preview)
  35 	})
  36 	r.With(ratelimit.Middleware(invitationsLimiter)).Post("/api/v1/auth/invitations/accept", func(w http.ResponseWriter, req *http.Request) {
  37 		var input struct {
  38 			Token    string `json:"token"`
  39 			Name     string `json:"name"`
  40 			Password string `json:"password"`
  41 		}
  42 		if err := httpx.DecodeJSON(w, req, &input); err != nil {
  43 			return
  44 		}
  45 		session, err := users.AcceptInvitation(req.Context(), input.Token, input.Name, input.Password, requestMeta(req))
  46 		if writeUserError(w, req, err, logger) {
  47 			return
  48 		}
  49 		user, err := users.Me(req.Context(), session.Principal)
  50 		if writeUserError(w, req, err, logger) {
  51 			return
  52 		}
  53 		company, err := companies.SummaryFor(req.Context(), session.Principal.TenantID)
  54 		if writeUserError(w, req, err, logger) {
  55 			return
  56 		}
  57 		identity.SetSessionCookie(w, session)
  58 		writeUsersJSON(w, http.StatusCreated, struct {
  59 			User        identity.CurrentUser `json:"user"`
  60 			Tenant      tenant.Summary       `json:"tenant"`
  61 			Permissions []authz.Permission   `json:"permissions"`
  62 		}{user, company, authz.Permissions(session.Principal.Role)})
  63 	})
  64 	r.Group(func(r chi.Router) {
  65 		r.Use(identity.Authenticate(users, logger), authz.RequirePermission(authz.SettingsManage))
  66 		r.Get("/api/v1/users", func(w http.ResponseWriter, req *http.Request) {
  67 			p, _ := authz.PrincipalFrom(req.Context())
  68 			items, err := users.ListUsers(req.Context(), p)
  69 			if writeUserError(w, req, err, logger) {
  70 				return
  71 			}
  72 			writeUsersJSON(w, http.StatusOK, struct {
  73 				Items []identity.User `json:"items"`
  74 			}{items})
  75 		})
  76 		inviteQuotas := r.With(ratelimit.Middleware(inviteIPLimiter), ratelimit.MiddlewareBy(inviteTenantLimiter, tenantKey))
  77 		inviteQuotas.Post("/api/v1/users/invitations", func(w http.ResponseWriter, req *http.Request) {
  78 			var input struct {
  79 				Email string     `json:"email"`
  80 				Role  authz.Role `json:"role"`
  81 			}
  82 			if err := httpx.DecodeJSON(w, req, &input); err != nil {
  83 				return
  84 			}
  85 			p, _ := authz.PrincipalFrom(req.Context())
  86 			user, reissued, err := users.Invite(req.Context(), p, input.Email, input.Role, requestMeta(req))
  87 			if writeUserError(w, req, err, logger) {
  88 				return
  89 			}
  90 			status := http.StatusCreated
  91 			if reissued {
  92 				status = http.StatusOK
  93 			}
  94 			writeUsersJSON(w, status, user)
  95 		})
  96 		r.Put("/api/v1/users/{userId}/role", func(w http.ResponseWriter, req *http.Request) {
  97 			id, err := uuid.Parse(chi.URLParam(req, "userId"))
  98 			if err != nil {
  99 				httpx.WriteProblem(w, req, httpx.CodeNotFound)
 100 				return
 101 			}
 102 			var input struct {
 103 				Role authz.Role `json:"role"`
 104 			}
 105 			if err := httpx.DecodeJSON(w, req, &input); err != nil {
 106 				return
 107 			}
 108 			p, _ := authz.PrincipalFrom(req.Context())
 109 			user, err := users.ChangeRole(req.Context(), p, id, input.Role, requestMeta(req))
 110 			if writeUserError(w, req, err, logger) {
 111 				return
 112 			}
 113 			writeUsersJSON(w, http.StatusOK, user)
 114 		})
 115 		r.Post("/api/v1/users/{userId}/deactivate", func(w http.ResponseWriter, req *http.Request) {
 116 			id, err := uuid.Parse(chi.URLParam(req, "userId"))
 117 			if err != nil {
 118 				httpx.WriteProblem(w, req, httpx.CodeNotFound)
 119 				return
 120 			}
 121 			p, _ := authz.PrincipalFrom(req.Context())
 122 			user, err := users.Deactivate(req.Context(), p, id, requestMeta(req))
 123 			if writeUserError(w, req, err, logger) {
 124 				return
 125 			}
 126 			writeUsersJSON(w, http.StatusOK, user)
 127 		})
 128 		inviteQuotas.Post("/api/v1/users/{userId}/reactivate", func(w http.ResponseWriter, req *http.Request) {
 129 			id, err := uuid.Parse(chi.URLParam(req, "userId"))
 130 			if err != nil {
 131 				httpx.WriteProblem(w, req, httpx.CodeNotFound)
 132 				return
 133 			}
 134 			p, _ := authz.PrincipalFrom(req.Context())
 135 			user, err := users.Reactivate(req.Context(), p, id, requestMeta(req))
 136 			if writeUserError(w, req, err, logger) {
 137 				return
 138 			}
 139 			writeUsersJSON(w, http.StatusOK, user)
 140 		})
 141 	})
 142 }

 144 // tenantKey is the company quota key of DD-48. ratelimit cannot import authz, so the key is built
 145 // here, from the Principal that Authenticate injected.
 146 func tenantKey(r *http.Request) (string, bool) {
 147 	p, ok := authz.PrincipalFrom(r.Context())
 148 	if !ok {
 149 		return "", false
 150 	}
 151 	return p.TenantID.String(), true
 152 }

 154 func writeUserError(w http.ResponseWriter, req *http.Request, err error, logger *slog.Logger) bool {
 155 	if err == nil {
 156 		return false
 157 	}
 158 	var invalid *identity.UserValidationError
 159 	var invalidPassword *identity.PasswordValidationError
 160 	switch {
 161 	case errors.Is(err, identity.ErrUnauthenticated):
 162 		identity.ClearSessionCookie(w)
 163 		httpx.WriteProblem(w, req, httpx.CodeUnauthenticated)
 164 	case errors.Is(err, authz.ErrForbidden):
 165 		httpx.WriteProblem(w, req, httpx.CodeForbidden)
 166 	case errors.Is(err, identity.ErrEmailNotVerified):
 167 		httpx.WriteProblem(w, req, httpx.CodeEmailNotVerified)
 168 	case errors.Is(err, identity.ErrUserNotFound):
 169 		httpx.WriteProblem(w, req, httpx.CodeNotFound)
 170 	case errors.Is(err, identity.ErrLastAdmin):
 171 		httpx.WriteProblem(w, req, httpx.CodeLastAdmin)
 172 	case errors.Is(err, identity.ErrInvalidTransition):
 173 		httpx.WriteProblem(w, req, httpx.CodeInvalidState)
 174 	case errors.Is(err, identity.ErrEmailTaken):
 175 		httpx.WriteProblem(w, req, httpx.CodeEmailTaken)
 176 	case errors.Is(err, identity.ErrTokenInvalid):
 177 		httpx.WriteProblem(w, req, httpx.CodeTokenInvalid)
 178 	case errors.As(err, &invalid):
 179 		httpx.ValidationError(w, req, invalid.Fields)
 180 	case errors.As(err, &invalidPassword):
 181 		httpx.ValidationError(w, req, map[string]string{"password": invalidPassword.Code})
 182 	default:
 183 		httpx.WriteDBError(w, req, err, logger)
 184 	}
 185 	return true
 186 }

 188 func writeUsersJSON(w http.ResponseWriter, status int, value any) {
 189 	body, err := json.Marshal(value)
 190 	if err != nil {
 191 		panic(err)
 192 	}
 193 	w.Header().Set("Content-Type", "application/json")
 194 	w.WriteHeader(status)
 195 	_, _ = w.Write(body)
 196 }
