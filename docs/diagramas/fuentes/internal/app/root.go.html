  14 // RootDeps are the four destinations of the root mux (DD-22).
  15 type RootDeps struct {
  16 	API       http.Handler // chi router serving everything under /api/ (INV-22)
  17 	Liveness  http.Handler // GET /healthz
  18 	Readiness http.Handler // GET /readyz
  19 	SPA       http.Handler // everything else; web.NewHandler(dist) in production, a stub in backend tests
  20 }

  22 // CommonMiddleware wraps the whole root mux, so it applies to the API, ops and the SPA alike.
  23 type CommonMiddleware func(http.Handler) http.Handler
  24 
  25 // NewCommonMiddleware builds the middlewares shared by every destination: request id, client IP,
  26 // recover, logging, security headers and cross-origin protection, outermost first (T-B204). local
  27 // omits HSTS (DD-24). trusted is explicit (not variadic, M5 of the PR #8 review): an omitted
  28 // argument used to silently ignore TRUSTED_PROXIES; every caller must now say "nil" on purpose.
  29 func NewCommonMiddleware(logger *slog.Logger, local bool, trusted []netip.Prefix) CommonMiddleware {
  30 	csrf := http.NewCrossOriginProtection()
  31 	csrf.SetDenyHandler(httpx.CSRFDenyHandler(logger))
  32 	chain := []func(http.Handler) http.Handler{
  33 		httpx.RequestID,
  34 		httpx.ClientIP(trusted, logger),
  35 		httpx.Recover(logger),
  36 		httpx.Logging(logger),
  37 		httpx.SecurityHeaders(!local),
  38 		csrf.Handler,
  39 	}
  40 	return func(next http.Handler) http.Handler {
  41 		for _, mw := range slices.Backward(chain) {
  42 			next = mw(next)
  43 		}
  44 		return next
  45 	}
  46 }

  48 // NewRootHandler builds the root mux: /api/ → chi, /healthz and /readyz (GET only) → ops, and
  49 // everything else → the SPA. The SPA is never registered in chi, so an unknown /api/... path is a
  50 // 404 problem+json and never index.html (DD-22, INV-22).
  51 func NewRootHandler(deps RootDeps, common CommonMiddleware) http.Handler {
  52 	for name, h := range map[string]http.Handler{
  53 		"API": deps.API, "Liveness": deps.Liveness, "Readiness": deps.Readiness, "SPA": deps.SPA,
  54 	} {
  55 		if h == nil {
  56 			panic("app: RootDeps." + name + " is nil")
  57 		}
  58 	}
  59 	mux := http.NewServeMux()
  60 	mux.Handle("/api/", deps.API)
  61 	// Not "GET /healthz": with the catch-all "/" below, a POST would match "/" and reach the SPA
  62 	// instead of getting a 405 from the mux. The method check is done here.
  63 	mux.Handle("/healthz", httpx.WithRoute("ops", getOnly(deps.Liveness)))
  64 	mux.Handle("/readyz", httpx.WithRoute("ops", getOnly(deps.Readiness)))
  65 	mux.Handle("/", spaUnlessAPI(deps.API, deps.SPA))
  66 	if common == nil {
  67 		return mux
  68 	}
  69 	return common(mux)
  70 }

  72 // getOnly answers 405 (with Allow) to anything but GET and HEAD.
  73 func getOnly(next http.Handler) http.Handler {
  74 	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
  75 		if r.Method != http.MethodGet && r.Method != http.MethodHead {
  76 			w.Header().Set("Allow", "GET, HEAD")
  77 			http.Error(w, "Method Not Allowed", http.StatusMethodNotAllowed)
  78 			return
  79 		}
  80 		next.ServeHTTP(w, r)
  81 	})
  82 }

  84 // spaUnlessAPI serves the SPA, except for anything that is under /api/ once decoded. The ServeMux
  85 // routes on the escaped path, so "/api%2Fv1%2Fx" would otherwise fall into the SPA's catch-all with
  86 // URL.Path "/api/v1/x" (INV-22): it goes to the API, which answers 404 problem+json.
  87 func spaUnlessAPI(api, spa http.Handler) http.Handler {
  88 	spa = httpx.WithRoute("spa", spa)
  89 	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
  90 		if r.URL.Path == "/api" || strings.HasPrefix(r.URL.Path, "/api/") {
  91 			api.ServeHTTP(w, r)
  92 			return
  93 		}
  94 		spa.ServeHTTP(w, r)
  95 	})
  96 }

  98 // NewAPIRouter returns the chi router of the API with its own 404 and 405 in problem+json.
  99 // Modules register their routes on it; the SPA is never registered here.
 100 func NewAPIRouter() *chi.Mux {
 101 	r := chi.NewRouter()
 102 	r.Use(httpx.NoStore)
 103 	// First middleware: once the request has been routed (or not), report the chi pattern to the
 104 	// request log. It runs on 404 and 405 too, and lets chi build its own route context.
 105 	r.Use(func(next http.Handler) http.Handler {
 106 		return http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
 107 			defer func() {
 108 				if pattern := chi.RouteContext(req.Context()).RoutePattern(); pattern != "" {
 109 					httpx.SetRoute(req, pattern)
 110 				}
 111 			}()
 112 			next.ServeHTTP(w, req)
 113 		})
 114 	})
 115 	r.NotFound(func(w http.ResponseWriter, req *http.Request) {
 116 		httpx.WriteProblem(w, req, httpx.CodeNotFound)
 117 	})
 118 	r.MethodNotAllowed(func(w http.ResponseWriter, req *http.Request) {
 119 		// chi's default 405 sets Allow, a custom handler does not (RFC 9110 requires it).
 120 		for _, m := range []string{http.MethodGet, http.MethodHead, http.MethodPost, http.MethodPut,
 121 			http.MethodPatch, http.MethodDelete, http.MethodOptions} {
 122 			if r.Match(chi.NewRouteContext(), m, req.URL.Path) {
 123 				w.Header().Add("Allow", m)
 124 			}
 125 		}
 126 		httpx.MethodNotAllowed(w, req)
 127 	})
 128 	return r
 129 }
